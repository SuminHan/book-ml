\noindent{\bf 1. 교과목 개요}

\vspace{2mm}


\noindent\fbox{\begin{minipage}{\dimexpr\textwidth-2\fboxsep-2\fboxrule}
\vspace{2mm}
기계학습과 심층학습 교과에서는 데이터로부터 패턴을 학습하는 기계학습의 기본 원리부터 최신 딥러닝 기술까지를 다루는 과목이다. 회귀와 분류 등 고전적인 지도학습 기법, 군집화 등 비지도학습 기법을 익힌 뒤, 신경망 기반의 딥러닝으로 확장하여 합성곱 신경망, 시퀀스 모델, 어텐션과 트랜스포머, 생성모델에 이르는 최신 인공지능 기술의 원리를 학습한다. 학기 중반과 종료 시점에 팀 프로젝트를 통해 배운 이론을 실제 데이터에 적용하고 그 결과를 정직하게 검증하고 보고하는 경험을 쌓는다.

\vspace{3mm}

\noindent{\bf 가. 교과목 목표}

\begin{itemize}[leftmargin=6mm,itemsep=0.8mm,topsep=1mm]

\item 지도학습(회귀, 분류)과 비지도학습(군집화, 차원축소)의 원리를 이해하고 구현하며, 정규화와 교차검증 등 모델 선택과 검증 원칙을 적용한다.

\item 신경망의 학습 원리(역전파)를 이해하여 CNN, RNN, 트랜스포머를 구현하고, 팀 프로젝트를 통해 머신러닝 프로젝트를 정직한 절차로 수행하는 능력을 기른다.

\end{itemize}

\vspace{1mm}

\noindent{\bf 나. 교과목 내용}

\begin{itemize}[leftmargin=6mm,itemsep=1.2mm,topsep=1mm]

\item 회귀와 분류 등 지도학습, 클러스터링 등 비지도학습

\begin{itemize}[leftmargin=5mm,itemsep=0.3mm,topsep=0.5mm]

\item[--] \small 선형회귀, 로지스틱회귀, 나이브베이즈, GDA, kNN 등 기초 알고리즘

\item[--] \small k-means, GMM/EM 등 클러스터링 기법

\end{itemize}

\item SVM, 정규화, 트리 기반 모델(GBDT) 등 모델 선택과 검증

\begin{itemize}[leftmargin=5mm,itemsep=0.3mm,topsep=0.5mm]

\item[--] \small 마진 최대화와 커널 트릭, L1/L2 정규화

\item[--] \small 교차검증과 결정 트리, 랜덤 포레스트, GBDT

\end{itemize}

\item 신경망 기초와 역전파, CNN, 시퀀스 모델

\begin{itemize}[leftmargin=5mm,itemsep=0.3mm,topsep=0.5mm]

\item[--] \small 순전파, 역전파, 활성함수, 초기화, 정규화 기법

\item[--] \small 합성곱 신경망(CNN)과 RNN, LSTM/GRU

\end{itemize}

\item 어텐션과 트랜스포머

\begin{itemize}[leftmargin=5mm,itemsep=0.3mm,topsep=0.5mm]

\end{itemize}

\item 표현학습과 생성모델(GNN, PCA, VAE, GAN, Diffusion 등)

\begin{itemize}[leftmargin=5mm,itemsep=0.3mm,topsep=0.5mm]
