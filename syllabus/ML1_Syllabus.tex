% ---------- 1. Course Descriptions ----------
\syltitle{1. Course Descriptions}
\fbox{\begin{minipage}{\dimexpr\textwidth-2\fboxsep-2\fboxrule}
\vspace{1mm}
\begin{itemize}[leftmargin=5mm,itemsep=1mm]
\item To understand and implement classical supervised learning
  (regression, classification) and unsupervised learning (clustering,
  dimensionality reduction) methods
\item To correctly apply model selection and validation principles
  (regularization, cross-validation) to real data
\item To understand the training principle behind neural networks
  (backpropagation) and design/implement representative deep learning
  architectures such as CNNs, RNNs, and Transformers
\item To understand the basic principles of generative models
\item To carry out the full pipeline of a machine learning project ---
  problem definition, implementation, honest validation, and
  presentation --- as a team
\end{itemize}
\end{minipage}}

B. Evaluation Method
\begin{enumerate}[leftmargin=7mm,itemsep=0.5mm]
\item Absolute evaluation
\item Weekly quiz and homework for each class
\item Midterm exam: written test (2 hrs)
\item Final exam: written test (2 hrs)
\item Team project: two capstone projects (Week 8: classical ML, Week
  16: deep learning), each graded on a written report, a presentation,
  and peer-review. Grading emphasizes the honesty of the experimental
  procedure (baseline comparison, proper train/validation/test
  separation, honest reporting of results) over raw performance.
\end{enumerate}

% ---------- 6. Criteria for Awarding Credits ----------
\syltitle{6. Criteria for Awarding Credits}
\begin{itemize}[leftmargin=5mm,itemsep=1.2mm]
\item \textbf{Pass:} In the case of receiving a score of \textbf{70 or more out of 100} according to the criteria of the evaluation method.
\item \textbf{Academic Excellence Award:} Students with excellent academic achievement are recommended for the Academic Excellence Award and the Principal's Award, and are recorded according to core competencies in the Subject Details. However, students with poor academic attitude or poor attendance are excluded.
\item \textbf{Attendance:} At least \textbf{three-quarters} of the total class hours is required to earn credit.
\item \textbf{Pass/Fail courses:} If the instructor determines that a specific evaluation element is insufficient and requires a separate evaluation, a separate evaluation may be conducted.
\item \textbf{Excused absences:} For official duties or medical reasons, the make-up score for a missed exam shall be calculated as follows: the percentile score corresponding to the student's total score in other assessments of the same course shall be applied to the missed exam, multiplied by a weighting factor (\textbf{100\%} for official absences, \textbf{80\%} for medical absences). If there are no available reference scores, a re-assessment shall be conducted.
\end{itemize}
