\begin{frame}{3.3절 마무리}
  \begin{block}{MDP는 ``지금 한 행동이 미래의 상태 자체를
  바꾼다''는 사실 하나를 Week 2의 밴딧에 추가한 것뿐이지만,
  이 한 가지 차이가 강화학습을 \textbf{훨씬 더 어렵고, 훨씬 더
  흥미로운} 문제로 만든다.}
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item \kb{가치함수} $V^\pi(s), Q^\pi(s,a)$ = 리턴 $G_t$ 의
      \textbf{기댓값} (상태 고정 / 상태+첫 행동 고정).
    \item \kb{벨만 기대방정식} = 마르코프 성질을 ``한 스텝만''
      적용한 결과 --- 무한합을 ``지금 보상 + 할인된 다음 상태
      가치''의 \textbf{재귀식}으로 압축.
    \item \kb{반복 대입(정책평가)} = $\gamma < 1$ 이면 \textbf{고정점
      하나로 수렴}(수축 연산 + 바나흐) --- 다음 장의 알고리즘
      전체의 뼈대.
  \end{itemize}
\end{frame}
