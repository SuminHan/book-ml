\begin{frame}{``계산이 큰 문제''가 아니라 ``성질이 다른 문제''}
  \begin{itemize}
    \item $\pi_2$가 \textbf{정해져 있으면} 에이전트 1의 최적
      정책 $\pi_1^* = \arg\max_{\pi_1} J_1(\pi_1, \pi_2)$도
      \textbf{존재}
    \item 그런데 $\pi_2$가 \textbf{바뀔 때마다} ``최적
      정책''이라는 \textbf{것 자체가 바뀜}
    \item Week 6 Q-learning이 수렴하던 표적 $Q^*(s,a)$는
      \textbf{고정}되어 있었는데, 멀티에이전트에서는 내가
      수렴해야 할 \textbf{``정답''이 상대의 학습과 함께 계속
      이동}
  \end{itemize}
  \vskip0.6em
  \begin{block}{이번 세 실습이 보여줄 것}
    학습 알고리즘은 \textbf{모두 똑같은 단순한 Q-learning /
    fictitious play}이고, 결과가 \textbf{완전히} 달라지는 것은
    오직 \textbf{보상 구조(게임의 규칙)} 때문.
  \end{block}
\end{frame}
