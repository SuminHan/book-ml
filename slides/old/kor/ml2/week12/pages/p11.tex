\begin{frame}{배포 실패: ``시뮬레이션에선 돌아갔는데''}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{시작 $(2,0)$} — 시연 경로를 그대로 따감:
      \vskip0.4em
      {\scriptsize $(2,0)$ up$\to$ $(1,0)$ up$\to$ $(0,0)$
      right$\to$ $(0,1)$ right$\to$ $(0,2)$ right$\to$ $(0,3)$}
      \vskip0.6em
      \textbf{목표 도착 (5스텝)} — \textit{학습 데이터 위에서}
      완벽하다.
      \vskip0.8em
      \textbf{시작 $(1,3)$으로} 변경: 시연에 \textbf{한 번도 등장
      하지 않은} 상태. 기본값 \texttt{right}인데, $(1,3)$에서
      right는 \textbf{벽}이다.
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \fbox{\scriptsize $(1,3)$ right$\to$ $(1,3)$ right$\to$
      $(1,3)$ right$\to$ $\cdots$}
      \vskip1em
      \kq{벽에 부딪히며 제자리를 돌다가 \textbf{타임아웃}.}
      \vskip0.8em
      {\footnotesize 같은 패턴: $(2,2)$도 기본값 right로
      $(2,3)$까지 갔다 벽에 막혀 죽는다.}
    \end{column}
  \end{columns}
  \vskip0.6em
  {\footnotesize 시작이 2칸만 달라졌을 때. 핵심: ``시연 데이터가
  적어서''가 아니라 \textbf{시연이 커버하지 못한 상태 분포} 문제.}
\end{frame}
