\begin{frame}{손으로 한 번: 세 스텝 직접 계산}
  \small
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      아주 작은 데이터셋 $X=[1,2,3]$, $y=[3,5,7]$ (참값 $y=2x+1$),
      $w_0 = w_1 = 0$ 에서 시작, $\alpha = 0.1$.
      \vskip0.3em
      초기 비용:
      \[
      J(0,0) = \tfrac{1}{6}\left[3^2+5^2+7^2\right] = \tfrac{83}{6}
      \approx 13.83
      \]
      그래디언트:
      \[
      \frac{\partial J}{\partial w_0} = -5, \quad
      \frac{\partial J}{\partial w_1} \approx -11.33
      \]
      한 스텝 이동: \\
      $w_0 \leftarrow 0 - 0.1(-5) = 0.5$ \\
      $w_1 \leftarrow 0 - 0.1(-11.33) \approx 1.13$
    \end{column}
