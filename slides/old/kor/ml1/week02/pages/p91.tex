\begin{frame}{자주 묻는 질문 (2): 임계값 0.5는 항상 최선인가}
  \textbf{아니다} --- 임계값은 문제의 성격에 맞춰 조정하는
  \textbf{하이퍼파라미터}.
  \vskip0.6em
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{암 검진}: 양성 놓침 (FN) 비용 $\gg$ 오진 (FP) 비용
      $\to$ 임계값을 0.5보다 \textbf{낮춰서 Recall 올리는 쪽}.
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{스팸 필터}: 정상 메일 오거름 (FP) 비용이 더 부담
      $\to$ 임계값을 \textbf{높여서 Precision 올리는 쪽}.
    \end{column}
  \end{columns}
  \vskip0.6em
