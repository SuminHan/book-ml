\begin{lstlisting}
w0, w1 = 0.0, 0.0
alpha_big = 1.5
for i in range(1, 6):
    g0, g1 = grad(w0, w1, X, y)
    w0 -= alpha_big * g0
    w1 -= alpha_big * g1
    print(f"step {i}: w0={w0:.4f} w1={w1:.4f} cost={cost(w0, w1, X, y):.4f}")
\end{lstlisting}
