\begin{frame}{학습(training)의 표기: $m$ 과 $(i)$}
  \begin{block}{학습의 정의}
    주어진 데이터 $(\boldsymbol{x}^{(1)}, y^{(1)}), \dots, (\boldsymbol{x}^{(m)}, y^{(m)})$ 에 대해
    $h(\boldsymbol{x}^{(i)})$ 가 $y^{(i)}$ 에 최대한 가깝도록 $h$ 의
    \textbf{파라미터를 조정}하는 과정.
  \end{block}
  \vskip0.4em
  \begin{itemize}
    \item 위 첨자 $(i)$ = ``$i$ 번째 데이터 샘플'' (이 학기 내내 반복되는
      표기). $\boldsymbol{x}^{(3)}$ = 세 번째 집의 특징 벡터.
    \item $m$ = 학습 데이터의 \textbf{전체 개수}(샘플 수).
    \item 예: 집 200채면 $m=200$, \; $\boldsymbol{x}^{(37)}$ = 37번째 집의
      (평수, 방 개수, 연식), \; $y^{(37)}$ = 그 집의 실제 거래가.
  \end{itemize}
  \vskip0.4em
  학습은 ``학습 데이터를 \textbf{외우는 것}''이 아니라,
  \textbf{본 적 없는 201번째 집에도 일반화되는 규칙을 찾는 것} ---
  일반화 능력의 엄밀한 검증은 Week 6.
\end{frame}
