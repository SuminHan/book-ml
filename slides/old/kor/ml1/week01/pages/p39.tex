\begin{frame}{세 시대가 주는 메시지}
  \centering
  \vskip1em
  \large\kq{``정의를 먼저 하라''는 매핑($x \to y$)의 어느 부분을
  사람이 쓰고, 어느 부분을 기계가 배우게 할 것인가를 정한다는 뜻.}
  \vskip1.5em
  \begin{flushleft}
  \small
  \begin{itemize}
    \item ``평가함수를 쓴다'' $\to$ ``학습한다'' (1959, 사무엘)
    \item ``인식 규칙을 쓴다'' $\to$ ``특징을 학습한다'' (1998, LeNet)
    \item ``정의를 바꾸는'' 순간 불가능했던 문제가 가능해졌다.
  \end{itemize}
  \end{flushleft}
\end{frame}
