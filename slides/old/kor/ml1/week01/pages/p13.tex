\begin{frame}{두 갈래의 역사 --- 신경망}
  \small
  \begin{itemize}
    \item 흥미로운 점: 역사가 직선으로 발전한 게 아니라
      \kb{침체기(``AI 겨울'')를 두 번이나 거쳤다}는 것.
  \end{itemize}
  \vskip0.4em
  \centering{\textbf{신경망 --- 퍼셉트론 $\to$ LLM}}\\[0.4em]
  \begin{tabular}{@{}ll@{}}
    \toprule
    연도 & 사건 \\
    \midrule
    1943 & 맥컬록-피터즈 신경원 모형 \\
    1950s & 퍼셉트론(단층 신경망) 등장 \\
    1969 & XOR 한계 증명 $\to$ \textbf{첫 AI 겨울} \\
    1986 & 역전파 재발견 $\to$ 다층 학습 가능 \\
    1990s$\sim$2000s & SVM$\cdot$커널기법에 밀려 \textbf{두 번째 AI 겨울} \\
    1998 & (겨울 중에도) LeCun의 CNN(LeNet) \\
    2012 & AlexNet --- \textbf{딥러닝 시대의 시작} \\
    2013 & word2vec$\cdot$VAE \\
    2016 & AlphaGo(이세돌) --- 강화학습의 상징적 승리 \\
    2017 & Transformer \\
    2022 & ChatGPT(사전학습+RLHF) \\
    2025 & 멀티모달 LLM 대중화 \\
    \bottomrule
  \end{tabular}
\end{frame}
