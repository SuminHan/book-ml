\begin{frame}{공분산 기준 vs 상관 기준}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \kb{공분산 기준 (기본)}
      \begin{itemize}
        \item 공분산 행렬 $\Sigma$의 고유분해.
        \item \texttt{sklearn} \texttt{PCA}의 기본 --- 중심화만 해주고,
          \textbf{스케일링은 안 해줌}.
        \item ``수치가 큰 축''이 지배함.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \kb{상관 기준 (표준화 후)}
      \begin{itemize}
        \item 특징을 \textbf{먼저 표준화}한 뒤 PCA.
        \item 고유분해의 대상이 공분산이 아니라
          \kb{상관행렬}(correlation matrix).
        \item 각 축이 동등한 무게를 가짐.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip1em
  \kq{단위가 다른 특징이라면, 표준화 없이 돌리는 것은 버그가
  아니라 버그의 원인이다.}
  \; 두 기준이 언제, 얼마나 달라지는지는 앞의 키·몸무게 예가 그대로.
\end{frame}
