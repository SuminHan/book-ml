\section{1.1 아주 짧은 역사와 이번 학기 로드맵}

\begin{frame}{이 두 과목이 다루는 질문}
  2012년 ImageNet 대회에서 \kb{AlexNet} 이 등장하기 전까지, 컴퓨터에게
  ``이 사진에 고양이가 있는가''를 알려주는 방법은 \textbf{사람이 직접 특징을
  설계}하는 것이었다 --- 가장자리 검출기, 색 히스토그램, 손으로 조합한 필터.
  \vskip0.8em
  AlexNet은 그런 손설계 없이, 수백만 장의 사진과 정답 라벨만 보고
  \textbf{스스로 무엇을 봐야 하는지 학습}해서 기존 방법을 압도했다.
  \vskip1em
  \begin{center}
  \kq{정답이 있는 데이터로부터, 규칙을 사람이 짜는 대신\\
      기계가 데이터에서 직접 찾아내게 하려면 어떻게 해야 하는가?}
  \end{center}
\end{frame}

  \begin{itemize}

    \item \kb{2012 — 딥러닝 시대의 개막.} AlexNet은 새 수학이 아니었다.

\section{1.2 머신러닝 문제 정식화와 세 갈래 분류}

\begin{frame}{문제를 푸는 게 아니라, 먼저 ``정의''부터}
  알고리즘을 고르기 전에 문제 자체를 세 가지로 정의한다:
  \begin{enumerate}
    \item \kb{입력 $x$} --- 무엇을 특징으로? (예: 평수, 방 개수, 연식)
    \item \kb{출력 $y$} --- 무엇을 맞히나? 연속값(회귀)인가 범주(분류)인가?
    \item \kb{가설 $h$} --- 입력$\to$출력 함수를 어떤 형태로 근사하나?
      (직선? 트리? 신경망?)
  \end{enumerate}
  \vskip0.8em
  \begin{block}{학습(training)의 정의}
    데이터 $(x^{(1)},y^{(1)}),\dots,(x^{(m)},y^{(m)})$ 에 대해
    $h(x^{(i)})$ 가 $y^{(i)}$ 에 최대한 가깝도록 $h$ 의 파라미터를 조정.
    \\ $m$ = 샘플 수, \; $x^{(i)}$ = $i$번째 샘플의 특징 벡터.
  \end{block}
\end{frame}

\begin{frame}{같은 틀, 다른 손실 --- 그리고 일반화}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{집값 예측}: $x$=평수·방·위치, \; $y$=거래가(\textbf{연속}) $\Rightarrow$ 회귀 \\[0.6em]
      \textbf{스팸 필터}: $x$=제목의 `무료'·발신자·느낌표 수, \;
      $y$=스팸(1)/정상(0) (\textbf{범주}) $\Rightarrow$ 분류 \\[0.8em]
      $x,y$ 의 형태가 다르니 \textbf{손실함수도 달라야} 한다.
      \\[0.6em]
      학습은 ``데이터를 외우는 것''이 아니라
      \textbf{본 적 없는 데이터에 일반화되는 규칙}을 찾는 것.
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch01_overfitting.png}
      \\[0.3em]{\footnotesize degree 1 vs degree 24 --- 오른쪽은 훈련점 사이
      잡음까지 외워 테스트에서 흔들린다.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{세 갈래 분류 --- 무엇을 데이터로 갖고 있는가}
  알고리즘 이름이 아니라 \textbf{데이터의 형태 · 학습 신호}로 갈린다.
  \vskip0.6em
  \small
  \begin{tabular}{@{}llll@{}}
    \toprule
     & 데이터 형태 & 목표 & 이번 학기 예시 \\
    \midrule
    지도학습 & $(x,y)$ 쌍 & 새 $x$의 $y$ 예측 & 회귀·분류, kNN, SVM, 트리, 신경망 \\
    비지도학습 & $x$ 만 & 숨은 구조 발견 & k-means, PCA, EM/GMM \\
    강화학습 & 상태·행동·보상 & 누적 보상 최대화 정책 & ML1에서 다루지 않음 (ML2) \\
    \bottomrule
  \end{tabular}
  \vskip1em
  \begin{block}{한 줄로 구분하는 두 질문}
    \begin{enumerate}
      \item ``$y$(정답 라벨)이 있는가?'' \; 있다 $\to$ \textbf{지도}. 없다 $\to$ 2번.
      \item ``보상이라는 신호라도 있는가?'' \; 있다 $\to$ \textbf{강화}. 없다 $\to$ \textbf{비지도}.
    \end{enumerate}
  \end{block}
\end{frame}

\section{1.3 데이터사이언스 파이프라인과 사전지식 리뷰}

\begin{frame}{모델은 파이프라인의 한 조각일 뿐}
  \begin{enumerate}
    \item \kb{수집} --- 어디서? (Kaggle, UCI, HF Datasets) + 라이선스·개인정보 확인
    \item \kb{정제} --- 결측치 · 중복 · 이상치 처리 (현실 데이터는 거의 항상 지저분)
    \item \kb{EDA} --- 분포 · 상관 · 클래스 비율을 \textbf{눈으로} 확인
    \item \kb{모델링} --- 정식화한 문제에 맞는 $h$ 선택 · 학습
    \item \kb{평가} --- 학습에 안 쓴 데이터로 검증. 나쁘면 종종 1$\sim$3단계로 되돌아감
  \end{enumerate}
  \vskip0.8em
  \begin{block}{왜 이 순서인가}
    앞 단계의 결함은 뒤에서 아무리 좋은 모델을 써도 \textbf{수리되지 않는다}.
    파이프라인은 위$\to$아래로 흐르되, \textbf{결함도 위$\to$아래로 증폭}된다.
  \end{block}
\end{frame}

\begin{frame}{더티 데이터의 세 가지 표준 모양}
  \begin{enumerate}
    \item \kb{결측치} --- 빈 값, 혹은 0 / $-1$ / $-999$ 같은 \textbf{센티넬 값}.
      센티넬은 ``비어 보이지 않아'' 가장 위험 $\to$ \texttt{describe()} 의
      최솟값 · 분포 확인.
    \item \kb{중복} --- 완전 중복은 \texttt{duplicated()} 로 잡히지만,
      부분 중복은 ``어떤 열 조합을 동일 사례로 볼지'' 사람이 먼저 결정
      (예: 부동산 = 지번+거래일).
    \item \kb{이상치} --- 범위는 합리적이나 분포에서 극단적으로 벗어난 값.
      버릴지 남길지는 \textbf{도메인 지식}으로만 판단.
  \end{enumerate}
  \vskip0.6em
  {\footnotesize 1992년 Wolpert \& Kibler, ``Do We Know the Data We're Using?''
  --- 같은 특징에 라벨이 모순되는 행 100건. \kq{성능이 이상할 때 첫 의심은
  ``모델이 약해서''가 아니라 ``데이터에 문제가 있어서''.}}
\end{frame}

\begin{frame}[fragile]{EDA: 모델을 돌리기 전에 데이터와 ``대화''하기}
  네 가지 질문 --- 모두 \textbf{모델이 아니라 데이터}에 묻는다:
  스케일 차이 / 목표와의 연관 / 목표 분포의 꼬리 / 클래스 균형.
\begin{lstlisting}
import pandas as pd
df = pd.DataFrame({
    "size_sqm":     [59, 84, 112, 46, 132, 78],
    "rooms":        [2, 3, 4, 1, 4, 3],
    "year_built":   [2015, 2008, 1998, 2020, 1995, 2011],
    "price_million":[320, 410, 505, 250, 560, 390],
})
print(df.describe())                                  # 스케일 한눈에
print(df.corr(numeric_only=True)["price_million"])    # 목표와의 연관
print(df.isna().sum())                                # 열별 결측 개수
\end{lstlisting}
  \vskip0.2em
  {\footnotesize EDA에 쓰는 10분이, 잘못된 가정 위에 쌓은 모델을 디버깅하는
  몇 시간을 아낀다.}
\end{frame}

\begin{frame}{데이터 나누기 --- 순서가 중요한 이유}
  EDA가 끝나면 모델링 전 \textbf{첫 실제 결정}: 훈련 셋 / 테스트 셋 분리.
  \vskip1em
  \begin{block}{실무 원칙 (데이터 누수 방지)}
    데이터에 대한 통계를 쓰는 모든 조작은 \\[0.3em]
    \hspace{1em}\kb{훈련 셋에서 fit(계산)} $\;\to\;$ \kb{테스트 셋에 apply(적용)} \\[0.3em]
    순서로. 결코 ``전체에서 fit $\to$ 훈련 셋에 apply'' 하지 않는다.
  \end{block}
  \vskip0.8em
  예: 결측치를 평균으로 채울 때 \textbf{전체 평균}을 쓰면 테스트 셋 정보가
  훈련에 새어든다 --- 훈련 때만 비현실적으로 정확하고 실전에서 무너지는,
  디버깅이 거의 불가능한 모델이 된다. (엄밀한 이유는 Ch.\,6.)
\end{frame}
