\begin{frame}{점을 세어 보면: 비율은 S 자}
\markpage{rate}
\vfill
\centering\includegraphics[width=0.54\linewidth]{chess_rate.png}\par\vskip0.2em
{\large 100점 구간마다 백 승률: $-500$ 점이면 10\%, 비슷하면 반반, $+500$ 점이면 92\%}\\[0.3em]
{\large 예측할 것은 0 / 1 이 아니라 \kb{이 비율 = 확률} \quad 0 과 1 사이에서 \kb{S 자}로 눕는다}\\[0.2em]
{\small 차이가 클수록 승률이 오르지만, 아무리 강해도 100\% 는 못 넘는다: 직선은 끝에서 빗나간다}
\vfill
\note{점 하나하나는 0 아니면 1이라 패턴이 잘 안 보입니다. 그래서 레이팅 차이 100점 구간마다 백이 몇 퍼센트 이겼는지 세어 봤습니다. 백이 500점 낮으면 10퍼센트, 비슷하면 반반, 500점 높으면 92퍼센트를 이깁니다. 차이가 클수록 승률이 오르지만, 아무리 실력 차이가 커도 100퍼센트를 넘을 수는 없으니 양 끝에서 눕습니다. 그래서 비율은 S 자입니다. 우리가 예측할 것은 0이나 1이 아니라 바로 이 비율, 확률입니다. 가운데에서는 직선도 괜찮지만, 끝으로 가면 직선은 0과 1 밖으로 빗나갑니다. 그러면 S 자 곡선을 쓰면 되겠죠.}
\end{frame}

\section{모델}
% --- 시그모이드
\begin{frame}{시그모이드: 아무 수나 0과 1 사이로}
\markpage{sigmoid}
\vfill
\begin{columns}[c]
\begin{column}{0.55\textwidth}\centering\includegraphics[width=\linewidth]{logit_sigmoid.png}\end{column}
\begin{column}{0.42\textwidth}
{\LARGE\[ \sigma(z) = \frac{1}{1 + e^{-z}} \]}\vskip0.4em
$z$ 가 크면 $\to 1$, \quad $z$ 가 작으면 $\to 0$\\[0.3em]
$z = 0$ 이면 정확히 $0.5$
\end{column}
\end{columns}
\vfill
\note{그래서 직선의 값을 시그모이드라는 함수에 한 번 통과시킵니다. 시그모이드는 어떤 수를 넣어도 0과 1 사이 값을 내놓습니다. 큰 수를 넣으면 1에 가까워지고, 작은 수를 넣으면 0에 가까워지고, 0을 넣으면 정확히 0.5가 나옵니다. 2를 넣으면 0.88, 마이너스 2를 넣으면 0.12입니다.}
\end{frame}

% --- 모델
\begin{frame}{로지스틱회귀 모델}
\vfill
\centering
{\LARGE\[ \hat y = \sigma(\underbrace{w_0 + w_1 x}_{z}) = P(\text{white wins} \mid x) \]}\vskip0.6em
{\large 선형회귀의 직선 $z$ 를 그대로 쓰고, 마지막에 $\sigma$ 만 씌운다 \quad $\to$ S 자 곡선}\\[0.6em]
{\large $x$ = 레이팅 차이 (백 $-$ 흑) \quad 출력 $\hat y$ = \kb{백이 이길 확률}}\\[0.4em]
{\small 이름은 ``회귀''지만 하는 일은 \kb{분류} \quad 수식에서는 백 승을 `white wins' 로 쓴다}
\vfill
\note{로지스틱회귀는 선형회귀의 직선 z를 그대로 계산하고, 마지막에 시그모이드만 씌운 모델입니다. 직선을 시그모이드에 넣으니 결과가 S 자 곡선이 되고, 0과 1 밖으로 나가지 않습니다. 입력 x는 레이팅 차이, 출력 와이햇은 백이 이길 확률로 읽습니다. 이름에 회귀가 들어가 있지만, 확률을 보고 누가 이길지 나누니까 실제로 하는 일은 분류입니다.}
\end{frame}

% --- 학습 결과
\begin{frame}{12만 판으로 학습하면}
\markpage{fit}
\vfill
\centering\includegraphics[width=0.56\linewidth]{chess_fit.png}\par\vskip0.3em
{\large $\hat y = \sigma(0.09 + 0.0047\,x)$ \qquad \kq{백이 200점 높으면 $\to$ 백 승률 74\%}}\\[0.2em]
{\small 레이팅을 정하는 \textbf{Elo 공식} $\frac{1}{1 + 10^{-x/400}} = \sigma(0.0058\,x)$ 도 같은 꼴: 레이팅 시스템이 곧 로지스틱회귀}\\[0.1em]
{\scriptsize\color{gray} Elo: 이 방식을 만든 아르파드 엘로 (Arpad Elo) 의 이름, 약자가 아니다. 레이팅 숫자를 ``Elo 점수''라고도 부른다}
\vfill
\note{12만 판으로 학습하면 w0는 0.09, w1은 0.0047이 나옵니다. 파란 곡선이 레이팅 차이에 따른 백 승률이고, 회색 점은 앞에서 센 구간별 실제 비율입니다. 숫자 두 개짜리 곡선이 12만 판의 비율을 거의 그대로 따라갑니다. 200점 높으면 74퍼센트를 이깁니다. 재미있는 건 아래 줄입니다. 체스의 레이팅을 계산하는 Elo 공식은, 참고로 Elo는 약자가 아니라 이 방식을 만든 헝가리 출신 물리학 교수이자 체스 선수 아르파드 엘로의 이름입니다. 그래서 레이팅 숫자 자체를 Elo 점수라고도 부릅니다. 이 공식은 1 나누기 1 더하기 10의 마이너스 x 나누기 400제곱인데, 이걸 정리하면 시그모이드 0.0058 x, 바로 로지스틱 곡선입니다. 체스, 바둑, 게임 랭크 점수는 처음부터 로지스틱회귀의 모양으로 설계된 겁니다. 데이터에서 배운 기울기 0.0047이 공식의 0.0058과 비슷하게 나왔습니다.}
\end{frame}

\section{결정 경계}
% --- 결정 경계
\begin{frame}{결정 경계: $z = 0$ 인 곳}
\vfill
\centering
{\Large\[ \hat y \ge 0.5 \;\Longleftrightarrow\; z = w_0 + w_1 x \ge 0 \;\Longleftrightarrow\; x \ge -\frac{w_0}{w_1} = -\frac{0.09}{0.0047} \approx -20 \]}\vskip0.8em
{\large 백은 \kb{20점 약해도} 반반: 먼저 두는 쪽이 유리하다 \quad (레이팅이 같으면 $\sigma(0.09) \approx$ \kb{52\%})}\\[0.5em]
{\large 시그모이드는 $z=0$ 에서 0.5 \pref{sigmoid} $\to$ \kb{경계는 직선 $z = 0$ 이 정한다}}\\[0.5em]
{\small 절편 $w_0$ 에도 뜻이 있다: 실력이 같을 때의 백의 이점}
\vfill
\note{확률이 0.5 이상이면 백 승이라고 판정할 때, 그 경계가 어디인지 식으로 구해 봅시다. 시그모이드는 z가 0일 때 0.5니까, 와이햇이 0.5 이상인 것은 z가 0 이상인 것과 같습니다. z는 w0 더하기 w1 x이니까 x가 마이너스 0.09 나누기 0.0047, 약 마이너스 20점 이상이면 됩니다. 백은 20점 약해도 반반이라는 뜻입니다. 체스는 백이 먼저 두니까 조금 유리하죠. 레이팅이 같을 때 백의 승률은 시그모이드 0.09, 약 52퍼센트입니다. 이렇게 절편 w0에도 뜻이 있습니다.}
\end{frame}

% --- 두 특징
\begin{frame}{특징이 둘이면: 경계는 직선}
\vfill
\begin{columns}[c]
\column{0.44\linewidth}\centering
\includegraphics[width=\linewidth]{chess_2d.png}
\column{0.54\linewidth}
{\large 특징을 백 레이팅 $a$, 흑 레이팅 $b$ 따로 넣으면}\\[0.3em]
{\large $z = 0.15 + 0.00466\,a - 0.00470\,b$}\\[0.8em]
{\large 경계 $z = 0$ 은 평면 위의 \kb{직선} \quad $a \approx b - 32$}\\[0.3em]
{\large 두 계수가 \kb{크기는 같고 부호만 반대}}\\[0.2em]
{\large $\to$ 모델이 스스로 ``\textbf{차이}만 중요하다''를 찾았다}\\[0.6em]
{\small 1,000 점끼리든 2,000 점끼리든 차이가 같으면 승률도 같다 (Elo 의 생각)}
\end{columns}
\vfill
\note{특징이 두 개면 어떨까요? 이번에는 차이를 미리 빼지 않고, 백 레이팅 a와 흑 레이팅 b를 따로 넣어 봅니다. 파란 점이 백 승, 빨간 점이 흑 승입니다. 학습하면 z는 0.15 더하기 0.00466 a 빼기 0.00470 b이고, z가 0인 곳이 경계니까 평면 위의 직선, 거의 a가 b 빼기 32인 대각선입니다. 두 계수가 크기는 거의 같고 부호만 반대입니다. 따로 넣어 줬는데도 모델이 스스로 둘의 차이만 중요하다는 걸 찾아낸 겁니다. 1000점끼리 두든 2000점끼리 두든, 차이가 같으면 승률이 같다. 이게 Elo 레이팅의 생각입니다. 특징이 더 많아지면 경계는 평면, 초평면이 되지만 여전히 곧습니다. 그래서 로지스틱회귀를 선형 분류기라고 부릅니다.}
\end{frame}

% --- 오즈
\begin{frame}{$w_1$ 을 읽는 법: 오즈}
\vfill
\centering
{\Large\[ z = \ln \frac{p}{1-p} \qquad \text{odds} = \frac{p}{1-p} = \frac{\text{백 승 확률}}{\text{흑 승 확률}} \]}\vskip0.6em
{\large 100점 높으면 $z$ 가 $100\,w_1$ 만큼 커진다 $\to$ 오즈가 $e^{0.47} \approx$ \kq{1.6배}}\\[0.6em]
{\large Elo 공식은 \textbf{400점마다 오즈 10배}로 정해 둔 것 ($10^{x/400}$) \quad 데이터에서는 6.5배}\\[0.6em]
{\small 예: 반반 (오즈 1) $\to$ 100점 높으면 오즈 1.6, 확률 $1.6/2.6 \approx 62\%$}
\vfill
\note{w1은 어떻게 해석할까요? 시그모이드를 거꾸로 풀면 z는 로그 오즈가 됩니다. 오즈는 백이 이길 확률을 흑이 이길 확률로 나눈 값으로, 내기에서 몇 대 몇이라고 할 때 쓰는 그 비율입니다. 레이팅이 100점 높으면 z가 0.47만큼 커지니까, 오즈는 e의 0.47제곱, 약 1.6배가 됩니다. Elo 공식은 아예 400점 차이마다 오즈가 10배가 되도록 정해 둔 겁니다. 이 데이터에서 400점이면 오즈 6.5배로, 공식보다 조금 덜 가파릅니다. 예를 들어 반반이던 대국에서 100점 높은 사람이 두면 오즈가 1에서 1.6으로, 확률로는 약 62퍼센트가 됩니다. 계수 하나를 오즈 배수로 읽을 수 있다는 것이 로지스틱회귀의 큰 장점입니다.}
\end{frame}

\end{columns}
\vfill
\note{벌점은 남은 칩 비율에 마이너스 로그 2를 씌운 값입니다. 다 남으면 0점, 절반이 남으면 1점, 4분의 1이 남으면 2점입니다. 칩이 절반으로 줄 때마다 벌점이 1씩 오릅니다. 그래서 90개 남은 A는 0.15점, 1개 남은 C는 6.64점입니다. 100개에서 1개가 되려면 절반을 6.6번 해야 하니까요. 처음 질문의 답이 6.64점입니다. 한 가지 짚어 두면, 로그의 밑이 꼭 2일 필요는 없습니다. 밑을 바꾸면 값 전체에 같은 상수가 곱해질 뿐이라 어느 AI가 더 나쁜지 순서도, 학습 결과도 같습니다. 밑이 2이면 단위를 비트라고 부르고, 절반마다 1점이라 직관적이어서 오늘 비유에서는 2를 씁니다. 실제 머신러닝 코드는 보통 자연로그를 쓰고, 단위는 냇이라고 합니다.}
\end{frame}

\begin{frame}{정답에 0개를 걸었다면}
\vfill
\centering\includegraphics[width=0.8\linewidth]{ce_allin.png}\par\vskip0.6em
{\large 정답에 하나도 안 걸면 \kq{파산: 벌점 $\infty$}. 모를 땐 반반이 낫다 (틀려도 1점)}
\vfill
\note{정답인 고양이에 하나도 안 걸고 강아지에 100개를 다 걸었다면 칩이 하나도 안 남습니다. 파산이고, 로그 0은 마이너스 무한대라서 벌점은 무한대입니다. 그래서 교차 엔트로피로 학습한 모델은 모를 때 함부로 확신하지 않습니다. 반반에 걸면 어느 쪽이 정답이든 1점만 잃기 때문입니다. 확신하고 틀리는 것을 아주 크게 벌하는 벌점입니다.}
\end{frame}

\begin{frame}{로지스틱회귀의 손실: 교차 엔트로피}
\markpage{celoss}
\vfill
\centering
{\Large\[ L = -\big[\, y \ln \hat y + (1-y) \ln (1-\hat y) \,\big] \]}\vskip0.4em
$y$: 정답 (1 또는 0) \qquad $\hat y = \sigma(z)$: 1일 확률 (시그모이드 출력)\\[0.8em]
\begin{tabular}{ll}
$y = 1$ 이면 & $L = -\ln \hat y$ \quad (1에 건 칩)\\[0.3em]
$y = 0$ 이면 & $L = -\ln (1-\hat y)$ \quad (0에 건 칩)
\end{tabular}\vskip0.8em
\kb{실제 코드는 $\log_2$ 대신 자연로그 $\ln$} \quad {\small 6.64 비트 $= 6.64 \times \ln 2 \approx 4.60$ (값만 상수배)}
\vfill
\note{로지스틱회귀에서 쓰는 손실이 바로 이 칩 걸기 벌점, 교차 엔트로피입니다. 답이 두 개뿐이라 y가 1이면 앞 항만, y가 0이면 뒤 항만 남습니다. 결국 정답 쪽에 건 칩에 마이너스 로그를 씌운 것과 같습니다. 한 가지 차이는 로그의 밑입니다. 비유에서는 절반마다 1점이 되도록 로그 2를 썼지만, PyTorch나 scikit-learn은 자연로그를 씁니다. 값이 상수배만 다르고 최솟값의 위치는 같습니다.}
\end{frame}

% --- 데이터 전체의 손실
\begin{frame}{12만 판 전체의 손실: 평균}
\vfill
\centering
{\Large\[ J(\boldsymbol w) = -\frac{1}{m}\sum_{i=1}^{m} \Big[\, y^{(i)} \ln \hat y^{(i)} + (1-y^{(i)}) \ln (1-\hat y^{(i)}) \,\Big] \]}\vskip0.6em
{\large 판마다 \kb{정답 쪽에 건 확률}에 $-\ln$, 그리고 평균}\\[0.5em]
{\small 예: 백이 이긴 판 ($y=1$) 에 $\hat y = 0.74$ 였다면 $-\ln 0.74 = 0.30$, \; 졌다면 ($y = 0$) $-\ln 0.26 = 1.35$}
\vfill
\note{데이터 전체의 손실은 판마다 벌점을 구해서 평균 낸 것입니다. 정답이 1이면 1에 건 확률, 0이면 0에 건 확률에 마이너스 로그를 씌웁니다. 예를 들어 모델이 백 승률을 0.74로 봤는데 백이 이겼으면 벌점은 마이너스 로그 0.74, 0.30입니다. 졌다면 0에 건 확률 0.26에 마이너스 로그, 1.35입니다. 이 평균 J를 가장 작게 만드는 w를 경사하강법으로 찾습니다.}
\end{frame}

\begin{frame}{왜 오차 제곱(MSE)이 아닐까: 벌점의 크기}
\vfill
\centering
{\large
\begin{tabular}{lcc}
 & \textbf{MSE} $(1-\hat y)^2$ & \textbf{교차 엔트로피} $-\ln \hat y$\\[0.3em]\hline\\[-0.6em]
정답에 0.9 & 0.01 & 0.11\\[0.3em]
반반 0.5 & 0.25 & 0.69\\[0.3em]
\kq{확신하고 틀림 0.01} & \kq{0.98} & \kq{4.60}\\[0.3em]
정답에 0 & 1 (최대) & $\infty$
\end{tabular}}\vskip1.0em
\kq{MSE는 아무리 크게 틀려도 벌점이 1을 넘지 못한다}
\vfill
\note{회귀에서 쓰던 오차 제곱을 분류에도 쓰면 안 될까요? 정답이 1이고 확률 예측을 와이햇라 하면 MSE는 1 빼기 와이햇의 제곱입니다. 확률은 0과 1 사이라서 아무리 크게 틀려도 벌점이 1을 넘지 못합니다. 강아지 99퍼센트라고 확신하고 틀린 AI도 0.98점밖에 안 받습니다. 교차 엔트로피는 4.6점, 정답에 0을 줬다면 무한대입니다. 확신하고 틀린 것을 MSE는 충분히 벌하지 못합니다.}
\end{frame}

\begin{frame}{더 큰 이유: 기울기}
\markpage{gradce}
\vfill
{\small 정답 $y=1$, \; $\hat y = \sigma(z)$, \; $\dfrac{\partial \hat y}{\partial z} = \hat y(1-\hat y)$ \; (시그모이드의 미분)}\vskip0.6em
\centering
{\large
\begin{align*}
\text{교차 엔트로피:}\quad \frac{\partial L}{\partial z} &= -\frac{1}{\hat y}\cdot \hat y(1-\hat y) = \boldsymbol{\textcolor{ksablue}{-(1-\hat y)}} && \text{약분}\\[0.4em]
\text{MSE:}\quad \frac{\partial L}{\partial z} &= -2(1-\hat y)\cdot \hat y(1-\hat y) && \text{그대로}
\end{align*}}\vskip0.4em
\kq{확신하고 틀려서 $\hat y \approx 0$ 이면, MSE의 기울기는 $\hat y$ 가 곱해져 0에 가깝다}
\vfill
\note{더 큰 이유는 기울기입니다. 출력 와이햇는 z에 시그모이드를 씌운 값이고, 시그모이드의 미분은 와이햇 곱하기 1 빼기 와이햇입니다. 교차 엔트로피는 로그를 미분한 1/와이햇가 시그모이드 미분의 와이햇와 약분되어서, 기울기가 깔끔하게 마이너스 1 빼기 와이햇가 됩니다. 크게 틀릴수록 기울기도 큽니다. MSE는 약분이 안 되고 와이햇가 곱해진 채로 남습니다. 그래서 확신하고 틀려서 와이햇가 0에 가까우면 기울기도 0에 가까워지고, 가장 많이 고쳐야 할 때 학습이 거의 멈춥니다.}
\end{frame}

\begin{frame}{그림으로: 손실과 기울기}
\vfill
\centering\includegraphics[width=0.92\linewidth]{ce_vs_mse.png}\par\vskip0.3em
{\small 오른쪽: 교차 엔트로피는 틀릴수록 기울기가 크고, MSE는 확신하고 틀린 왼쪽 끝에서 기울기가 사라진다 \pref{gradce}}
\vfill
\note{그림으로 보면 차이가 분명합니다. 왼쪽 손실을 보면 교차 엔트로피는 정답 확률이 0에 가까워질수록 끝없이 커지고, MSE는 1에서 멈춥니다. 오른쪽 기울기를 보면 교차 엔트로피는 틀릴수록 기울기가 커지는 직선입니다. MSE는 가운데서 잠깐 커졌다가, 가장 크게 틀린 왼쪽 끝에서 0으로 떨어집니다. 실습에서 이 차이를 실제 데이터로 확인합니다.}
\end{frame}

% --- 기울기
\begin{frame}{기울기: 선형회귀와 똑같은 모양}
\markpage{grad}
\vfill
\centering
{\Large\[ \frac{\partial J}{\partial w_j} = \frac{1}{m}\sum_{i=1}^{m} \big(\hat y^{(i)} - y^{(i)}\big)\, x_j^{(i)} \]}\vskip0.6em
{\large 선형회귀: $\hat y = z$ \qquad 로지스틱회귀: $\hat y = \sigma(z)$}\\[0.4em]
{\large \kb{``오차 $\times$ 입력의 평균''}이라는 모양은 같다 $\to$ 경사하강법 코드도 한 줄만 다르다}\\[0.6em]
{\small 시그모이드의 미분 $\hat y(1-\hat y)$ 이 교차 엔트로피의 $1/\hat y$, $1/(1-\hat y)$ 과 약분되어 생긴 결과}
\vfill
\note{놀라운 점이 하나 있습니다. 교차 엔트로피를 w로 미분하면, 오차 와이햇 빼기 y에 입력을 곱해서 평균 낸 값이 나옵니다. 선형회귀의 기울기와 모양이 똑같습니다. 다른 점은 와이햇이 직선 값이냐, 시그모이드를 씌운 값이냐 하나뿐입니다. 시그모이드의 미분이 교차 엔트로피의 분모와 약분되어서 생긴 결과입니다. 그래서 선형회귀 경사하강법 코드에서 예측하는 줄 하나만 바꾸면 로지스틱회귀가 됩니다. 실습에서 직접 해 봅니다.}
\end{frame}
