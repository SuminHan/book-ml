\section{셀프 어텐션}
% --- RNN vs 트랜스포머
\begin{frame}{RNN 은 차례로, 트랜스포머는 한 번에}
\markpage{seq}
\vfill
\centering\includegraphics[width=0.9\linewidth]{tf_seq.png}\par\vskip0.3em
\renewcommand{\arraystretch}{1.3}
{\large\begin{tabular}{lcc}
 & RNN & 트랜스포머 (셀프 어텐션)\\\hline
단어 $n$ 개를 읽는 차례 & $n$ 걸음 (앞이 끝나야 뒤) & \kb{1} 걸음 (모두 동시에, GPU)\\
첫 단어 $\leftrightarrow$ 끝 단어 & $n - 1$ 걸음 거쳐 전달 & \kb{바로} 연결\\
계산량 & $n$ 에 비례 & 모든 쌍: $n^2$ 에 비례
\end{tabular}}
\vfill
\note{RNN은 단어를 하나씩 차례로 읽습니다. 앞 단어를 다 계산해야 다음 단어로 갈 수 있으니, 다섯 단어면 다섯 걸음입니다. 첫 단어의 정보가 끝 단어에 가려면 중간 칸을 모두 거쳐야 해서, LSTM을 써도 멀면 흐려집니다. 트랜스포머는 셀프 어텐션으로 모든 단어 쌍을 한 번에 봅니다. 순서를 기다릴 필요가 없으니 GPU로 모든 칸을 동시에 계산하고, 첫 단어와 끝 단어도 바로 연결됩니다. 대신 모든 쌍을 보니 계산량이 단어 수의 제곱에 비례합니다. 긴 글이 비싼 이유입니다.}
\end{frame}

% --- 셀프 어텐션 + 멀티헤드
\begin{frame}{셀프 어텐션, 여러 개의 머리로}
\vfill
\centering
{\large 단어 벡터 $X$ ($n$ × 512) 에서 \quad $Q = X W_Q, \ K = X W_K, \ V = X W_V$}\\[0.3em]
{\Large\[ \text{head} = \text{softmax}\!\left(\frac{Q K^\top}{\sqrt{64}}\right) V \]}\vskip0.2em
\renewcommand{\arraystretch}{1.4}
{\large\begin{tabular}{ll}
\textbf{멀티헤드} & 512칸을 \kb{8개 머리} × 64칸으로 나눠 따로 $\to$ 이어 붙여 $W_O$\\
왜 여러 개? & 머리마다 다른 관계 (주어 $\leftrightarrow$ 동사, ``피곤하다'' $\leftrightarrow$ 밤새: 이유)
\end{tabular}}\\[0.6em]
{\small 지난 시간과 달리 $Q, K, V$ 를 \textbf{학습하는 행렬}로: 무엇을 찾을지 · 내걸지 · 줄지를 배운다}
\vfill
\note{트랜스포머의 셀프 어텐션은 지난 시간 피곤하다 예제와 같지만 한 가지가 다릅니다. 쿼리, 키, 밸류를 단어 벡터 그대로 쓰지 않고, 각각 학습하는 행렬 W_Q, W_K, W_V를 곱해 만듭니다. 그래서 무엇을 찾을지, 무엇을 내걸지, 무엇을 줄지를 배웁니다. 또 원 논문은 512칸을 64칸씩 8개로 나눠 어텐션을 8번 따로 합니다. 이걸 멀티헤드라고 합니다. 머리마다 주어와 동사, 대명사와 가리키는 말처럼 서로 다른 관계를 볼 수 있습니다. 8개 결과를 이어 붙여 행렬 W_O를 곱하면 다시 512칸이 됩니다.}
\end{frame}

\section{그래프로 보기}
% --- GNN 한 층
\begin{frame}{다른 눈으로: 그래프 신경망 (GNN) 한 층}
\markpage{gnn}
\vfill
\begin{columns}[c]
\column{0.5\linewidth}\centering
\includegraphics[width=\linewidth]{tg_gnn.png}
\column{0.48\linewidth}
{\large 점 = 사람, \quad 선 = 친구 관계}\\[0.6em]
{\large 한 층에서 점마다 \textbf{이웃의 정보를 모아} 자기 정보를 고친다 \quad (메시지 패싱)}\\[0.6em]
{\Large\[ \mathbf{h}_i' = \sum_{j \in N(i)} \textcolor{ksablue}{a_{ij}} \, W \mathbf{h}_j \]}
{\large $N(i)$ = $i$ 의 이웃 \quad 가장 단순하게: 친구 4명이면 $a_{ij} = \frac14$ 씩 평균}\\[0.4em]
{\small E 는 친구의 친구: 이 층에서는 안 보이고, 층을 하나 더 쌓으면 닿는다 \quad 자세히는 다음 시간 GCN}
\end{columns}
\vfill
\note{트랜스포머를 다른 눈으로 봅시다. 그래프 신경망, GNN은 점과 선으로 된 데이터를 다룹니다. 점은 사람, 선은 친구 관계라고 해 봅시다. GNN 한 층에서는 점마다 이웃의 정보를 모아 자기 정보를 새로 고칩니다. 이걸 메시지 패싱이라고 합니다. 가장 단순하게는 이웃을 똑같이 평균 냅니다. 나에게 친구가 네 명이면 한 명당 4분의 1씩, 1, 4, 2, 5를 평균 내 3이 됩니다. E는 친구의 친구라 이 층에서는 보이지 않고, 층을 하나 더 쌓으면 B를 거쳐 닿습니다. 그래프 신경망은 다음 시간 GCN에서 자세히 배웁니다.}
\end{frame}

% --- 문장 = 완전 그래프
\begin{frame}{트랜스포머 = 모든 단어가 이어진 그래프 위의 GNN}
\markpage{tfgnn}
\vfill
\begin{columns}[c]
\column{0.42\linewidth}\centering
\includegraphics[width=\linewidth]{tg_sentence.png}\\
{\small 왜 피곤할까? 밤새 0.45 (어텐션 시간 손 계산)}
\column{0.56\linewidth}
\renewcommand{\arraystretch}{1.5}
{\normalsize\begin{tabular}{lll}
 & GNN & 트랜스포머\\\hline
점 & 사람 & \textbf{단어}\\
선 & 주어진 관계 & \kb{모든 짝} (완전 그래프)\\
무게 $a_{ij}$ & $\frac{1}{\text{이웃 수}}$ 고정 & $\text{softmax}\!\left(\frac{\mathbf{q}_i \cdot \mathbf{k}_j}{\sqrt{d}}\right)$\\
모으는 것 & $W \mathbf{h}_j$ & $\mathbf{v}_j$
\end{tabular}}\\[0.6em]
{\large 어텐션 = 선마다 \textbf{계산한} 무게}\\[0.2em]
{\small Joshi (2020) ``Transformers are Graph Neural Networks''}
\end{columns}
\vfill
\note{이제 문장을 그래프로 그려 봅시다. 단어 하나가 점 하나이고, 모든 단어를 서로 잇습니다. 지난 시간 그 문장, 나는 밤새 게임을 해서 피곤하다에서, 피곤하다는 왜 피곤한지 알려면 밤새를 가장 크게, 해서는 작게 봐야 합니다. 그림의 선 굵기가 어텐션 무게입니다. 숫자는 지난 시간 어텐션 손 계산에서 나온 값입니다. 표로 비교하면, GNN은 선이 데이터에 주어지고 무게가 이웃 수로 정해지지만, 트랜스포머는 모든 단어 짝을 선으로 잇고, 무게를 쿼리와 키의 내적으로 매번 계산합니다. 그리고 그 무게대로 밸류를 섞어 자기 정보를 고칩니다. GNN 한 층과 똑같은 일입니다. 그래서 2020년 차이타냐 조시는 트랜스포머는 그래프 신경망이다라는 글로 이 관점을 정리했습니다.}
\end{frame}

% --- 위치 정보
\begin{frame}{순서는? 위치 정보를 더한다}
\markpage{pe}
\vfill
\centering
{\large 그래프에는 순서가 없다: Dog bites man = Man bites dog (\textbf{같은 그래프})}\vskip0.3em
\includegraphics[width=0.72\linewidth]{tf_pe.png}\par
{\large 입력 = 단어 벡터 + \kb{위치 벡터} \quad $\text{PE}(pos, 2i) = \sin\!\left(pos / 10000^{2i/d}\right)$, \quad $2i + 1$ 칸은 $\cos$}\\[0.2em]
{\small 칸마다 주기가 다른 물결 $\to$ 위치마다 다른 무늬 \quad 위치 벡터를 학습으로 정해도 비슷하다 (논문, GPT)}
\vfill
\note{그래프로 보면 문제가 하나 보입니다. 그래프에는 순서가 없습니다. 점의 자리를 섞어도 같은 그래프죠. 셀프 어텐션도 단어 쌍의 비슷함만 보니까 순서를 모릅니다. 12주차에 본 Dog bites man과 Man bites dog을 구분하지 못합니다. 그래서 단어 벡터에 위치를 나타내는 벡터를 더합니다. 원 논문은 사인과 코사인을 썼습니다. 칸마다 주기가 다른 물결이라, 왼쪽 그림처럼 위치마다 다른 무늬가 생깁니다. 앞 칸은 빨리, 뒤 칸은 천천히 바뀝니다. 시계의 초침, 분침, 시침처럼요. 위치 벡터를 학습으로 정해도 결과가 비슷하다고 논문에 나오고, GPT는 학습하는 방식을 씁니다.}
\end{frame}

\end{frame}

GPT-3은 d가 12288, 96층이라 12 곱하기 12288 제곱 곱하기 96, 약 1740억 개입니다. 발표된 1750억 개가 거의 이 식 하나로 나옵니다.}
\end{frame}

\section{GPT}
% --- 앞만 보기
\begin{frame}{GPT: 앞 단어만 보고 다음 단어 맞히기}
\vfill
\begin{columns}[c]
\column{0.4\linewidth}\centering
\includegraphics[width=\linewidth]{tg_mask.png}
\column{0.58\linewidth}
{\large 다음 단어를 맞히는데 뒤 단어를 보면 반칙}\\[0.3em]
{\large $\to$ 점수 행렬의 위쪽을 $-\infty$ 로 (softmax 후 0)}\\[0.8em]
\renewcommand{\arraystretch}{1.4}
{\large\begin{tabular}{ll}
원 논문 & 인코더 6층 + 디코더 6층 (번역)\\
BERT (2018) & 인코더만: 양쪽 다 본다\\
GPT (2018 $\sim$) & 디코더만: \kb{앞만} 본다
\end{tabular}}\\[0.5em]
{\small 그래프로는: 선에 \textbf{방향}이 생겨 앞 단어에서만 정보가 온다}\\[0.2em]
{\small 앞만 보기로 길이 $n$ 문장에서 $n$ 개 문제를 한 번에 학습}
\end{columns}
\vfill
\note{GPT는 앞 단어들을 보고 다음 단어를 맞히도록 학습합니다. 그런데 셀프 어텐션은 모든 단어를 한 번에 보니까, 그대로 두면 뒤 단어, 즉 정답을 봐 버립니다. 그래서 점수 행렬의 위쪽 절반을 마이너스 무한대로 바꿔 softmax 뒤 0이 되게 합니다. 그림처럼 같은 문장에서 게임을은 나는, 밤새, 게임을까지만 보고, 마지막 피곤하다만 모든 단어를 봅니다. 원 논문은 번역을 위해 인코더 6층과 디코더 6층을 썼고, BERT는 인코더만 써서 양쪽을 다 보고, GPT는 디코더만 써서 앞만 봅니다. 앞만 보게 하면 길이 n 문장 하나로 다음 단어 문제 n개를 한 번에 학습할 수 있습니다.}
\end{frame}

base 모델 & & GPU 8개로 \textbf{12시간}
\end{tabular}}\vskip0.8em
{\large RNN 없이 어텐션만으로 더 잘, 더 빨리 \quad $\to$ 제목: Attention Is All You Need}
\vfill
\note{2017년 논문의 결과입니다. 영어 독일어 번역에서 BLEU 28.4점으로 앙상블을 포함한 기존 최고 기록보다 2점 넘게 높았고, 영어 프랑스어 번역에서는 41.8점을 GPU 8개로 3.5일 학습해 얻었습니다. 기존 모델들의 학습 비용보다 훨씬 적은 계산입니다. 작은 base 모델은 GPU 8개로 12시간이면 충분했습니다. RNN 없이 어텐션만으로 더 잘, 더 빨리 학습했다, 그래서 제목이 어텐션만 있으면 된다입니다.}
\end{frame}

\section{정리}
% --- 정리
\begin{frame}{정리}
\vfill
\centering\large\begin{tabular}{l}
\textbf{트랜스포머}: RNN 없이 셀프 어텐션으로 모든 단어를 \kb{한 번에} (2017)\\[0.5em]
그래프로 보면: 모든 단어가 이어진 그래프 위의 GNN, \kb{어텐션 = 선의 무게}\\[0.5em]
입력 = 단어 + 위치 \quad 블록 = 멀티헤드 어텐션 $\to$ 피드포워드, 지름길 · LayerNorm\\[0.5em]
블록 하나 $\approx 12 d^2$ \quad GPT-3: $12 \times 12288^2 \times 96 \approx 1{,}750$억\\[0.5em]
GPT = 앞만 보는 디코더 \quad 셰익스피어 검증 손실 1.64 (LSTM) $\to$ 1.61
\end{tabular}
\vfill
\note{정리합니다. 트랜스포머는 RNN 없이 셀프 어텐션으로 모든 단어를 한 번에 봅니다. 그래프로 보면 모든 단어가 서로 이어진 그래프 위의 GNN이고, 어텐션은 선마다 계산한 무게입니다. 그래프에는 순서가 없으니 단어 벡터에 위치 벡터를 더하고, 블록 하나는 멀티헤드 어텐션으로 주고받고 피드포워드로 다듬으며, 지름길과 LayerNorm이 붙습니다. 블록 하나의 가중치는 약 12 d 제곱이고, GPT-3의 1750억 개가 이 식으로 나옵니다. GPT는 앞만 보는 디코더이고, 셰익스피어 실험에서는 LSTM의 1.64보다 낮은 1.61을 얻었습니다.}
\end{frame}
