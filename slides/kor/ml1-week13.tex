\begin{frame}{이번 주차 개요}
  \begin{itemize}
    \item \kb{13.1 Random Walk 기초} — 그래프 위 \kb{무작위 걷기}가 마르코프
      체인임을 정의하고, 카라테 클럽에서 걷기 말뭉치를 만들고,
      노드에 머무는 장기 비율이 도(degree)에 비례함을 유도한다.
    \item \kb{13.2 Node2Vec} — 걷기 경로 = ``문장'', 노드 = ``단어''로

    \item \kb{13.3 PageRank} — ``영원히 돌아다닌 뒤 각 노드에 머무는
      확률은?''이라는 다른 질문. 댐핑 팩터 + 순간이동, 거듭제곱법
      손계산, 그리고 두 가지 흔한 실수.
    \item \kb{13.4 그래프 학습의 확장} — 두 아이디어를 실제 문제로:
      단백질(AlphaFold $\to$ DeepFRI), 금융(AML World), 교통(STGCN).
  \end{itemize}
  \vskip0.5em
  \kq{그래프로 표현된 데이터를, 신경망이나 통계 모델이 다룰 수 있는
      벡터로 어떻게 바꾸는가?}
\end{frame}

\section{13.1 Random Walk 기초}

\begin{frame}{이 장은 무엇을 묻는가 --- AlphaFold가 준 관점}
  2021년 딥마인드의 \kb{AlphaFold}(Jumper et al., 2021)는 아미노산
  서열만 보고 단백질의 3차원 구조를 원자 단위 정확도로 예측해,
  50년 가까운 생물학 난제를 사실상 해결했다.
  \vskip0.6em
  \begin{itemize}
    \item 핵심 관점 중 하나: 단백질을 \kb{그래프}로 본다.
    \item 각 아미노산(잔기, residue) = \textbf{노드},
      공간적으로 가까운 잔기 쌍 사이의 관계 = \textbf{엣지}.
    \item ``어떤 잔기가 어떤 잔기와 가까운가''를 알면
      3차원 구조가 따라 나온다.
  \end{itemize}
  \vskip0.6em
  이 장의 질문은 AlphaFold 구조 자체를 유도하는 게 아니라 더 근본적인 것 ---
  \kq{그래프로 표현된 데이터를, 벡터로 어떻게 바꾸는가?}
\end{frame}

\begin{frame}{핵심 실: word2vec 스킵그램의 그래프 재사용}
  \begin{block}{14.2절의 스킵그램 원리}
    ``문맥 안에서 \textbf{함께 나타나는} 단어는 \textbf{비슷한 벡터}를
    갖는다''
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item 이 원리를 그래프에 \textbf{그대로} 재사용한다.
    \item ``문장''의 역할은 그래프 위의 \kb{무작위 걷기}(random walk).
    \item 이 장은 14장(표현학습, PCA·word2vec)과 순환 구조:
      14장에서 ``함께 자주 나타나는 것은 가까운 벡터로 모인다''의
      \textbf{텍스트} 버전을 배울 예정이고, 이 장은 그 적용 대상을
      \textbf{텍스트가 아닌 그래프}로 바꾼 것을 먼저 보여준다.
  \end{itemize}
  \vskip0.4em
  {\footnotesize 뒤로 보면: ``그래프 위에서 값을 반복해서 갱신한다''는
  감각은 15장(잠재변수 생성모델)의 확률 모델과 ML2의 벨만 방정식
  (반복 대입으로 고정점 수렴)으로 이어진다.}
\end{frame}

\begin{frame}{그래프 위 무작위 걷기 = 마르코프 체인}
  \begin{block}{정의}
    노드 집합 $\{0,\dots,N-1\}$의 그래프 위에서 ``지금 서 있는 노드''를
    \kb{상태}로 보고, 매 단계에서 \textbf{지금 노드의 이웃 중 하나를
    골라 이동}하는 확률과정 = \kb{마르코프 체인}(Markov chain).
  \end{block}
  \vskip0.4em
  \begin{itemize}
    \item 상태 $i$에서 $j$로 건너가는 \kb{전이확률}
      (transition probability):
      \[ P(i \to j) = \frac{1}{\text{deg}(i)} \quad ((i,j) \in \mathcal{E}), \qquad
         P(i \to j) = 0 \quad (\text{그 외}) \]
    \item 이웃을 ``동등한 확률로 골라'' 이동 = \kb{일괄적}(uniform)
      무작위 걷기.
  \end{itemize}
\end{frame}

\begin{frame}{같은 체인, 다른 파라미터 --- 다음 두 절의 예고}
  \begin{itemize}
    \item \kb{13.2 Node2Vec}: 이 전이확률에 $p, q$ 두 계수를 붙여
      \textbf{편향}(bias)시키는 변형.
    \item \kb{13.3 PageRank}: 이 전이확률에 ``\textbf{순간이동}''을
      섞은 변형.
  \end{itemize}
  \vskip0.5em
  \begin{block}{한 줄 정리}
    ``\textbf{같은 마르코프 체인} 위에 파라미터만 바꾸는 것'' ---
    \kq{같은 무작위 걷기에 서로 다른 질문을 던진다.}
  \end{block}
\end{frame}

\begin{frame}{마르코프 체인의 핵심 성질: 무자기성}
  다음 상태를 결정하는 것은 ``지금 어디에 있는지''뿐이다 ---
  그전에 어떤 경로를 거쳐 왔는지는 \textbf{아니다}.
  \vskip0.8em
  \[ P(x_{t+1} \mid x_t, x_{t-1}, \dots, x_0) = P(x_t \to x_{t+1}) \]
  \vskip0.8em
  \begin{itemize}
    \item 이 한 줄이 무작위 걷기가 ``단순한 반복''으로 분석될 수 있는
      이유 --- 그래서 \textbf{수십억 노드의 웹에도 거듭제곱법 한 줄로}
      돌릴 수 있다.
    \item 예: 카라테 클럽에서 node 16이 node 5를 거쳐 왔는지 node 6을
      거쳐 왔는지 \textbf{기억하지 않고}, 오직 ``지금 node 16에 서 있고
      내 이웃은 5와 6이다''만 아는 것이 바로 이 성질.
  \end{itemize}
\end{frame}

\begin{frame}{상태 분포를 계속 곱하면 --- 정상분포}
  \begin{itemize}
    \item 전이확률 행렬 $P$를 ``각 노드에 서 있을 확률''인
      상태 분포 $\pi$에 계속 곱한다:
      \[ \pi_{t+1} = \pi_t P \]
    \item 분포는 결국 멈추는 \kb{정상분포}(stationary distribution)
      $\pi^*$ 에 도달한다:
      \[ \pi^* P = \pi^* \]
    \item ``\textbf{영원히 돌아다닌 뒤 각 노드에 머무는 비율}''이
      바로 이 $\pi^*$ --- 아래에서 도(degree)에 비례함을 직접 유도.
  \end{itemize}
\end{frame}

\begin{frame}{같은 무작위 걷기, 다른 질문}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{\color{ksablue}13.2 Node2Vec}
      \begin{itemize}
        \item ``이 걷기가 만들어내는 \textbf{노드의
          순서(시퀀스)}는 어떤 구조를 담고 있는가?''
        \item 그 시퀀스를 ``문장'' 취급해 word2vec의
          skip-gram을 학습 $\to$ \textbf{노드 임베딩}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\color{ksablue}13.3 PageRank}
      \begin{itemize}
        \item ``이 걷기를 영원히 돌리면 \textbf{각 노드에 머무는
          확률} $\pi^*$는 얼마인가?''
        \item 그 정상분포 자체를 노드의 ``\textbf{중요도}''로 읽는다.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.6em
  \begin{block}{한 줄}
    같은 재료(그래프, 무작위 걷기)에서 하나는 \textbf{시퀀스}를 뽑아내고,
    하나는 \textbf{정상분포}를 뽑아낸다 --- 벡터와 스칼라,
    두 가지 다른 표현.
  \end{block}
\end{frame}

\begin{frame}{왜 ``무작위 걷기''가 ``문장''을 대신하는 자연스러운 선택인가}
  \begin{itemize}
    \item word2vec에서 ``문장''이 가진 본질: \textbf{어떤 것들이 어떤
      순서로 함께 나타나는지}를 담는 것.
    \item 그래프 노드의 ``함께 나타남'' = ``엣지로 연결되어,
      서로를 따라다니면 만나는 것''.
    \item 어떤 노드에서 출발해 이웃을 무작위로 골라 반복 이동하는
      \textbf{무작위 걷기}가 바로 이 ``함께 나타남의 순서''를 만들어내는
      \textbf{최소 장치}.
    \item 걷기 안에서 \textbf{인접한 노드는 서로 엣지로 연결되어} 있고,
      어떤 노드 주위에 자주 등장하는 노드 = 그 노드의 구조적 ``이웃''.
  \end{itemize}
\end{frame}

\begin{frame}{머무는 비율은 도(degree)에 비례한다}
  \begin{block}{핵심 성질}
    무방향 그래프에서 일괄적 무작위 걷기를 무한히 계속하면,
    노드 $i$에 머무는 장기 비율은 그 노드의 \kb{도}(degree,
    연결된 엣지 수)에 비례:
    \[ \pi(i) = \frac{\text{deg}(i)}{2\,|\mathcal{E}|} \]
    ($|\mathcal{E}|$ = 엣지 수, $2|\mathcal{E}|$ = 모든 노드의 도의 합)
  \end{block}
  \vskip0.4em
  \begin{itemize}
    \item 즉 \textbf{허브 노드는 걷기 말뭉치에 더 자주 등장} $\to$
      skip-gram 학습에서 \textbf{더 많이 업데이트}.
  \end{itemize}
\end{frame}

\begin{frame}{허브는 더 많이 ``나타나는 단어''}
  \begin{itemize}
    \item 14.2절의 ``\textbf{빈도 높은 단어의 임베딩 벡터 노름이 크다}''
      는 현상과 \textbf{정확히 같은 메커니즘}.
    \item 카라테 클럽에서: node 33(관장) 도 17, node 0(감독) 도 16 ---
      두 파벌 리더가 구조적으로 가장 많이 ``나타나는'' 단어.
    \item 임베딩 학습이 자동으로 \textbf{구조적으로 중요한 노드에 더 많은
      ``학습 예산''}을 배정해준다고 볼 수 있다.
  \end{itemize}
  \vskip0.6em
  \begin{block}{주석 --- 이 절 후반과 13.3의 연결}
    ``영원히 돌아다닌 뒤 각 노드에 머무는 비율''이라는 질문은
    댐핑을 빼면 곧바로 \textbf{PageRank} 그 자체.
    이 절의 후반부가 바로 이 질문에 정교한 답을 붙인다.
  \end{block}
\end{frame}

\begin{frame}[fragile]{코드: 일괄적 무작위 걷기로 말뭉치 만들기}
\begin{lstlisting}
def random_walk(neighbors, start, length):
    walk = [start]
    for _ in range(length - 1):
        cur = walk[-1]
        if not neighbors[cur]:
            break
        walk.append(random.choice(neighbors[cur]))
    return walk

def build_walks(edges, n_nodes, walks_per_node=10, walk_length=8):
    neighbors = {i: [] for i in range(n_nodes)}
    for a, b in edges:
        neighbors[a].append(b)
        neighbors[b].append(a)
    walks = []
    for node in range(n_nodes):
        for _ in range(walks_per_node):
            walks.append([str(n) for n in random_walk(neighbors, node,
                                                      walk_length)])
    return walks
\end{lstlisting}
  \vskip0.2em
  {\footnotesize \texttt{random\_choice} = 이웃을 동등한 확률로
  고르는 일괄적 걷기. 각 걷기 = ``문장'', 각 노드 번호 = ``단어''.}
\end{frame}

\begin{frame}{걷기 $\to$ 말뭉치 $\to$ skip-gram $\to$ 임베딩}
  \begin{enumerate}
    \item \kb{걷기} --- \texttt{build\_walks} 로 34개 노드 $\times$
      노드당 10개 걷기(길이 8) 생성.
    \item \kb{말뭉치} --- 이어붙인다: \texttt{corpus = sum(walks, [])}.
    \item \kb{학습} --- 14.2절의 \texttt{train\_skipgram}에
      \textbf{그대로} 넣으면 \textbf{34개 노드 각각의 임베딩 벡터}가
      나온다.
    \item \kb{시각화} --- 14.1절의 \textbf{PCA}로 2차원까지 압축해
      그리면, 실제로 두 파벌이 공간적으로 갈라진다.
  \end{enumerate}
  \vskip0.5em
  \begin{block}{같은 아이디어의 두 얼굴}
    word2vec과 Node2Vec은 서로 다른 데이터(텍스트 vs. 그래프)에
    적용됐을 뿐, ``\textbf{함께 자주 나타나는 것들은 비슷한 벡터를
    갖도록 학습한다}''는 \textbf{같은 아이디어의 두 얼굴}.
  \end{block}
\end{frame}

\begin{frame}{해석: 리더는 ``경계''가 아니라 군집 \textbf{중심}에}
  \begin{itemize}
    \item 두 파벌이 한쪽/다른 쪽으로 갈라지고, 두 파벌 리더
      (node 0, node 33)이 \textbf{각 군집의 중심 부근}에 있다.
    \item 두 리더의 임베딩이 자기 파벌 군집 중심에서 각각
      \textbf{약 0.2, 0.3} 거리 --- 나머지 멤버의 중앙값은
      감독파 약 1.1, 관장파 약 0.4.
    \item ``리더는 두 파벌의 \textbf{경계}에 있어야 할 것'' 같은
      직관과 달리, 실제로는 군집 \textbf{내부에 잘 박혀 있다}.
    \item 군집 경계 근처에 두 파벌을 잇는 ``\textbf{다리}'' 역할을
      하는 멤버들이 섞여 있어도 \textbf{정상} --- 그래프에서
      실제로 두 클러스터 사이를 잇고 있다.
  \end{itemize}
\end{frame}

\section{13.2 Node2Vec: 걷기에 skip-gram을 그대로}

\begin{frame}{13.1을 한 줄로: 걷기 안의 인접 = ``함께 나타남''}
  \begin{block}{확립된 것}
    그래프 위의 \kb{무작위 걷기}는 마르코프 체인이고,
    그 걷기 안의 \textbf{인접 노드들은 서로를 ``함께 나타남''으로 잇는다}.
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item 이제 그 ``함께 나타남의 순서'', 곧 \textbf{걷기 시퀀스}를
      14.2절 word2vec의 \kb{skip-gram} 학습에 \textbf{그대로} 넣으면 된다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{\kb{Node2Vec}(Grover \& Leskovec, 2016)}
    random walk로 생성한 경로들에 word2vec의 skip-gram을
    \textbf{그대로 학습시킨 것}.
  \end{block}
\end{frame}

\begin{frame}{word2vec의 세 요소를 그래프 버전으로 치환}
  \small
  \begin{tabular}{@{}lll@{}}
    \toprule
    word2vec & $\to$ & Node2Vec \\
    \midrule
    문장 & $\to$ & 한 노드에서 출발해 이웃을 무작위로 따라가며
                만든 \textbf{걷기 경로}(노드 시퀀스) \\
    단어 & $\to$ & 그 경로 위에 순서대로 나오는 \textbf{노드} \\
    학습 & $\to$ & 14.2절의 \texttt{train\_skipgram}을 \textbf{한 줄도
                고치지 않고} 그대로 (토큰이 단어 대신 노드 번호일 뿐) \\
    \bottomrule
  \end{tabular}
  \vskip1em
  ``비슷한 문맥에 나오는 단어는 비슷한 벡터를 갖는다'' $\to$
  ``\textbf{비슷한 이웃 구조를 가진 노드는 비슷한 벡터를 갖게
  된다}''는 문장으로 바뀐 것뿐.
\end{frame}

\begin{frame}{학습이 끝나면: 은닉층 가중치 = 노드 임베딩}
  \begin{itemize}
    \item 14.2절과 똑같이 \textbf{은닉층 가중치 행이 각 노드의
      임베딩 벡터}.
    \item 14.2절에서 word2vec은 ``작은 \textbf{신경망}을 학습시킨다''고
      소개됐다 --- one-hot 입력층 · 임베딩 차원 은닉층 · 어휘 크기
      출력층, 이진 교차엔트로피 최소화.
  \end{itemize}
  \vskip0.5em
  \begin{block}{핵심 재료가 신경망을 \textbf{전제로 하지 않는다}}
    ``걷기로 시퀀스를 만들고 그 위에서 함께-나타남을 학습한다''는
    틀 자체는 \textbf{신경망 없이도 성립}. skip-gram을 쓰든,
    더 단순한 ``함께 나타난 횟수'' 카운트 기반의 선형 분해
    (14.2절의 LSI 같은)를 쓰든 --- Node2Vec은 그 \textbf{더
    일반적인 틀} 위에 14.2절의 skip-gram이라는 하나의 실용적
    학습자를 골랐을 뿐.
  \end{block}
\end{frame}

\begin{frame}{Node2Vec의 진짜 기여: 두 계수 $p$와 $q$}
  \begin{itemize}
    \item 13.1절 코드의 일괄적 걷기 = 원본 논문의 \textbf{가장 단순한
      특수한 경우} ($p{=}q{=}1$).
    \item 논문의 실제 기여: 걷기에 \kb{$p$}(return)과 \kb{$q$}(in-out)
      두 계수를 붙여 ``\textbf{이웃 탐색의 스타일}''을 조절.
  \end{itemize}
  \vskip0.5em
  걷기가 노드 $x_t$에 도달했을 때(직전 $x_{t-1}$), 다음 노드
  $x_{t+1}$를 고르는 확률은 가중치 $w_{x_{t-1}}(x_{t+1})$에
  \textbf{비례}:
  \begin{itemize}
    \item \textbf{돌아가기} --- 방금 있던 노드로 복귀
      ($x_{t+1} = x_{t-1}$): 가중치 $1/p$
    \item \textbf{인근 머물기} --- $x_{t-1}$의 이웃(이미 탐색한
      로컬 영역 안): 가중치 $1/q$
    \item \textbf{새 영토로} --- 나머지(아직 탐색 안 한 이웃):
      가중치 $1$
  \end{itemize}
\end{frame}

\begin{frame}{계수를 움직이면 탐색 스타일이 기울어진다}
  \begin{itemize}
    \item $p = q = 1$: 모든 가중치가 1 $\to$ 정확히
      \textbf{일괄적 무작위 걷기}.
    \item \kb{$p < 1$} ($1/p > 1$): 직전 노드로의 \textbf{뒤로 감}
      (backtracking) 확률이 올라간다.
    \item \kb{$q < 1$} ($1/q > 1$): 방금 탐색한 \textbf{로컬 영역 안에
      머물기} 확률이 올라간다.
    \item \kb{$q > 1$} ($1/q < 1$): 이미 본 영역을 다시 밟는 것이
      \textbf{억제}되어 걷기가 \textbf{새 영토로 확장}하려 한다.
  \end{itemize}
\end{frame}

\begin{frame}{원본 논문의 두 표준 프리셋}
  \small
  \begin{tabular}{@{}p{0.3\textwidth}p{0.25\textwidth}p{0.33\textwidth}@{}}
    \toprule
    설정 & 탐색 스타일 & 강조하는 구조 \\
    \midrule
    $(p{=}1, q{=}16)$ --- ``node2vec'' & \textbf{BFS형}(넓이 우선)
      & \kb{구조적 동치}(structural equivalence) --- 역할의 유사성 \\
    $(p{=}16, q{=}1)$ --- ``deep walk'' & \textbf{DFS형}(깊이 우선)
      & \kb{동질성}(homophily) --- 같은 커뮤니티 속한 것의 유사성 \\
    \bottomrule
  \end{tabular}
  \vskip1em
  {\footnotesize 일괄적 무작위 걷기는 둘을 \textbf{섞어서} 담고,
  $p, q$가 이 중 어느 쪽에 더 초점을 맞추게 하는 ``\textbf{다이얼}''
  인 셈.}
\end{frame}

\begin{frame}{동질성 vs 구조적 동치 --- 서로 다른 두 유사도}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{\color{ksablue}동질성 (homophily)}
      \begin{itemize}
        \item 같은 커뮤니티의 노드는 \textbf{동질}하다.
        \item 예: 카라테 클럽에서 \textbf{한 파벌의 멤버끼리}
          서로 비슷.
        \item $\to$ DFS형 $(16,1)$ 가 강조.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\color{ksablue}구조적 동치 (structural equivalence)}
      \begin{itemize}
        \item \textbf{서로 다른} 커뮤니티의 두 노드도 \textbf{역할이
          같을} 수 있다.
        \item 예: 다른 부서 두 곳에서 각각 ``외부와의 \textbf{유일한
          연결고리}'' 역할을 하는 두 멤버.
        \item $\to$ BFS형 $(1,16)$ 가 강조.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{코드: 편향 무작위 걷기 (일괄적 걷기를 대체)}
\begin{lstlisting}
def biased_walk(nb, start, length, p, q, rng):
    """Node2Vec의 편향 무작위 걷기. nb: 노드 -> 이웃 목록"""
    walk = [start]
    prev = None
    for _ in range(length - 1):
        cur = walk[-1]
        weights = []
        for nxt in nb[cur]:
            if prev is None:
                weights.append(1.0)
            elif nxt == prev:
                weights.append(1.0 / p)      # 돌아가기
            elif nxt in nb[prev]:
                weights.append(1.0 / q)      # 로컬 영역 머무르기
            else:
                weights.append(1.0)          # 새 영토로
        r = rng.random() * sum(weights)      # 가중치에 비례해 뽑기
        acc, chosen = 0.0, nb[cur][0]
        for nxt, w in zip(nb[cur], weights):
            acc += w
            if r <= acc:
                chosen = nxt
                break
        walk.append(chosen)
        prev, cur = cur, chosen
    return walk
\end{lstlisting}
\end{frame}

\begin{frame}{작은 실험: $p$와 $q$가 임베딩을 얼마나 바꾸는가}
  세 가지 설정으로 걷기를 \textbf{재생성}해서 동일하게 Node2Vec을
  학습(시드 고정):
  \vskip0.6em
  \begin{itemize}
    \item \kb{일괄적} $(1,1)$ --- 일괄적 무작위 걷기
    \item \kb{BFS형} $(1,10)$ --- 새 영토 확장 강조
    \item \kb{DFS형} $(10,1)$ --- 로컬 영역 머물기 강조
  \end{itemize}
  \vskip0.4em
  {\footnotesize 공통 하이퍼파라미터: \texttt{walks\_per\_node=10},
  \texttt{walk\_length=8}, dim=8 (노트북).}
\end{frame}

    \item 카라테 그래프는 \textbf{무방향}, PageRank는

    \item 표준 기법: ``친분 관계 1개''를 \textbf{양방향 링크 2개}로

      --- 달라진다.
  \end{itemize}
\end{frame}

    대칭화는 \textbf{한 줄}이지만, 이 실수는

  \end{itemize}
\end{frame}

\section{13.3 PageRank: 정상분포를 중요도로}

\begin{frame}{같은 무작위 걷기, 다른 질문: PageRank}
  \begin{itemize}
    \item 13.1: 걷기를 ``노드가 이웃을 얼마나 자주 만나는지''를 담는
      \textbf{말뭉치}로.
    \item 13.2: 그 말뭉치에 skip-gram을 학습 $\to$ \textbf{임베딩}.
    \item 이 절: 말뭉치가 아니라, 13.1의 마지막 주석
      ``영원히 돌아다닌 뒤 각 노드에 \textbf{머무는 비율}''에
      정식으로 답 --- 그 답, 즉 \textbf{무작위 걷기의 정상분포
      그 자체}를 노드 하나당 \textbf{스칼라 ``중요도''}로 쓰는 것이
      \kb{PageRank}.
  \end{itemize}
  \vskip0.4em
  \begin{block}{같은 재료, 갈라진 결과물}
    \kq{같은 무작위 걷기를 13.2와 이 절이 모두 쓰지만, 묻는 질문이
    달라서} 나온 결과물이 \textbf{임베딩 벡터}와 \textbf{스칼라
    점수}로 갈라진 것.
  \end{block}
\end{frame}

\begin{frame}{1998년: 래리 페이지와 세르게이 브린}
  \begin{itemize}
    \item 1996--97년, 스탠퍼드 대학원생이던 \kb{래리 페이지}
      (Larry Page)와 \kb{세르게이 브린}(Sergey Brin).
    \item 당시 웹 검색엔진들은 ``페이지에 \textbf{키워드가 있는가}''로만
      순위를 매김 $\to$ 웹이 너무 커져 \textbf{쓸모없는 결과만 뒤섞여}
      나온다는 문제를 직감.
    \item 그들의 답: \textbf{웹 자체를 데이터로 보는 것} ---
      수억 개 웹페이지 사이의 \textbf{링크 구조}는 그 자체로
      ``\textbf{웹의 투표 기록}''
    \item 다른 사이트의 주인들이 자발적으로 그 페이지를 링크한 것이
      ``이 페이지가 중요하다''는 \textbf{증거}.
  \end{itemize}
\end{frame}

\begin{frame}{PageRank의 핵심 아이디어와 정식화}
  \begin{itemize}
    \item 이 그래프 위를 \textbf{무작위로 영원히 돌아다닌다면},
      각 노드에 머물러 있을 확률은 각각 얼마나 될까?
    \item 그 답 = 그 노드의 ``\textbf{중요도}'' (1998, Brin \& Page).
    \item 많은 페이지가 링크하는 페이지일수록, 또 \textbf{중요한
      페이지가} 링크하는 페이지일수록, \kb{무작위 서퍼}(random
      surfer)가 더 자주 머문다.
  \end{itemize}
  \vskip0.5em
  \begin{itemize}
    \item ``누가 누구를 링크하는가''를 \textbf{무작위 서퍼의 이동
      행렬}로 정식화 $\to$ 1998년 논문
      \textit{``The Anatomy of a Large-Scale Hypertextual Web Search
      Engine''}이 실용 알고리즘으로 완성.
  \end{itemize}
  {\footnotesize (사명 ``Google''은 $10^{100}$을 뜻하는
  ``googol''에서 온 말장난 --- 조직하려던 웹의 규모에 대한 선언.)}
\end{frame}

\begin{frame}{PageRank 방정식}
  노드 $i$의 페이지랭크 $PR(i)$는 다음을 만족해야 한다 ---
  \textbf{자신을 가리키는 모든 노드 $j$로부터, 그 노드가 가진 점수를
  그 노드의 바깥 링크 수만큼 나눠 받은 합}:
  \vskip0.8em
  \[ PR(i) = \frac{1-d}{N} + d\sum_{j \to i} \frac{PR(j)}{\text{outdeg}(j)} \]
  \vskip0.8em
  \begin{itemize}
    \item $N$ = 전체 노드 수.
    \item $d$ (보통 \textbf{0.85}) = \kb{댐핑 팩터}(damping factor):
      확률 $d$로 링크를 따라가고, 확률 $1-d$로 \textbf{아무 노드로나
      순간이동}(teleport).
    \item 순간이동 = \textbf{막다른 골목}에 갇히거나 \textbf{순환}에
      갇히는 것을 막기 위한 장치.
  \end{itemize}
\end{frame}

\begin{frame}{재귀식이라 한 번에 못 풀고 --- 거듭제곱법}
  \begin{itemize}
    \item 이 식은 우변에 $PR$ \textbf{자신이 다시 등장}하는
      \textbf{재귀식}이라 한 번에 풀 수 없다.
    \item 그래서 아무 값(모든 노드에 $1/N$)에서 시작해, 우변에
      \textbf{반복해서 대입}하며 값이 더 이상 바뀌지 않을 때까지
      되풀이하는 \kb{거듭제곱법}(power iteration)으로 푼다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{형태가 수렴을 보장}
    ``\textbf{새 값 = teleport 항 + 옛 값들의 가중 평균}''이라는
    방정식의 형태 --- 이 형태가 반복법의 수렴을 보장하는 구조이며,
    바로 뒤에 온다.
  \end{block}
\end{frame}

\begin{frame}[fragile]{코드: pagerank (거듭제곱법)}
\begin{lstlisting}
def pagerank(edges, n_nodes, d=0.85, iters=100):
    outgoing = {i: [] for i in range(n_nodes)}  # directed: a -> b
    for a, b in edges:
        outgoing[a].append(b)
    outdeg = {i: max(1, len(outgoing[i])) for i in range(n_nodes)}
    incoming = {i: [] for i in range(n_nodes)}  # who points to me?
    for a, b in edges:
        incoming[b].append(a)

    pr = {i: 1 / n_nodes for i in range(n_nodes)}
    for _ in range(iters):
        pr = {i: (1 - d) / n_nodes + d * sum(pr[j] / outdeg[j]
                                             for j in incoming[i])
              for i in range(n_nodes)}
    return pr
\end{lstlisting}
\end{frame}

\begin{frame}{손으로 한 번: 노드 4개, 순환 하나, 외딴 끝 하나}
  \kb{$d = 0.5$}인 방향 그래프로 최소 예제:
  \vskip0.5em
  \begin{itemize}
    \item $A \to B$, $B \to C$, $C \to A$ --- \textbf{3-노드 순환}
    \item $D \to A$ --- D는 A만 가리킨다 (들어오는 링크는 \textbf{없음})
  \end{itemize}
  \vskip0.5em
  \begin{block}{teleport 항}
    \[ \frac{1-d}{N} = \frac{0.5}{4} = \textbf{0.125} \]
  \end{block}
\end{frame}

\begin{frame}{1단계 --- D는 들어오는 링크가 없다}
  \begin{itemize}
    \item D를 가리키는 노드가 없으니 합 $\sum_{j \to D}$ 는 비어 있다:
    \[ PR(D) = \frac{1-d}{N} = 0.125 \]
  \end{itemize}
  \vskip0.5em
  \begin{block}{읽기}
    D는 \textbf{반복을 아무리 돌려도 0.125에 머문다} ---
    ``나를 가리키는 이가 없으니 \textbf{teleport분밖에} 받을 수
    없다''는 뜻.
  \end{block}
\end{frame}

\begin{frame}{2단계 --- 순환 속 3개 노드에 방정식}
  \small
  \[ PR(A) = 0.125 + 0.5\left(\frac{PR(C)}{1} + \frac{PR(D)}{1}\right) \]
  \[ PR(B) = 0.125 + 0.5\,PR(A), \qquad
     PR(C) = 0.125 + 0.5\,PR(B) \]
  \vskip0.6em
  (각 노드의 outdeg가 1이라 \textbf{나누기는 일을 하지 않는다}.)
\end{frame}

\begin{frame}[fragile]{3단계 --- 대입으로 푼다}
  \scriptsize
  $PR(B), PR(C)$를 $PR(A)$ 식에 대입:
  \[ PR(A) = 0.125 + 0.5\left(\underbrace{0.125 + 0.5\bigl(0.125 + 0.5\,PR(A)\bigr)}_{PR(C)} + \underbrace{0.125}_{PR(D)}\right)
  = 0.28125 + 0.125\,PR(A) \]
  \[ PR(A) = \frac{0.28125}{0.875} = \frac{9}{28} \approx 0.321 \]
  \vskip0.6em
  \begin{lstlisting}
PR(B) = 0.125 + 0.5 * 0.321  # ≈ 0.286
PR(C) = 0.125 + 0.5 * 0.286  # ≈ 0.268
# 검산: 0.321 + 0.286 + 0.268 + 0.125 = 1.000
\end{lstlisting}
\end{frame}

\begin{frame}{이 작은 예제에서 읽어둘 것 두 가지}
  \begin{itemize}
    \item \textbf{첫째.} A·B·C는 모두 도 2(들어오는 1, 나가는 1)로
      ``겉모습은 똑같은''데, 점수는 \textbf{균등하지 않다}
      ($0.321 > 0.286 > 0.268$).
    \item ``$D \to A \to B \to C$'' 흐름을 받는 A가 \textbf{D의
      점수를 직접 받아} 가장 크고, \textbf{순환을 돌수록 우위가
      조금씩 소진}된다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{핵심}
    \kq{PageRank는 ``엣지가 몇 개인가''가 아니라 ``누가, 얼마나
    중요한 위치에서 나를 가리키는가''의 문제.}
  \end{block}
\end{frame}

\begin{frame}{거듭제곱법이 왜 수렴하는가 --- 바나흐 고정점 정리}
  \begin{itemize}
    \item 이 거듭제곱법의 수렴 논리는, \kb{ML2 Chapter 4}에서
      벨만 최적방정식의 해가 유일하게 존재함을 보일 때 쓸
      \kb{바나흐 고정점 정리}(Banach fixed point theorem)와
      \textbf{정확히 같은 논리}.
    \item ``\textbf{어떤 변환을 값이 안 바뀔 때까지 반복해서 적용하면
      결국 고정점}(fixed point)\textbf{에 도달한다}''는 같은 수학:
    \item 여기서 = ``\textbf{그래프 위 확률분포}'',
      강화학습에서는 = ``\textbf{상태의 가치함수}''.
  \end{itemize}
\end{frame}

\begin{frame}{수렴이 얼마나 빠른가 --- 지수적으로}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item 매 반복의 변화량(전체 노드 중 최대 $|\Delta PR|$)은
          \textbf{지수적으로} 줄어든다.
        \item 카라테 클럽: $d{=}0.85$ 일 때 대략 \textbf{45회}로
          $10^{-9}$ 이내 수렴. $d{=}0.5$ 이면 \textbf{19회}.
        \item $d$가 클수록 서퍼가 더 ``\textbf{장기 기억}''을 하려
          하므로 수렴이 \textbf{느려진다}.
        \item 작~중형 그래프: ``\textbf{100회 반복}''이면 충분.
        \item 수십억 페이지 규모의 웹: 구글이 \textbf{분산 계산 ·
          희소화} 같은 별도 공학이 필요했던 이유.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch14_3_pagerank_convergence.png}
      {\footnotesize 매 반복의 최대 변화량 max $|\Delta PR|$ 을
      로그 스케어로. $d{=}0.85$ (파랑)가 $d{=}0.5$ (주황)보다
      더 오래 걸리며, 두 곡선 모두 직선(= 지수적 감소)을 따라
      $10^{-9}$ 아래로.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{실무 1: ``수렴했는지'' 확인 없이 반복 횟수만 고정하기}
  위의 코드는 단순화를 위해 \texttt{iters=100}을 \textbf{고정}했다 ---
  실제 사용에서는 \textbf{변화량이 기준 $\varepsilon$ 아래로 떨어질
  때까지} 돌리는 것이 정석:
  \vskip0.4em
\begin{lstlisting}
def pagerank_converged(edges, n_nodes, d=0.85, tol=1e-9,
                       max_iters=1000):
    # ... (위 pagerank와 동일, 단 반복 조건이 다름)
    pr = {i: 1 / n_nodes for i in range(n_nodes)}
    for _ in range(max_iters):
        new = {i: (1 - d) / n_nodes + d * sum(pr[j] / outdeg[j]
                 for j in incoming[i]) for i in range(n_nodes)}
        delta = max(abs(new[i] - pr[i]) for i in range(n_nodes))
        pr = new
        if delta < tol:
            break
    return pr
\end{lstlisting}
\end{frame}

\begin{frame}{실무 2: dangling node (나가는 링크가 0인 노드)}
  \begin{itemize}
    \item $\max(1, \cdot)$ 가 막는 것은 출도가 0인 \textbf{전달자}가
      \textbf{분모 0}으로 만들지 않는 것이지, 그 노드 \textbf{자신}이
      받는 teleport분 $\frac{1-d}{N}$ 을 없애는 것이 \textbf{아니다}.
    \item 그래서 ``dangling 노드가 있으면 전체 PR의 합이 1이 안
      된다''는 주장은 \textbf{잘못}. (2-노드 예제 $A \to B$ 만
      있는 그래프에서도 B가 dangling이건 말건 합은 정확히 1.)
    \item 다만 출도가 0인 노드의 점수가 우변의 합에서
      \textbf{어디로도 전달되지 않는다}는 사실은 \textbf{변함없다}.
    \item 카라테 그래프에는 그런 노드가 없어 합 = 1이 정확히 나왔지만,
      웹처럼 ``\textbf{링크가 하나도 없는 죽은 페이지}''가 흔한
      그래프에서는 dangling 노드의 점수를 \textbf{모든 노드에
      균등하게 재분배}해 주는 보정이 \textbf{표준}.
  \end{itemize}
  {\footnotesize 댐핑 팩터와 같은 계열의 공학 ---
  ``\textbf{확률이 새는 구멍을 하나하나 막아주는}'' 장치들.}
\end{frame}

\begin{frame}{자주 하는 실수: ``PageRank = 도(degree)''로 읽기}
  \begin{itemize}
    \item ``가리키는/가리켜지는 링크가 많을수록 점수가 크다''는
      설명 때문에 PageRank를 ``\textbf{도 순위}''로 단순화하는 경우
     가 많다.
    \item 무방향 그래프의 \textbf{대칭화}된 PageRank에서는 둘의
      순위가 \textbf{매우 강하게} 상관난다(카라테 그래프에서도
      상위권이 거의 도 순서와 일치).
    \item 그러나 이 둘은 \textbf{같은 것이 아니다} ---
      $\sum_{j \to i} \frac{PR(j)}{\text{outdeg}(j)}$ 는
      \textbf{\textit{어떤}} 노드가 나를 가리키느냐를 \textbf{가중}.
  \end{itemize}
  \vskip0.4em
  \begin{block}{본래 설계 목적}
    방향 그래프에서는 차가 확 벌어진다:
    ``중요하지 않은 페이지 10개가 링크''보다 ``\textbf{가장 중요한
    페이지 1개가 링크}''가 더 높은 PageRank를 만드는 것.
    \textbf{인기(도)}와 \textbf{권위}(가리켜지는 쪽의 중요도)는
    다른 개념이며, PageRank가 측정하는 쪽은 \textbf{후자}.
  \end{block}
\end{frame}

  \begin{itemize}
    \item 본업: \textbf{웹 검색 랭킹}. 그러나 ``\textbf{방향 그래프
      위 무작위 걷기의 정상분포 = 중요도}''라는 틀 자체는 \textbf{범용
      도구}.
    \item 학술 \textbf{인용 네트워크}(논문 · 저널 랭킹),

    \item \kb{개인화 PageRank}(Personalized PageRank): 무작위 걷기를
      \textbf{주어진 노드에서 다시 시작}하는 변형 ---
      \textbf{추천 시스템}과 \textbf{커뮤니티 감지}에 사용.

      벡터 $\to$ 일반 ML 모델로 전달}'' 파이프라인의 첫 단계로

\begin{frame}{하나의 수학적 패턴}
  \begin{block}{공유되는 패턴}
    \kb{PCA · word2vec · Node2Vec · PageRank}는 얼핏 서로 다른
    문제처럼 보이지만, 전부
    \kq{``무언가를 어떤 변환에 \textbf{반복해서} 통과시키면
    \textbf{특별한 지점}(고유벡터, 임베딩, 정상분포)에 수렴한다''}
    는 하나의 수학적 패턴을 공유한다.
  \end{block}
  \vskip0.6em
  \begin{block}{지도 vs 비지도}
    \textbf{지도학습}이 ``\textbf{정답을 맞히는 법}''을 배운다면,
    \textbf{비지도학습}은 ``\textbf{데이터가 스스로 어떤 모양을
    하고 있는지}''를 배운다 --- 둘은 서로 다른 질문에 답한다.
  \end{block}
\end{frame}

\begin{frame}{확인 문제: 2-노드 그래프}
  2-노드 방향 그래프 $A \leftrightarrow B$ ($A \to B$, $B \to A$)에
  $d = 0.5$, $N = 2$ 로 PageRank를 손으로 계산.
  \vskip0.5em
  \begin{enumerate}
    \item \textbf{단계 1} --- 방정식 두 개:
      $PR(A) = 0.25 + 0.5\,PR(B)$, \;
      $PR(B) = 0.25 + 0.5\,PR(A)$
    \item \textbf{단계 2} --- 대칭성: $PR(A) = PR(B)$ 로 놓으면
      $PR(A) = 0.25 + 0.5\,PR(A)$, 즉
      $PR(A) = PR(B) = 0.5$ --- \textbf{$d$의 값과 무관하게}
      각 0.5. (코드로 확인)
    \item \textbf{단계 3 (핵심)} --- $B \to A$ 엣지를 지우면 (B는
      \textbf{dangling node}가 됨):
      $PR(A) = 0.25$, \;
      $PR(B) = 0.25 + 0.5 \times 0.25 = 0.375$
      --- 합이 $0.25 + 0.375 = \textbf{0.625}$ 로, \textbf{1이 아니다}.
      \kq{왜 그런지, 그리고 본문의 어떤 \textbf{공학 보정}을
      추가하면 합이 1로 돌아올지 이름으로 답하라.}
      (힌트: ``자주 하는 실수'' 섹션의 두 번째 세부 사항)
  \end{enumerate}
\end{frame}

\begin{frame}{자주 묻는 질문 (1/2)}
  \small
  \begin{itemize}
    \item \textbf{Q. PageRank도 임베딩인가요?} --- 아니요.
      Node2Vec은 각 노드를 \textbf{벡터 공간의 한 점}(임베딩)으로,
      PageRank는 각 노드에 \textbf{스칼라 값}(중요도 점수) 하나만.
      ``구조적으로 얼마나 비슷한가''(Node2Vec) vs ``전체에서 얼마나
      중요한가''(PageRank).
    \item \textbf{Q. 댐핑 팩터 $d$가 없으면?} --- $d{=}1$ 이면
      순간이동이 전혀 없어 막다른 골목에서 서퍼가 \textbf{영원히
      갇히거나}, 그래프가 여러 조각으로 나뉘어 있으면 \textbf{시작
      조각에 따라 결과}가 달라짐. $d{<}1$ 의 순간이동 확률은
      확률분포가 조각 사이를 계속 넘나들게 만들어 \textbf{유일한
      해로 수렴}을 보장.
    \item \textbf{Q. 걷기 개수·길이는?} --- 엄격한 공식은 없지만
      원본 논문의 \textbf{노드당 10개 걷기, 길이 40, 차원 128}을
      출발점으로 데이터 규모에 맞춰 줄인다(34개 노드에서는
      10개 $\times$ 길이 8 · 차원 8이면 충분). 너무 짧으면
      ``문맥''이 바로 옆 이웃에 갇히고, 너무 길면 ``출발 지점''을
      잊어 문맥이 노이즈가 된다.
  \end{itemize}
\end{frame}

\begin{frame}{자주 묻는 질문 (2/2)}
  \small
  \begin{itemize}
    \item \textbf{Q. 무방향 그래프에 써도 되나요?} --- 된다. 표준
      기법은 무방향 엣지 1개를 \textbf{방향 링크 2개}로 바꿔주는 것.
      이렇게 하면 ``도 순서에 가깝되, \textbf{누가 나를 가리키느냐를
      가중한}'' 순서가 되며, 일반적으로 도 순위와 동일하지는 않다.
    \item \textbf{Q. $p, q$는 그래프마다 튜닝해야 하나요?} ---
      보편적 최적값은 없다. 두 프리셋 $(1,16), (16,1)$ 도 ``역할
      구조 vs 커뮤니티 구조''에 맞춘 \textbf{출발점}에 불과 ---
      몇 가지 설정을 돌려보고 \textbf{downstream 태스크} 성능이나
      시각화에서 기대된 구조가 보이나로 검증한다.
  \end{itemize}
\end{frame}

\section{13.4 그래프 학습의 확장}

\begin{frame}{두 아이디어 $\to$ 세 실전 방향}
  \begin{itemize}
    \item 이 장에서 배운 두 아이디어: \kb{무작위 걷기 기반 그래프
      임베딩}(Node2Vec)과 \kb{정상분포}(PageRank).
    \item 오프너의 AlphaFold 이상으로 \textbf{훨씬 넓은 문제}에
      적용된다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{세 가지 실전 방향 (지도식 수업)}
    \begin{enumerate}
      \item \kb{(a) 단백질}: 구조를 알면, 그 구조로 기능을 분류
    \item \kb{(b) 금융}: 거래 그래프에서 자금세탁 탐지
      (IBM AML World)
      \item \kb{(c) 교통}: 도로망 + 시계열의 스페이오템포럴 그래프
        교통 예측(STGCN)
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{(a) 단백질: 구조를 알면, 그 구조로 기능을 분류한다}
  \begin{itemize}
    \item 오프너의 \kb{AlphaFold}는 잔기 쌍 사이의 \textbf{거리 ·
      각도 관계}를 그래프로 표현해 3차원 구조를 예측.
    \item 구조를 알고 나면 자연스러운 다음 질문:
      \textbf{``이 구조는 어떤 기능을 하는가?''}
    \item \kb{DeepFRI}는 단백질 3차원 구조에서 얻은 \kb{잔기 접촉
      그래프}(contact graph)를 \kb{그래프 신경망}(GCN)에 넣어,
      그 단백질이 어떤 \textbf{생물학적 기능}을 갖는지 분류.
  \end{itemize}
  \vskip0.5em
  \begin{block}{자연스러운 파이프라인}
    \textbf{구조 예측(AlphaFold)} $\to$ \textbf{구조 기반 기능
    분류(DeepFRI)}
  \end{block}
\end{frame}

\begin{frame}{이 장의 도구가 어디에 쓰이는가}
  \begin{itemize}
    \item 구조를 3차원 좌표로 돌려받는 순간, ``\textbf{공간적으로
      가까운 잔기 쌍}''을 엣지로 묶으면 \kb{접촉 그래프}가
      만들어진다.
    \item 카라테 클럽의 친분 엣지를 ``친하다''로 바꾼 것 같은 ---
      \textbf{도메인 지식이 만든 그래프}.
    \item 그래프 신경망 계열은 이 그래프에 13장 전체를 관통한
      ``\textbf{이웃과 값을 합치면서 반복한다}''는 연산 ---
      \kb{그래프 컨볼루션}(graph convolution) --- 을 적용.
    \item 각 잔기(노드)의 벡터를 \textbf{이웃 잔기의 정보와 섞어
      몇 라운드 갱신}하면, ``이 잔기는 구조적으로 어떤 위치에 있는가''
      가 인접성만큼 벡터에 스며든다.
    \item DeepFRI는 이렇게 쌓인 노드 표현을 받아 \textbf{수십만 개의
      기능 라벨}(GO term) 중 해당 단백질을 설명하는 것을 골라내는
      \textbf{분류 문제}로 만든다.
  \end{itemize}
\end{frame}

\begin{frame}{왜 서열을 바로 분류하지 않고 구조 그래프를 거치는가}
  \begin{block}{``왜?''를 한 번}
    동일한 기능을 가진 단백질은 \textbf{서열이 많이 달라}도
    \textbf{3차원 형태(구조)는 비슷하게} 유지되는 경우가 흔하다.
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item 구조 그래프는 ``\textbf{서열과 무관한 기능의 공통 요인}''을
      더 \textbf{직접적으로} 담는 표현.
    \item 같은 구조에 다른 서열의 단백질(\textbf{유사 동종단백질})
      기능을 추론할 때 \textbf{서열 기반 모델보다 유리}.
    \item AlphaFold의 구조 예측이 생물학 난제를 풀기만 한 게 아니라,
      \kq{\textbf{그래프 기반 추론의 입력을 대량으로 공급하는
      인프라}}가 된 이유.
  \end{itemize}
\end{frame}

\begin{frame}{노트북 Part 1: 파이프라인의 뼈대}
  \begin{itemize}
    \item 작은 \textbf{합성 접촉 그래프}(20개 노드)를 만들고.
    \item ``\textbf{접촉 수(도)를 특징으로 한}'' 매우 간단한
      분류기를 돌린다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{주의: 이것은 DeepFRI의 GCN을 직접 학습하는 것이
  아니다}
    \textbf{``그래프 구조 특징 $\to$ 기능 분류'' 파이프라인의
    뼈대}를 느끼는 데모.
  \end{block}
\end{frame}

\begin{frame}{(b) 금융: 거래 그래프에서 자금세탁 탐지}
  \begin{itemize}
    \item \kb{IBM AML World}: \textbf{계좌를 노드, 거래를 엣지}로
      하는 \textbf{대규모 합성} 거래 그래프 데이터셋.
    \item 자금세탁은 흔히 ``여러 계좌를 거쳐 돈의 출처를 세탁''하는
      패턴(\kb{레이어링})으로 나타나는데 ---
    \item \textbf{개별 거래 하나만 봐서는 안 보이고},
      \textbf{그래프 구조 전체}(누가 누구와 얼마나 자주, 어떤 패턴으로
      거래하는가)를 봐야 드러난다.
    \item Node2Vec 같은 \textbf{그래프 임베딩}이나 그래프 신경망으로
      ``\textbf{의심스러운 부분그래프}''를 탐지하는 방향의
      \textbf{대표적 실전 응용}.
  \end{itemize}
\end{frame}

\begin{frame}{왜 합성 데이터셋인가 + 레이어링의 특징}
  \begin{block}{합성인 것의 의미}
    ``\textbf{답이 알려진} 거래 그래프''를 만들어서 탐지 모델의
    성능을 \textbf{측정}할 수 있게 하는 것 --- 실제 금융 데이터는
    \textbf{프라이버시} 때문에 쉽게 공개될 수 없으니,
    \textbf{구조와 노이즈 수준만 조절}해서 ``탐지 가능한
    패턴''을 심어놓은
    합성 데이터셋이 연구용으로 나온 것.
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item 레이어링 패턴의 특징: 불법 자금의 흐름이 \textbf{1--2홉이
      아니라 3--4홉 이상}으로 뻗어 있다.
    \item ``계좌 A가 B로, B가 C로, C가 해외로'' 이동하는 동안
      \textbf{각 구간은 정상 거래와 구분하기 어렵다}.
  \end{itemize}
\end{frame}

\begin{frame}{이 장의 두 도구를 거래 그래프에}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{\color{ksablue}노드 임베딩 (Node2Vec 계열)}
      \begin{itemize}
        \item 계좌를 ``거래 네트워크에서 \textbf{어떤 위치에
          있는가}''로 벡터화.
        \item 동일한 패턴으로 세탁에 쓰인 계좌들이 임베딩 공간에서
          \textbf{같은 군집}에 모이면, 그 군집 자체가
          ``의심 패턴''의 \textbf{시그니처}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\color{ksablue}중요도 점수 (PageRank 계열)}
      \begin{itemize}
        \item ``다른 계좌들이 \textbf{가장 많이 거쳐가는 통로}''에
          높은 점수.
        \item 자금세탁의 \textbf{중계 계좌}(여러 출처를 여러
          목적지로 합치고 흩어주는 허브) = 정확히 그런
          \textbf{구조적 허브}.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.4em
  \begin{block}{두 신호를 결합하면}
    ``무슨 특징인지 아직 이름이 붙지 않았지만'' \textbf{구조적으로
    수상한 계좌}를 선별 --- 규칙 기반 탐지(``거래액 $>$ 1억이면
    신고'')가 놓치는 \textbf{패턴 전체가 정상인 개별 구간}까지
    커버.
  \end{block}
\end{frame}

\begin{frame}{노트북 Part 2: 합성 거래 그래프}
  \begin{itemize}
    \item 작은 \textbf{합성 거래 그래프}(20개 계좌, \textbf{정상
      클러스터} + \textbf{의도적으로 심은 레이어링 경로})를
      만들고.
    \item \textbf{PageRank}로 ``\textbf{중계 허브}''를 찾아보며.
    \item \textbf{1홉 특징만으로는 왜 발견 못 하는지} 보여준다.
  \end{itemize}
\end{frame}

\begin{frame}{(c) 교통: 도로망 위의 시계열 --- spatio-temporal 그래프}
  \begin{itemize}
    \item 지금까지: 그래프 \textbf{구조 자체가 고정}
      (카라테 클럽의 친분 관계는 안 바뀜).
    \item 교통 예측처럼 \textbf{그래프 위에 시계열이 얹히는}
      문제도 있다:
    \item \textbf{도로망(그래프)은 고정}이지만, 각 도로 구간의
      \textbf{교통량은 시간에 따라 변한다}.
    \item \kb{STGCN}(Spatio-Temporal Graph Convolutional Network):
      \textbf{그래프 컨볼루션}(공간 방향: ``이웃 도로의 정보를
      합친다'')와 \textbf{시계열 모델}(시간 방향: ``과거 몇 스텝을
      본다'')을 결합.
  \end{itemize}
  \vskip0.5em
  \begin{block}{주의}
    비슷한 계열의 \kb{DCRNN}도 있는데, 그 논문의
    ``\textbf{diffusion convolution}''은 15장 생성모델의
    diffusion과는 \textbf{다른 개념} --- 여기서는
    \textbf{그래프 위의 확산 과정}(무작위 걷기와 비슷한 발상)을
    가리킨다.
  \end{block}
\end{frame}

\begin{frame}{왜 두 방향이 모두 필요한가}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{\color{ksablue}공간 방향만} --- ``이웃의 교통량을
      합친다''
      \begin{itemize}
        \item \textbf{시간에 따른 변화}를 담을 수 없다:
          출근 · 퇴근 시간대의 \textbf{파도}, 신호에 따른
          \textbf{국소 정체}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\color{ksablue}시간 방향만} --- ``이 도로의 과거
      교통량만 본다''
      \begin{itemize}
        \item \textbf{다른 도로의 영향}을 놓친다: 인접 구간의 정체가
          이 구간으로 \textbf{전파}.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.5em
  도로망은 그 \textbf{자신이 그래프}다 --- 교차로가 노드이고
  도로가 엣지.
\end{frame}

\begin{frame}{STGCN: 공간 $\times$ 시간을 순환적으로 결합}
  \begin{enumerate}
    \item 한 단계에서 (1) \textbf{공간 방향}으로 이웃 도로의 정보를
      합치고,
    \item (2) \textbf{시간 방향}으로 과거 몇 단계의 \textbf{자기
      자신}을 본다.
  \end{enumerate}
  \vskip0.5em
  \begin{itemize}
    \item 이것을 \textbf{반복}하면 ``\textbf{인접 도로의 과거 흐름이
      이 도로의 미래 흐름에 어떤 영향을 미치는가}''를 학습.
    \item 이 ``\textbf{공간 $\times$ 시간 순환}'' 구조는 13장 전체의
      ``\textbf{반복해서 값을 갱신한다}''는 패턴의 자연스러운 확장
      (\textbf{노드 반복 $\to$ 노드 + 시간 반복}).
  \end{itemize}
\end{frame}

\begin{frame}{노트북 Part 3: 합성 도로망}
  \begin{itemize}
    \item 작은 \textbf{합성 도로망}($4\times 4$ 격자, 교차로 = 노드)에서
      인공적인 ``\textbf{출근 파도}'' 시계열을 만들고.
    \item 두 모델을 비교:
    \item (a) \textbf{1홉 이웃 평균}만 보는 모델
    \item (b) \textbf{3홉 이웃 평균 + 과거 3단계 자기 평균}을
      보는 모델
    \item 결과: \textbf{이웃 범위를 넓히고 과거를 함께 보면}
      \textbf{오차가 줄어든다}.
  \end{itemize}
\end{frame}

\begin{frame}{재료가 그대로 실전에 옮겨진다}
  \begin{itemize}
    \item 세 케이스 모두, 이 장의 ``재료''인 \kb{그래프 자체}가
      \textbf{도메인 지식에서 직접 만들어지는} 형태
      (접촉 그래프, 거래 그래프, 도로망).
    \item 이 장의 ``연산''인 \kb{반복해서 이웃과 값을 합친다}는
      것이 각 도구의 \textbf{핵심}에 들어 있다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{그래프 학습에서 ``모델링''이 ``프로그래밍''보다
  중요한 이유}
    도메인이 달라질수록 \textbf{그래프를 어떻게 만드는가}
    (\textbf{어떤 것을 노드로, 어떤 것을 엣지로} 하는가)의 선택이
    \textbf{모델의 성능을 결정}한다.
  \end{block}
\end{frame}

  \end{block}
\end{frame}

\begin{frame}{이번 주차 정리}

    \item \kb{13.1 무작위 걷기 = 마르코프 체인} ---

      DFS형(커뮤니티) 다이얼.

      금융(AML World) · 교통(STGCN). ``반복해서 이웃과 값을
