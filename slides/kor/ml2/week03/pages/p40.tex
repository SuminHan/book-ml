\begin{frame}{베이스라인 해석: 같은 $\gamma$ 안에서만 비교}
  \begin{itemize}
    \item \textbf{버틴 스텝 수는 $\gamma$ 와 무관하게 평균 24.1스텝}
      (1.3절의 베이스라인).
    \item \textbf{리턴의 숫자는 $\gamma$ 에 따라 8.62 $\to$ 13.07
      $\to$ 20.77로 크게 바뀐다}.
    \item 이는 ``할인율이 보상의 \textbf{총량}이 아니라,
      \kb{언제의 보상을 얼마나 헤아리는가}를 정한다''는 뜻.
  \end{itemize}
  \vskip0.5em
  \begin{block}{주의: 서로 다른 ``화폐 단위''}
    $\gamma = 0.9$ 의 ``평균 8.62''와 $\gamma = 0.99$ 의 ``평균
    20.77''을 \textbf{같은 스케일로 비교하면 안 된다} --- 서로
    다른 \textbf{화폐 단위(할인 기준)}의 숫자.
    \vskip0.2em
    {\small \kq{같은 $\gamma$ 안에서만} 리턴의 크기를 비교하는 것이
    정석.}
  \end{block}
  \vskip0.4em
  {\footnotesize 이 ``화이트라벨'' 같은 주의는 Week 5--7에서
  \textbf{리플레이 버퍼}에 여러 환경$\cdot$다른 $\gamma$ 로 모은
  전이를 함께 쓸 때 실제로 문제가 된다.}
  \vskip0.3em
  {\footnotesize 그림(오른쪽): 매 스텝 보상 1의 10스텝 에피소드에서
  ``할인 없이 더함''은 $\gamma$ 와 무관하게 10, ``$\gamma=0.95$ 의
  정확한 할인''은 8.03, ``두 번 할인''은 6.58 --- 두 번 할인은
  사실상 $\gamma^2$ 를 할인율로 쓰는 것과 같으므로 곡선이 가장
  빨리 0으로 곤두박질한다.}
\end{frame}
