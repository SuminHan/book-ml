\begin{frame}{자주 하는 실수 (3.3)}
  \kb{실수 1: ``다음 상태의 가치''에 할인율 $\gamma$ 를 빼먹기}
  \begin{itemize}
    \item $V(s) = R(s) + V(s')$ 처럼 $\gamma$ 를 빼면, ``당장의
      보상''과 ``아주 먼 미래의 보상''을 \textbf{동등하게} 취급.
    \item 결정적 2상태 예에서 $\gamma$ 를 빼면 $V_0 = 1 + V_1$,
      $V_1 = 2 + V_0$ 가 되어 $V_0 - V_1 = 1$ 이면서도 $V_1 - V_0 =
      2$ 가 되어 \textbf{모순(해 없음)}이 된다 --- $\gamma < 1$ 이
      없으면 이 단순한 2상태 예의 방정식에도 해가 없다.
  \end{itemize}
  \vskip0.4em
  \kb{실수 2: ``가치함수''를 ``실제 에피소드 리턴''과 혼동}
  \begin{itemize}
    \item $V^\pi(s)$ 는 \textbf{기댓값}. ``상태 $s$ 에서 시작하면
      리턴이 \textbf{반드시} $V^\pi(s)$ 가 된다''가 \textbf{아니다}.
    \item ``가치가 14.737인 상태''에서 10번 시작하면 10번 중 10번이
      정확히 14.737이 아니라, 10번의 \textbf{평균이 대략} 14.737에
      가까울 뿐. (단, 결정적 MDP$\cdot$정책에서는 기댓값
      $=$ 실제값이 일치 --- 위 2상태 예가 그런 경우.)
  \end{itemize}
  \vskip0.4em
  \kb{실수 3: ``해는 하나''라고 단정하지 않기}
  \begin{itemize}
    \item $\gamma < 1$ 이면 고정점이 \textbf{유일}. \textbf{초기값에
      무관하게 같은 답}으로 수렴하는 것이 \textbf{정상}.
    \item 만약 ``초기값에 따라 다른 값으로 수렴했다''면:
      (1) $\gamma \geq 1$ 로 설정, (2) 코드에 할인율 곱셈이
      빠져(실수 1), (3) 전이확률 합이 1이 아닌 잘못된 MDP 중
      하나.
    \item \kq{``초기값 무관 수렴''은 벨만방정식 코드가 \textbf{맞는지}
    를 확인하는 간단한 건강체크.}
  \end{itemize}
\end{frame}
