\begin{frame}{확인 문제: MDP인가, 마르코프인가? (2/2)}
  \begin{itemize}
    \item \textbf{(c) 슬롯머신(Week 2의 밴딧)}
      \begin{itemize}
        \item 답: \kb{MDP이지만 특수한 경우} ---
          $\mathcal{S} = \{s_0\}$ (상태 1개),
          $P(s_0|s_0, a) = 1$ for all $a$.
        \item 밴딧은 ``\textbf{전이 없는 MDP}''로, 이 절의 모든
          정의가 \textbf{퇴화된 형태로} 적용(FAQ 참고).
      \end{itemize}
    \item \textbf{(d) 체스에서 다음 수를 고르는 문제(중반)}
      \begin{itemize}
        \item 답: \kb{MDP} --- $\mathcal{S} =$ 합법적인 모든 판
          배치(크기 $\sim 10^{40}$ 이상), $\mathcal{A} =$ 그 배치의
          합법 수, $P$ = 행동 후 판 배치(\textbf{결정적}이라 확률 1),
          $R =$ 매 스텝 0, 체크메이트 시 $\pm 1$.
        \item 체스는 마르코프 성질이 \textbf{정확히} 성립하는
          \textbf{희귀한 예} --- ``어떤 수순을 거쳐왔는지''가 다음
          선택지에 영향을 주지 않으므로.
      \end{itemize}
  \end{itemize}
  \vskip0.6em
  \begin{block}{네 문항의 공통 기준}
    \kq{``현재 상태(관측)만 알면, 앞으로 일어날 것의 전체
    확률분포가 결정되는가?''} 아니라면 hidden state가 있다는
    뜻이고, 상태 재설계(전략 1 또는 2)가 필요하다.
  \end{block}
\end{frame}
