    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch02_1_epsilon_tradeoff.png}
      \vskip0.3em
      {\footnotesize 왼쪽: 평균 총보상, 오른쪽: 평균 후회.
      $\varepsilon=0$ 은 후회 1754로 폭주, $\varepsilon=0.5$ 는

    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{세 가지 관찰}:
      \begin{enumerate}
        \item \textbf{$\varepsilon=0$ 의 후회 1754} ---
          ``단 한 번의 나쁜 운''이 약 2000스텝 $\times$ 2.0의
          절반에 달하는 손실. $\varepsilon=0.1$ 만 달이면
          후회 $\approx$137로 \textbf{약 13배} 감소.
        \item \textbf{최대가 $\varepsilon=0.05$} (3872.4).
          0.10(3862.8)은 ``아주 비슷''한 2위. ``$\varepsilon=0.1$
          이 마법 수''가 아니라 \textbf{문제에 따라 움직이는}
          절절점.
        \item \textbf{$\varepsilon=0.01$ 도 16\% 실패} ---
          ``좀만 섞어주면 되겠지''가 ``영원히 박히기''를
          \textbf{완전히} 막아주진 못함.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}
