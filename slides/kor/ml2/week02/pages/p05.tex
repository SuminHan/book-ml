\begin{frame}{밴딧 문제의 정적 설정}
  $k$개의 팔(arm), 각 팔 $a$는 \textbf{모르는} 진짜 평균 보상
  $q^*(a)$를 가진 ``보상 기계''.
  \vskip0.5em
  \begin{itemize}
    \item 매 스텝 $t = 1, 2, \ldots, T$: 팔 하나를 골라 당기고
      보상 $R_t$를 받는다.
    \item $R_t \sim \mathcal{N}(q^*(a_t),\, 1)$ --- 평균 $q^*(a_t)$,
      표준편차 1의 정규분포.
    \item \textbf{판별 가능한 신호는 ``평균'' 하나뿐} ---
      매번 받는 보상은 잡음에 가려져 있다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{목표}
    $T$스텝 동안의 \kb{총보상} $\sum_{t=1}^{T} R_t$ 를 \textbf{최대화}.
  \end{block}
\end{frame}
