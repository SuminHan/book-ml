\begin{frame}{비선형 활성이 있는 층을 쌓으면}
  \begin{itemize}
    \item 은닉 유닛 \textbf{하나}가 ``$x$가 이 방향에 있는가''를 확인하는
      한 개의 \kb{선형 경계}(hyperplane)를 만듦.
    \item 16개 유닛 $\to$ \textbf{16개의 서로 다른 경계}.
    \item 경계들이 $x$ 공간을 \textbf{여러 조각}(각각 다른 선형 영역)으로
      쪼개고, 마지막 선형 층이 각 조각 안에서
      \textbf{각각 다른 기울기}의 직선을 이어붙임.
  \end{itemize}
  \vskip0.6em
  \begin{block}{깊이가 곧 표현력의 지수}
    층이 하나 더 쌓일 때마다 이 ``조각 쪼개기''가
    \textbf{기하급수적으로} 세짐.
  \end{block}
\end{frame}
