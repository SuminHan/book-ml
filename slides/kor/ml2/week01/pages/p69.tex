\begin{frame}[fragile]{관찰 공간과 행동 공간: \texttt{Box}와 \texttt{Discrete}}
\begin{lstlisting}
env = gym.make("CartPole-v1")
obs, _ = env.reset(seed=0)
print("관측 공간 :", env.observation_space)
print("행동 공간 :", env.action_space)
print("초기 관측 :", obs)
\end{lstlisting}
  {\scriptsize\texttt{관측 공간 : Box([-4.8 -inf -0.41887903 -inf],
  [4.8 inf 0.41887903 inf], (4,), float32)} \\[0.2em]
  {\scriptsize\texttt{행동 공간 : Discrete(2)} \quad
  \texttt{초기 관측 : [ 0.0137 -0.023  -0.0459 -0.0483]}}
  \vskip0.2em
  \begin{itemize}
    \item \textbf{관측은 4차원} $[x,\,\dot{x},\,\theta,\,\dot{\theta}]$
      (카트 위치$\cdot$속도, 막대 각도$\cdot$각속도). \texttt{Box}로 각
      성분의 상$\cdot$하한($\texttt{low}/\texttt{high}$)을 표현.
      $x \in [-4.8, 4.8]$, $\theta \in [-0.419, 0.419]$,
      속도$\cdot$각속도는 $\pm\infty$.
    \item \textbf{행동}은 \texttt{Discrete(2)} --- \textbf{이산} 두 개
      (0: 왼쪽, 1: 오른쪽). \texttt{action\_space.n == 2}로 확인.
  \end{itemize}}
\end{frame}
