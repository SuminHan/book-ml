\begin{frame}{표의 각 행이 예고하는 한 장 (1)}
  \begin{itemize}
    \item \kb{``데이터를 스스로 만들어낸다''} --- 에이전트가
      \textbf{경험}을 생산. 시작에서 종료까지의 상태-보상-행동 일련을
      \kb{에피소드}(episode)라 부름.
      $\Rightarrow$ \textbf{3주차}에서 MDP 정식화로 발전.
    \item \kb{``보상(reward)''} --- 한 스텝의 점수.
      에이전트가 실제로 최대화하는 대상은 \kb{할인된 리턴(return)}
      \[G_t = R_t + \gamma R_{t+1} + \gamma^2 R_{t+2} + \cdots\]
      $\Rightarrow$ \textbf{3주차}.
  \end{itemize}
\end{frame}
