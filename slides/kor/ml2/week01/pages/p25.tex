\begin{frame}{표의 각 행이 예고하는 한 장 (2)}
  \begin{itemize}
    \item \kb{``지금 행동이 미래를 바꾼다''} ---
      \kb{마르코프 결정과정(MDP)}이 등장하는 이유.
      ``현재 상태가 미래의 전부''(마르코프 성질) +
      ``행동이 전이를 바꾼다''(결정적 구조). $\Rightarrow$ 3주차.
    \item \kb{``탐험 vs 활용''} --- 형태를 바꿔가며 \textbf{학기 내내} 등장:
      \begin{itemize}
        \item 2주차(밴딧): 가장 순수한 형태
        \item 6주차(TD): $\varepsilon$-greedy
        \item 9주차(DQN): 탐색 전략
        \item 11주차(PPO): 엔트로피 보너스
      \end{itemize}
  \end{itemize}
\end{frame}
