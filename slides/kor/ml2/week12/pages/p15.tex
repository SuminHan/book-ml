\begin{frame}{네 줄의 각 — 역할 분담이 DAgger의 정체}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{②가 전문가 정책으로 간다면}
      \begin{itemize}
        \item = ``시연을 더 많이 모으는 것''과 같음
        \item 위에서 본 바와 같이 \textbf{도움이 안 됨}
      \end{itemize}
      \vskip0.6em
      \textbf{③이 다른 상태에 라벨을 찍는다면}
      \begin{itemize}
        \item 배포 시 마주칠 확률이 \textbf{0인 상태}에 시간을 쓰는 것
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{정체}
        \textbf{에이전트가 운전하고, 전문가는 내비게이션만}
        해주는 구조.
      \end{block}
      \vskip0.6em
      \begin{itemize}
        \item 배포 오차 $\mathbb{E}_{s\sim d_\pi}L(s)$를 줄이려면
          학습 데이터가 $d_\pi$ 위에 놓여야 한다.
        \item $\pi$는 매 라운드 바뀐다 — DAgger는 ``달리는 말 위에서''
          이 문제를 \textbf{반복으로 안정화}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
