\section{7.1 결정 트리: 구조, 지니불순도, 정보이득}

\begin{frame}{스무고개에서 결정 트리}
  ``동물인가요? 네 다리로 걷나요? 야옹 소리를 내나요?'' ---
  \kb{스무고개(20 Questions)}는 예/아니오 질문을 잘 골라 20번 안에 답을
  좁혀내는 게임이다.
  \vskip0.8em
  \begin{itemize}
    \item \kb{결정 트리}(Decision Tree) = 이 게임을 \textbf{알고리즘}으로
      그대로 옮긴 것
    \item ``좋은 질문''을 고르는 기준(지니불순도 감소량)을 수학으로 정의하면,
      컴퓨터가 \textbf{스스로} 가장 잘 좁히는 질문을 찾아 나무 형태로 쌓는다
    \item 은행은 대출 거절 시 법(ECOA)에 따라 \textbf{구체적 사유}를
      알려줘야 한다 --- 트리 하나는 경로를 읽으면 되지만, GBDT는 ``왜?''에
      답하기 훨씬 어렵다
  \end{itemize}
  \vskip0.8em
  \kq{질문을 얼마나 ``잘 좁히는가''를 숫자로 재는 기준은 무엇인가?}
\end{frame}

\begin{frame}{결정 트리의 구조: 이진나무}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \kb{내부 노드} = 하나의 질문 (예: ``$x_2 > 5$?'')
        \item \kb{리프}(leaf) 노드 = \textbf{예측값}
          (분류: 클래스, 회귀: 숫자)
        \item 예측 = 루트에서 시작해 예/아니오에 따라 \textbf{아래로 내려가
          리프에 닿을 때까지} 경로를 따르는 것뿐
        \item 예: 6개 샘플 $x=[1,2,3,4,5,6]$, 라벨 $[A,A,A,B,B,B]$를
          ``$x \le 3$?'' \textbf{한 개의 질문}만으로 완벽 분류
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch07_1_decision_tree_structure.png}
      \vskip0.3em
      {\footnotesize 루트 ``$x \le 3$?'' — 예(1,2,3)$\to$ 클래스 A,
      아니오(4,5,6)$\to$ 클래스 B. 두 리프 모두 순수($G=0$).}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{경로 = 설명: 결정 트리의 강점}
  \begin{itemize}
    \item 새 샘플 들어오면: (1) 루트에서 질문을 평가 $\to$ (2) 예/아니오에
      따라 좌·우 자식으로 이동 $\to$ (3) 리프면 그 값을 내놓고 끝
    \item 이 \textbf{경로 자체가} ``왜 이렇게 예측했는가''의 설명이다
    \item 대출 심사 예: ``소득 $<$ 3000만원 $\to$ 연령 $>$ 45세 $\to$
      \textbf{거절}''이라는 경로를 \textbf{그대로 읽어}주면 됨
  \end{itemize}
  \vskip0.8em
  \begin{block}{\textbf{설명가능성}(interpretability)}
    루트에서 리프까지의 질문 경로를 사람이 읽을 수 있다. 이 장(7.3)에서는
    \textbf{수백 개의 트리}가 합쳐지면 이 ``경로 = 설명'' 특성을 잃는
    대가를 다시 논한다(SHAP).
  \end{block}
\end{frame}

\begin{frame}{왜 질문은 언제나 ``축에 평행''한가}
  \begin{itemize}
    \item 질문은 \textbf{특징 하나}만 쓴다: ``$x_j > t$?'' 의 형태
    \item 여러 특징을 섞은 ``$x_1 + x_2 > 5$?'' 같은 \textbf{대각선 경계}는
      한 번에 만들 \textbf{수 없다}
    \item 그래서 결정 경계는 축에 평행한 직선 조각이 이어진
      \kb{계단}(staircase)으로밖에 나오지 않는다
    \item 이 제약은 \textbf{강점}(가독성, 스케일 불변)이자 동시에
      \textbf{약점}(대각선 경계에 서툴다)
  \end{itemize}
  \vskip0.8em
  {\footnotesize 7.2절에서 ``트리 하나는 계단, 숲은 부드러운 곡선''을 볼 때
  이 계단이 재등장한다.}
\end{frame}

\begin{frame}{역사: 스무고개에서 CART까지}
  ``질문을 잘 골라 답을 좁혀간다''는 아이디어는 1960$\sim$80년대에
  통계학과 인공지능이 겹치는 장소에서 여러 이름으로 태어났다.
  \vskip0.8em
  \small
  \begin{tabular}{@{}llp{7.2em}@{}}
    \toprule
    시기 & 이름 & 내용 \\
    \midrule
    1960s & CHAID & 마케팅 연구: 고객을 ``질문으로 쪼개는'' 분할 \\
    1975 & TDIDT & Quinlan: 예·아니오 질문 트리로
      분류 문제를 푼다는 틀의 정식화 \\
    1984 & CART & Breiman 외: 회귀를 같은 틀로 확장, 지니불순도

    1986 & ID3 & Quinlan: 정보이득이 가장 큰 질문부터 나누는
      \textbf{탐욕스러운} 알고리즘 \\
    \bottomrule
  \end{tabular}
  \vskip1em
  \begin{block}{핵심}
    사람이 ``잘 좁히는 질문''을 \textbf{직감}으로 고르던 것을,
    ``지니불순도/엔트로피 감소량''이라는 \textbf{숫자}로 바꿔 컴퓨터가
    매 단계에서 가장 큰 것을 고르게 한 것. 그 숫자가 이 절의 전부다.
    (scikit-learn의 \texttt{DecisionTreeClassifier}는 CART 계열.)
  \end{block}
\end{frame}

\begin{frame}{지니불순도: 노드가 얼마나 ``섞여'' 있나}
  \begin{itemize}
    \item 스무고개를 잘하는 사람은 ``생물인가요?''처럼 \textbf{반반으로
      갈릴 만한} 질문을 먼저 던진다
    \item 그래서: ``노드가 얼마나 섞여 있는가(불순도)''를 재고,
      \textbf{나누기 전 불순도 $-$ 나눈 뒤 불순도}가 클수록 좋은 질문
  \end{itemize}
  \vskip0.8em
  \kb{지니불순도}(Gini Impurity): 클래스 $K$개, 클래스 $k$의 비율이 $p_k$일 때
  \[ G = 1 - \sum_{k=1}^{K} p_k^{\,2} \]
  \begin{itemize}
    \item 한 클래스로 \textbf{순수}하면($p_k=1$): $G = 0$ — 가장 좋은 상태
    \item \textbf{반반}이면($K=2$, $p_1=p_2=0.5$): $G = 1-0.25-0.25 = 0.5$
      — 이진 분류의 최댓값
  \end{itemize}
\end{frame}

\begin{frame}{지니불순도의 두 가지 해석}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \kb{(1) ``잘못 고를 확률'' 해석}
      \begin{itemize}
        \item 노드에서 항목을 \textbf{무작위 2개} 뽑아 \textbf{다른 클래스}일
          확률이 정확히 $G$
        \item 이진: $1 - p^2 - (1-p)^2 = 2p(1-p)$
        \item 순수($p=1$) $\to$ ``다를 확률 0'', 반반($p=0.5$) $\to$ 최대
          0.5
        \item 즉 $G$ = ``이 노드의 예측이 \textbf{틀릴 확률}''
      \end{itemize}
      \vskip0.5em
      \kb{(2) ``스무고개 정보'' 해석}
      \begin{itemize}
        \item 질문이 ``클래스1인가?''일 때, 답이 \textbf{반반으로 갈릴수록}
          정보가 가장 크다
        \item $G = 2p(1-p)$는 $p=0.5$에서 최대인 포물선
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch07_1_gini_entropy_curves.png}
      \vskip0.3em
      {\footnotesize 둘 다 $p=0.5$에서 최대, 순수에서 0.
      지니는 매끄러운 포물선, 엔트로피는 반반 근처가 더 완만.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{정보이득: 나누기 전 엔트로피 $-$ 가중평균}
  \kb{엔트로피}(entropy, Ch.\,2.3 섀넌 정보이론):
  \[ H = -\sum_{k=1}^{K} p_k \log_2 p_k \]
  지니불순도와 마찬가지로 순수할수록 0, 섞일수록 크다.
  \vskip0.6em
  노드를 왼쪽($L$)과 오른쪽($R$)으로 나눴을 때 \kb{정보이득}(Information Gain):
  \[ \text{IG} = H(\text{parent})
    - \left( \tfrac{|L|}{|L|+|R|} H(L) + \tfrac{|R|}{|L|+|R|} H(R) \right) \]
  \begin{itemize}
    \item IG가 클수록 그 질문이 데이터를 잘 나눴다는 뜻
    \item 트리를 만들 때 각 단계에서 \textbf{IG(또는 지니 감소량)가 가장 큰
      질문}을 고른다 — 스무고개의 ``가장 잘 좁히는 질문을 먼저''이
      알고리즘이 된 것
  \end{itemize}
\end{frame}

\begin{frame}{두 가지 ``왜'': 크기 가중과 $\log_2$}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \kb{왜 자식의 불순도를 ``크기로 가중''하는가}
      \begin{itemize}
        \item 자식 중 하나가 1개, 다른 하나가 99개 샘플일 때,
          1개짜리 순수 노드의 ``$G=0$''을 무겁게 세면 \textbf{불공정}
        \item 그래서 각 자식의 불순도를 \textbf{샘플 수 비율}로 곱해
          평균한다 — ``가중평균''의 이유 전부
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \kb{엔트로피의 밑은 왜 $\log_2$인가}
      \begin{itemize}
        \item 단위가 \textbf{비트}(bit): ``이 항목의 클래스를 알려면
          예/아니오 질문을 \textbf{평균 몇 개} 해야 하는가''의 답
        \item 반반($p=0.5$) $\to$ 1비트, 한쪽으로 쏠려 있으면 $\approx 0$비트
        \item 다른 밑을 써도 단위만 바뀌고 \textbf{분기 비교의 순서는
          변하지 않음}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{세 함수를 손으로 구현}
\begin{lstlisting}
def gini(labels):
    n = len(labels)
    if n == 0:
        return 0
    counts = {}
    for l in labels:
        counts[l] = counts.get(l, 0) + 1
    return 1 - sum((c / n) ** 2 for c in counts.values())

def entropy(labels):
    n = len(labels)
    if n == 0:
        return 0
    counts = {}
    for l in labels:
        counts[l] = counts.get(l, 0) + 1
    return -sum((c / n) * math.log2(c / n) for c in counts.values())

def information_gain(parent_labels, left_labels, right_labels):
    n = len(parent_labels)
    weighted_child = (len(left_labels) / n) * entropy(left_labels) + \
                      (len(right_labels) / n) * entropy(right_labels)
    return entropy(parent_labels) - weighted_child
\end{lstlisting}
  {\footnotesize \texttt{gini} = ``잘못 고를 확률'' 해석, \texttt{entropy} =
  ``비트 수'' 해석, \texttt{information\_gain} = IG 식(자식 엔트로피의
  \textbf{크기 가중평균}을 뺀 값)을 그대로 구현.}
\end{frame}

\begin{frame}{손으로 한 번: 두 분기 후보 비교}
  샘플 6개: $x = [1,2,3,4,5,6]$, 라벨 $[A,A,A,B,B,B]$.
  부모 노드는 반반이므로 $G_{\text{parent}} = 1 - 0.5^2 - 0.5^2 = 0.5$.
  \vskip0.8em
  \small
  \begin{tabular}{@{}llcc@{}}
    \toprule
    분기 후보 & 왼쪽 / 오른쪽 & 가중평균 불순도 & Gain \\
    \midrule
    $x \le 3$ & 3A / 3B (모두 순수, $G=0$) & $0$ & \textbf{0.5} \\
    $x \le 1$ & $[A]$ / $[A,A,B,B,B]$ ($G_R = 1-0.4^2-0.6^2 = 0.48$)
      & $\approx 0.4$ & $0.1$ \\
    \bottomrule
  \end{tabular}
  \vskip1em
  \begin{itemize}
    \item \kq{같은 데이터인데, 어디를 자르느냐에 따라 정보이득이
      5배(0.5 vs 0.1) 차이난다} — 후보 1이 훨씬 ``좋은 질문''
    \item 엔트로피 기준도 같은 결론: 부모 $H = 1$(비트),
      후보 1: $1.0$, 후보 2: $\approx 0.191$
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{코드로 확인: 손계산과 정확히 일치}
\begin{lstlisting}
X = [1, 2, 3, 4, 5, 6]
y = ["A", "A", "A", "B", "B", "B"]
print("parent  Gini=%.3f  Entropy=%.3f" % (gini(y), entropy(y)))
for t in (1, 3):
    L = [y[i] for i in range(6) if X[i] <= t]
    R = [y[i] for i in range(6) if X[i] > t]
    wG = (len(L)/6)*gini(L) + (len(R)/6)*gini(R)
    wH = (len(L)/6)*entropy(L) + (len(R)/6)*entropy(R)
    print(f"t={t}: Gini gain={gini(y)-wG:.3f} | Ent gain={entropy(y)-wH:.3f}")
\end{lstlisting}
  \vskip0.3em
  {\footnotesize \texttt{parent Gini=0.500 Entropy=1.000} \quad
  \texttt{t=1: Gini gain=0.100 | Ent gain=0.191} \\
  \texttt{t=3: Gini gain=0.500 | Ent gain=1.000} --- 손계산과
  \textbf{정확히} 일치.}
  \vskip0.4em
  \begin{itemize}
    \item \texttt{best\_split}은 \textbf{모든 threshold를 하나씩} 시도해
      최대 IG를 고른다 — 모든 특징에 걸쳐 반복하는 것이
      \texttt{DecisionTreeClassifier.fit} 각 노드에서 일어나는 일
    \item 연속값 특징: 인접 두 값의 \textbf{중간}만 후보면 충분
      $\to$ $n$개 고유값 특징당 후보 $n-1$개
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{실습: sklearn으로 트리를 배우고 ``읽기''}
  유방암 진단 데이터(양성/악성, 30개 특징), train/test 7:3, \textbf{깊이 3}.
\begin{lstlisting}
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier, export_text

data = load_breast_cancer()
X_train, X_test, y_train, y_test = train_test_split(
    data.data, data.target, test_size=0.3, random_state=0)

tree = DecisionTreeClassifier(max_depth=3, random_state=0).fit(X_train, y_train)
print(export_text(tree, feature_names=list(data.feature_names), max_depth=2))
\end{lstlisting}
  \vskip0.3em
  {\footnotesize
  \texttt{|--- worst concave points <= 0.14} \;
  \texttt{|--- worst concave points > 0.14} \ldots \\
  \texttt{train acc: 0.967 \quad test acc: 0.947}}
  \vskip0.4em
  \begin{itemize}
    \item \texttt{export\_text}가 \textbf{알고리즘이 스스로 골라낸 질문}을
      그대로 보여준다
    \item 루트가 \texttt{worst concave points}인 것 = \textbf{정보이득이
      가장 큰 질문}을 자동 탐색해 루트에 놓은 것
    \item 이 트리 하나는 사람이 \textbf{그대로 읽을 수 있다}:
      ``0.14 이하이면(악성 쪽) worst area로 한 번 더 나누고…''
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{예측 경로 추적 + 특징 중요도}

path = tree.decision_path(X_one)
print("경로를 지난 노드:", path.indices)     # 루트부터 리프까지 노드 번호

  \vskip0.3em
  {\footnotesize \texttt{worst concave points 0.817} \quad
  \texttt{worst area 0.098} \quad \texttt{area error 0.054}}
  \vskip0.4em
  \begin{itemize}
    \item \texttt{path.indices} = ``이 환자가 악성으로 예측된 \textbf{이유}''
      — 7.3절 SHAP은 \textbf{수백 개의} 트리 경로를 합성해야 하지만,
      \textbf{트리 하나}의 설명은 이 한 줄이면 끝
    \item \texttt{feature\_importances\_} = 각 특징이 분기에 쓰일 때마다

    \item 단, ``이 트리 전체''의 설명이지 ``한 환자의 예측'' 설명은 아님
      (SHAP이 그 격차를 메운다)
  \end{itemize}
\end{frame}

\end{frame}

\begin{frame}{언제 나누기를 멈추는가: 정지 조건}
  \begin{itemize}
    \item 끝까지 키우면 모든 리프가 순수해진다 — 학습 데이터는 100\%
      맞히지만 새 데이터에 취약한 \textbf{과적합} 트리
    \item \kb{정지 조건}(stopping criteria) 세 가지:
    \begin{enumerate}
      \item \texttt{max\_depth} — 트리의 \textbf{높이} 상한
      \item \texttt{min\_samples\_leaf} — \textbf{리프에 들어갈 최소 샘플 수}
        (2개뿐인 리프는 ``무엇이든 1:1 예측''하는 노이즈 외우기)
      \item \texttt{min\_samples\_split} — \textbf{분기할 수 있는 최소
        샘플 수}
    \end{enumerate}
    \item 또는 트리를 다 키운 뒤 불필요한 가지를 자르는 \kb{가지치기}
      (pruning)
  \end{itemize}
  \vskip0.8em
  \begin{block}{트리에서의 ``정규화''}
    셋 다 ``더는 나누지 마''라는 같은 신호 — Ch.\,6의 정규화강도 $\lambda$에
    대응한다. \textbf{트리에서 정규화란 트리를 작게(간단하게) 유지하는
    것}이다.
  \end{block}
\end{frame}

\begin{frame}{U자 곡선: 편향-분산 트레이드오프}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch07_1_depth_ucurve.png}
      \vskip0.3em
      {\footnotesize train은 1.0까지 단조 증가, test는 depth 2$\sim$6에서
      정점 후 depth 8에서 0.912로 떨어져 U자.}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{itemize}
        \item \textbf{왼쪽}(깊이 1$\sim$2, 너무 얕은): train·test 모두 낮음 —
          \kb{편향(bias)} 영역. 질문 1개(스텀)로는 부족
        \item \textbf{가운데}(깊이 2$\sim$6, 적당히): test \textbf{0.947}
          정점, 그 부근에서 평탄하게 머문다 — \textbf{최적 깊이}
        \item \textbf{오른쪽}(깊이 8$\sim$None, 너무 깊은): train
          \textbf{1.000}(100\% 외움) vs test \textbf{0.912} —
          \kb{분산(variance)} 영역
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.6em
  \begin{block}{실무 규칙}
    최적 깊이는 \textbf{test 정확도의 정점}(depth 2$\sim$6, 0.947).
    ``train이 1.0인 가장 깊은 트리''를 고르면 U자 \textbf{오른쪽 끝}
    (0.912)에 떨어져 \textbf{3.5\%p 나빠진다} — 검증셋(또는 7.2의 OOB)으로
    이 U자를 그려 골라야 한다.
  \end{block}
\end{frame}

\begin{frame}{자주 하는 실수 1: \texttt{max\_depth}를 제한 없이 두기}
  \small
  \begin{tabular}{@{}lcc@{}}
    \toprule
     & \texttt{max\_depth=None} & \texttt{max\_depth=3} \\
    \midrule
    train & \textbf{1.000} & 0.967 \\
    test & 0.912 & \textbf{0.947} \\
    \bottomrule
  \end{tabular}
  \vskip1em
  \begin{itemize}
    \item 깊이 제한이 없는 트리는 학습 데이터를 \textbf{완벽히}(100\%)
      외웠지만, 새 데이터에서는 오히려 \textbf{더 나쁨}(91.2\% vs 94.7\%)
    \item ``학습 데이터를 더 정확히 맞히는 트리가 더 좋은 트리''라는 직관이
      여기서 \textbf{완전히} 틀린다 — Ch.\,4.1 편향-분산 트레이드오프가
      트리에서 그대로 재현
    \item 깊이 제한이 없으면 분산이 극단적으로 커지고(노이즈 하나까지
      다 기억), 적당한 제한은 약간의 편향을 감수해 \textbf{분산을 크게 줄임}
    \item \kq{정지 조건은 선택 사항이 아니라, 반드시 검증셋으로 튜닝해야
      하는 \textbf{필수} 하이퍼파라미터}
  \end{itemize}
\end{frame}

\begin{frame}{자주 하는 실수 2: ``IG $> 0$''이면 무조건 나누기}
  \begin{itemize}
    \item 함정: 리프에 5개 샘플(4A, 1B)이 있는데 ``그 1개 B를 떼어내자''
      $\to$ IG는 양수(0.32 $\to$ 0)
    \item 하지만 \textbf{샘플 1개짜리 리프}를 만든 것은 그 1개 샘플의
      \textbf{라벨링 노이즈를 그대로 외운 것}
    \item 새 데이터에서 그 위치의 ``진짜'' 클래스가 A라면 틀린다
  \end{itemize}
  \vskip0.8em
  \begin{itemize}
    \item 막는 것: \texttt{min\_samples\_leaf}(리프 최소 샘플 수),
      \texttt{min\_samples\_split}(분기 최소 샘플 수)
    \item \texttt{max\_depth}는 트리의 \textbf{높이}를,
      \texttt{min\_samples\_*}는 \textbf{리프의 크기}를 막는다
    \item \texttt{max\_depth}만으로도 상당한 효과; \textbf{클래스 불균형}
      (99:1)이나 \textbf{고차원}(특징 수백 개)이면
      \texttt{min\_samples\_leaf}까지 함께
  \end{itemize}
  \vskip0.6em
  \begin{block}{수정된 규칙}
    ``IG가 양수이면 나눠도 된다''가 아니라, ``IG가 \textbf{충분히} 양수
    \textbf{이고} 나눈 리프가 \textbf{충분히 크고}''일 때만 나누는 것 —
    CART 탐욕스러운 분기의 현실적 한계.
  \end{block}
\end{frame}

\begin{frame}{자주 묻는 질문}
  \small
  \begin{itemize}
    \item \textbf{지니불순도 vs 엔트로피?} 실전에서 거의 같은 분기를 고른다.
      지니는 \textbf{로그 계산이 없어 더 빠르다} $\to$ scikit-learn 등
      기본값. (반반 근처에서 엔트로피는 더 완만해, 가끔 분기가 달라질
      수도.)
    \item \textbf{왜 스케일링이 필요 없나?} kNN/SVM은 거리를 쓰니
      스케일에 민감했지만, 결정 트리는 ``$t$보다 큰가?''라는
      \textbf{순서(비교) 정보}만 쓴다 — 2배로 늘려도, 미터를 센티미터로
      바꿔도 순서 관계가 안 변한다.
    \item \textbf{연속 특징도 예/아니오로?} 그렇다 — 인접 고유값 \textbf{중간}만
      후보 threshold($n$개 고유값 $\to$ $n-1$개). 실습의
      \texttt{worst concave points <= 0.14}가 자동 골라진 cutoff.
    \item \textbf{회귀에도?} CART = Classification and \textbf{Regression}
      Trees — 리프 = 라벨 \textbf{평균}, 분기 기준 = \textbf{제곱오차
      감소}(SSE 감소). 7.3절 GBDT에서 재등장.
    \item \textbf{탐욕스러운 분기 = 전역 최적?} \textbf{아님} — 매 단계
      ``그 순간 IG 최대''일 뿐. 전역 최적은 \textbf{지수 시간}에 탐색
      불가 $\to$ ``충분히 좋고, 빠르다''는 탐욕 근사. 이 한계가
      \textbf{랜덤 포레스트(7.2)와 GBDT(7.3)로 넘어가는 동기}.
  \end{itemize}
\end{frame}

\section{7.2 트리 하나에서 숲으로: 랜덤 포레스트}

\begin{frame}{트리 하나에서 \kb{숲}(랜덤 포레스트)으로}
  \begin{itemize}
    \item 트리 하나는 데이터를 \textbf{완벽히 외워버리기}(과적합) 쉽다
    \item \kb{랜덤 포레스트}(Random Forest, Breiman 2001)의 아이디어:
      서로 조금씩 다른 트리를 \textbf{여러 개} 만들고,
      \textbf{다수결}로 예측한다
  \end{itemize}
  \vskip0.8em
  \begin{block}{핵심 직관}
    \textbf{``완전히 같은 실수를 하는 전문가 100명''}보다
    \textbf{``서로 다른 실수를 하는 전문가 100명''}의 평균이 훨씬 안정적이다.
    두 가지 무작위성이 트리들을 서로 덜 닮게 만들어, 개별 트리의 과적합
    성향이 다수결로 \textbf{상쇄}된다.
  \end{block}
\end{frame}

\begin{frame}{두 가지 무작위성}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \kb{1. 배깅}(Bootstrap Aggregating)
      \begin{itemize}
        \item 각 트리는 원본에서 \textbf{복원추출}(중복 허용)로
          \textbf{같은 크기}의 데이터를 뽑아 학습
        \item $\to$ 트리마다 \textbf{조금씩 다른 데이터}를 봄
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \kb{2. 특징 무작위화}
      \begin{itemize}
        \item 각 분기(split)에서 \textbf{전체 특징이 아니라}
          무작위로 고른 \textbf{일부 특징} 중에서만 최선의 질문을 찾음
        \item $\to$ 같은 데이터를 봐도 \textbf{다른 질문}을 탐색
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.8em
  이 둘이 ``트리들이 서로 닮지 않게'' 만드는 방식:
  하나는 \textbf{데이터}를, 다른 하나는 \textbf{각 분기에서 고려하는
  특징}을 무작위화한다.
\end{frame}

\begin{frame}{역사 1: 골튼의 소 체중 맞추기 (1889)}
  \begin{itemize}
    \item 1889년, 영국 통계학자 프랜시스 골튼(Francis Galton)은 시골
      박람회의 \textbf{``시상 소'' 체중 맞추기} 이벤트 결과를 조사
    \item 787명이 쓴 추정치의 \textbf{평균}($1{,}251$파운드)은 실제
      ($1{,}198$파운드)과 불과 \textbf{1\% 이내} — 개별 추정치 중 거의
      어느 것보다 정확
  \end{itemize}
  \vskip0.8em
  \begin{block}{\textbf{군중의 지혜}(wisdom of crowds)}
    ``많은 사람 각각은 틀리지만 \textbf{그 평균은} 놀라울 만큼 정확하다.''
    골튼의 ``평균이 개인을 이긴다''는 발견은, 개별 트리의 흔들림(분산)을
    \textbf{다수결로 없애겠다는 배깅}의 목표와 정확히 같은 말.
  \end{block}
\end{frame}

\begin{frame}{역사 2: 부트스트랩(1993) $\to$ 배깅 $\to$ Breiman(2001)}
  \begin{itemize}
    \item 골튼의 원리는 약 100년 뒤, \kb{부트스트랩}(bootstrap,
      Efron \& Tibshirani 1993)이라는 도구로 돌아옴:
      \textbf{복원추출}로 원본과 같은 크기의 샘플을 계속 뽑으면 그
      통계량 분포가 \textbf{진짜 분포를 근사}
    \item 랜덤 포레스트는 이 부트스트랩을 \textbf{트리의 학습 데이터}에
      그대로 적용 — 각 트리가 ``다르게 보게 만드는 것'' =
      \kb{배깅}(B\"uhlmann \& H\"usler 1997)
    \item 마지막 조각, \textbf{특징 무작위화}는 2001년 Breiman의
      ``Random Forests''에서 완성:
      \textbf{배깅만으론}(특정 강한 특징이 있으면) 트리들이 너무
      닮는다는 문제를, ``각 분기에서 \textbf{특징의 무작위 하위 집합}만
      고려하자''는 두 번째 무작위성으로 풀었다
  \end{itemize}
  \vskip0.6em
  {\footnotesize (부트스트랩 + 특징 무작위화)의 \textbf{조합}이 바로 우리가
  쓰는 랜덤 포레스트다.}
\end{frame}

\begin{frame}[fragile]{실습: 트리가 어떤 질문을 던지나 (복습)}
  유방암 데이터, 깊이 3 — 7.1절과 같은 코드:
\begin{lstlisting}
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier, export_text

data = load_breast_cancer()
X_train, X_test, y_train, y_test = train_test_split(
    data.data, data.target, test_size=0.3, random_state=0)

tree = DecisionTreeClassifier(max_depth=3, random_state=0).fit(X_train, y_train)
print(export_text(tree, feature_names=list(data.feature_names), max_depth=2))
print("트리 정확도:", round(tree.score(X_test, y_test), 3))
\end{lstlisting}
  \vskip0.3em
  {\footnotesize 루트가 \texttt{worst concave points}(가장 오목한 부분의
  정도)로 먼저 나누는 것이 보임 — 정보이득이 가장 큰 질문을 알고리즘이
  스스로 찾아낸 것.}
  \vskip0.4em
  \begin{itemize}
    \item 대출 심사 예시처럼, 이 \textbf{트리 하나}는 ``왜 이렇게
      예측했는가''를 사람이 \textbf{그대로 읽을 수 있다}
  \end{itemize}
\end{frame}

\begin{frame}{손으로 한 번: ``여러 전문가의 평균''이 더 안정적이는가}
  같은 유방암 학습 데이터에서 \textbf{부트스트랩 샘플 5개}로 깊이 제한
  없는 트리를 \textbf{독립적으로} 5개 학습 (개별 트리 흔들림을 선명하게
  보기 위해):
  \vskip0.6em

    \toprule
     & tree 0 & tree 1 & tree 2 & tree 3 & tree 4 \\
    \midrule
    test acc & 0.871 & 0.918 & 0.895 & 0.889 & 0.883 \\
    \bottomrule
  \end{tabular}
  \vskip1em
  \begin{itemize}
    \item 5개는 \textbf{87.1\% $\sim$ 91.8\%}로 흩어짐
      (평균 89.1\%, \textbf{표준편차 1.55\%p}) — 어떤 부트스트랩 샘플을
      뽑았느냐에 따라 꽤 요동침
    \item 이 5개를 \textbf{100개 트리의 다수결}로 합친 랜덤 포레스트:

    \end{center}

\begin{frame}{왜? — 앙상블 분산 공식}
  Ch.\,1.1의 \kb{편향-분산 분해}: 예측 오차 = \textbf{편향}(본질적으로 못
  맞히는 부분) + \textbf{분산}(데이터가 조금만 바뀌어도 예측이 흔들리는
  부분). 깊이 제한이 없는 결정 트리는 \textbf{분산이 매우 큼}
  (7.1: \texttt{max\_depth=None} 91.2\% $<$ 깊이 3의 94.7\%).
  \vskip0.8em
  같은 설계의 트리를 $T$개, 각각 다른 부트스트랩 샘플로 학습.
  각 트리의 test 예측 오차 표준편차 $\sigma$, 두 트리 간 상관계수 $\rho$라면
  \textbf{$T$개 평균(다수결 앙상블)의 오차 분산}:
  \[ \mathrm{Var}[\text{앙상블}]
     = \frac{\sigma^2}{T}\,\big(1 + \rho\,(T-1)\big) \]
  (개별 예측 \textbf{평균}의 분산 공식: $T\sigma^2$ + $T(T-1)$개의
  공분산 $\rho\sigma^2$, $T^2$로 나누면 됨.)
\end{frame}

\begin{frame}{공식을 극한으로 — 그리고 $\rho$를 손으로}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{두 극한}
      \begin{itemize}
        \item \textbf{완전히 독립}($\rho=0$): 분산 $\sigma^2/T$,
          표준편차 $\sigma/\sqrt{T}$로 \textbf{줄어든다}
        \item \textbf{완전히 같으면}($\rho=1$): 분산 $\sigma^2$
          \textbf{그대로} — ``완전히 같은 실수를 하는 전문가 100명''은
          전문가 1명과 \textbf{같다}
      \end{itemize}
      \vskip0.5em
      현실은 둘 사이($0<\rho<1$):
      \textbf{무작위성이 클수록} $\rho$가 작아져 $\sigma/\sqrt{T}$에
      가까워진다.
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{손으로}
      \begin{itemize}
        \item 개별 트리 $\sigma \approx 1.55\%$p, 독립이라면 100개 평균은
          $\tfrac{1.55}{\sqrt{100}} = 0.155\%$p
        \item 실측: 5번 다른 시드 포레스트가 \textbf{0.68\%p}로 모였다
          — ``완전히 독립(0.155\%p)은 아니지만 개별 트리보다
          훨씬 안정''
        \item 0.68\%p를 공식에 대입하면
          $\rho \approx 0.2$ — 같은 강한 특징을 다 보지만 \textbf{상관이
          낮은 쪽}
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.6em
  \begin{block}{결론}
    두 가지 무작위성의 역할 = 정확히 이 \textbf{$\rho$를 줄이는 것}.
    $\rho$가 1에 가깝지 않다는 것이 ``앙상블이 단일 트리보다 좋은 이유''의
    \textbf{전부}.
  \end{block}
\end{frame}

\begin{frame}{결정 경계를 눈으로: ``계단''에서 ``곡선''으로}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch07_2_boundary_tree_vs_rf.png}
      \vskip0.3em
      {\footnotesize 왼쪽: 트리 하나(깊이 3), 오른쪽: 랜덤 포레스트(100개).
      2개 특징 300개 샘플.}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{itemize}
        \item 왼쪽: \textbf{축에 평행한 직선 조각이 이어진 계단형} —
          ``$x_1 > t$?'' 같은 축에 평행한 질문만 하므로 필연적
        \item 오른쪽: 100개 트리의 \textbf{다수결} $\to$ 각 트리의 계단이
          조금씩 달라 \textbf{훨씬 부드러운 곡선}
        \item 이 ``부드러짐'' = 앞 절 \textbf{분산 감소의 시각화}:

          (높은 분산), 100개 계단의 평균은 잡음이 \textbf{상쇄}되어

      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.5em
  {\footnotesize ``트리가 하나면 계단, 숲이면 곡선''이라는 앙상블의 본질을
  2D 데이터에서 눈으로 직접 보는 장면.}
\end{frame}

\begin{frame}{OOB(out-of-bag): 검증 세트 없이도 일반화 성능을}
  \begin{itemize}
    \item 각 트리는 \textbf{복원추출}된 부트스트랩 샘플로 학습하는데,
      복원추출은 원본 샘플 중 \textbf{약 37\%를 ``한 번도 안 고른다''}
    \item 그 37\% = \kb{OOB}(out-of-bag, bag 밖) 샘플 — 해당 트리가
      \textbf{보지 못한} 샘플
    \item 이 트리가 ``보지 못한'' 샘플을 예측하게 하면, 학습을 마치자마자
      \textbf{새 데이터에 대한 예측}을 할 수 있다 — \textbf{별도 검증
      세트를 떼어놓을 필요가 없음}
  \end{itemize}
  \vskip0.8em
  \begin{block}{실무적 중요성}
    \textbf{데이터가 적어 검증 세트를 따로 떼기 아까울 때}에도 ``이 모델이
    과적합이 아닌가''를 \textbf{학습 중 즉시} 확인할 수 있다.
  \end{block}
\end{frame}

\begin{frame}[fragile]{OOB 실습: test와 거의 붙는다}
\begin{lstlisting}
rf = RandomForestClassifier(n_estimators=100, oob_score=True, random_state=0)
rf.fit(X_train, y_train)
print("OOB score:", round(rf.oob_score_, 3))
print("test score:", round(rf.score(X_test, y_test), 3))
\end{lstlisting}
  \vskip0.3em
  {\footnotesize \texttt{OOB score: 0.955 \quad test score: 0.959}}
  \vskip0.5em
  \begin{itemize}
    \item OOB 점수(0.955)가 진짜 test(0.959)와 \textbf{거의 붙는다}
    \item \textbf{주의}: \texttt{oob\_score\_}는 \texttt{oob\_score=True}로
      학습했을 때만 \textbf{속성}으로 생기며, \texttt{oob\_score()}라는
      \textbf{메서드가 아니다}
  \end{itemize}
\end{frame}

        \item 표준편차(오차막대)는 트리가 늘수록 \textbf{줄어든다} —

        \item 정확도가 \textbf{떨어지는 지점이 없다} — 트리를 더

\begin{frame}{숲 전체가 어떤 특징을 보는가: 중요도}

        \item 100개 트리의 그것 = \textbf{각 트리가 분기마다 줄인

          ``숲 전체가'' 어떤 특징을 보는지
        \item 유방암: \texttt{worst perimeter}(0.151) $>$
          \texttt{worst concave points}(0.135) $>$
          \texttt{mean concave points}(0.131)
        \item ``worst(최악)'' 특징의 상위권 = 실제 진단에서
          ``가장 나쁜 상태''가 가장 결정적이었음을 뜻함
      \end{itemize}

\begin{frame}[fragile]{특징 무작위화: ``강한 특징''이 루트를 독점하는가}
  배깅만 하는 \texttt{BaggingClassifier}(특징 무작위화 \textbf{없음})와
  랜덤 포레스트(\textbf{있음})를 50개 트리로 만들고, \textbf{각 트리가
  루트에서 어떤 특징을 쓰는지} 세면:
\begin{lstlisting}
dom = list(data.feature_names).index("worst concave points")
bag = BaggingClassifier(
    estimator=DecisionTreeClassifier(max_depth=6, random_state=0),
    n_estimators=50, random_state=0).fit(X_train, y_train)
rf = RandomForestClassifier(n_estimators=50, max_depth=6, random_state=0) \
       .fit(X_train, y_train)

root_frac = lambda clf: np.mean([t.tree_.feature[0] == dom
                                 for t in clf.estimators_])
print("bagging-only:  ", round(root_frac(bag), 2))
print("random forest: ", round(root_frac(rf), 2))
\end{lstlisting}
  \vskip0.3em
  {\footnotesize \texttt{bagging-only: 0.40 \quad random forest: 0.20}}
  \vskip0.4em
  \begin{itemize}
    \item 배깅만: 50개 중 \textbf{40\%}가 루트에서
      \texttt{worst concave points} 사용 $\to$ RF는 \textbf{절반}(20\%)
    \item 이유: RF는 분기마다 30개 중 무작위 일부(기본
      \texttt{max\_features=$\sqrt{30}\approx$5})만 후보로 보므로 강한
      특징이 ``이번엔 후보에 안 들어가는'' 경우가 생김
    \item 결과: 트리들이 \textbf{서로 다른 질문}을 탐색 $\to$ $\rho$를
      낮추어 분산 감소를 \textbf{더 크게}
  \end{itemize}
\end{frame}

\begin{frame}{자주 하는 실수: \texttt{n\_estimators}를 너무 적게}
  \begin{itemize}
    \item 배깅의 효과는 트리 개수가 늘수록 (수확체감이 있긴 하지만)
      \textbf{꾸준히 안정} — $\to$ \texttt{n\_estimators=3}처럼 적게 쓰면
      앞 실습의 ``개별 트리 5개 흩어짐''과 크게 다르지 않은 \textbf{불안정}
      결과 (``전문가''가 너무 적으면 소수 의견에 쉽게 흔들림)
    \item 실전: \texttt{n\_estimators=100} 이상을 \textbf{기본}으로 시작,
      검증 성능이 ``정체 구간''이 되는 지점을 확인
  \end{itemize}
  \vskip0.8em
  \begin{block}{GBDT와의 대비}
    트리를 더 추가하는 것은 (\textbf{GBDT와 달리}) 과적합을 일으키지
    않으므로, ``너무 많이 쓰는'' 걱정은 \textbf{훨씬 적어도 됨}
    (다만 학습·예측 \textbf{속도}만 느려진다).
  \end{block}
\end{frame}

\begin{frame}{자주 묻는 질문}
  \small
  \begin{itemize}
    \item \textbf{두 무작위성 모두 필요한가?} 배깅만으론, \textbf{특정
      강한 특징}이 있으면 샘플이 달라져도 거의 모든 트리가 그 특징으로
      루트를 나눠 \textbf{너무 닮아버림} $\to$ 특징 무작위화까지 하면
      어떤 트리는 그 특징을 아예 후보에서 못 봄 (앞의 40\%$\to$20\%).
    \item \texttt{max\_features}는? \textbf{분류} 기본값 $\sqrt{n}$,
      \textbf{회귀} 기본값 $n/3$. 특징 1000개면 $\sqrt{\,}\to$ 31개.
      기본값은 \textbf{출발점}일 뿐 — 검증셋으로 $\sqrt{\,}$, $\log_2$,
      $1.0$(전체)를 비교. 특징이 독립적 $\to$ 더 많이(값 크게),
      상관관계 강하면 $\to$ 덜.
    \item \texttt{predict\_proba}는? 각 트리의 \textbf{클래스 확률
      평균} — 다수결(\texttt{predict})이 ``클래스를 더 많이 말았는가''를
      세는 반면 부드러운 \textbf{확률}을 줌. OOB 계산·임계값 튜닝에
      쓰임.
    \item \textbf{사람이 ``이 예측이 왜?'', 읽을 수 있나?} \textbf{아니요}
      — 100개 트리는 개별 예측 하나에 어떤 트리의 어떤 특징이 기여했는지
      눈으로 따지기 어렵다. \textbf{7.3의 SHAP}이 정확히 이 문제를 푼다.
    \item \texttt{feature\_importances\_}? 같은 방식(지니 감소 누적)이나,
      RF는 \textbf{독립적} 학습 / GBDT는 \textbf{순차적}(잔차) 학습이라
      같은 데이터에도 \textbf{중요도 순위가 달라질 수 있다}.
  \end{itemize}
\end{frame}

\section{7.3 GBDT와 SHAP}

\begin{frame}{다른 조직 방식: \kb{GBDT}}
  7.2의 랜덤 포레스트 = \textbf{``서로 독립적인 전문가 100명의 다수결''.}
  \kb{GBDT}(Gradient Boosted Decision Trees, Friedman 2001)는 전혀 다른
  조직 방식:
  \vskip0.8em
  \begin{center}
    \textbf{``선임이 실수를 남기면, 다음 직원이 바로 그 실수를 고치는'' 순환 근무}
  \end{center}
  \vskip0.8em
  \begin{itemize}
    \item 랜덤 포레스트: 트리를 \textbf{독립적으로} 만들고 \textbf{다수결}
    \item GBDT: 트리를 \textbf{하나씩 순차적으로} 추가하고,
      \textbf{매번} 새 트리는 ``지금까지의 예측이 아직 못 맞힌 부분
      (\kb{잔차}, residual)''을 목표로 학습
  \end{itemize}
\end{frame}

\begin{frame}{GBDT: 순차적으로 잔차를 추격}
  \begin{itemize}
    \item 목표: $T$개 트리 $f_1,\dots,f_T$를 \textbf{순차적으로 더}해
      예측 $F_T(x) = \sum_{t=1}^{T} f_t(x)$
    \item $t$번째 트리 학습 시: 지금까지 쌓은 예측 $F_{t-1}$의 \textbf{잔차}
      \[ r^{(i)} = y^{(i)} - F_{t-1}(x^{(i)}) \]
      을 \textbf{새 목표값}으로 삼아 그 잔차를 예측하는 트리를 학습
  \end{itemize}
  \vskip0.6em
  \begin{itemize}
    \item \kb{learning\_rate}(축소율, shrinkage): 각 트리의 기여를 일부러
      \textbf{줄여서} 한 트리가 과적합하는 것을 막고, 여러 트리가 조금씩
      나눠서 학습하게 함 — Ch.\,2 경사하강법의 \textbf{학습률}과 같은 역할
    \item ``잔차를 향해 조금씩 이동'' = \textbf{함수 공간에서의 경사하강법}
      $\to$ 그래서 이름이 \textbf{gradient} boosting
    \item \kb{XGBoost}(2016), \kb{LightGBM}(2017) = 실무 최적화 라이브러리,
      지금도 \textbf{정형(tabular) 데이터 대회(Kaggle)}에서 가장 자주
      우승하는 축
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{핵심 루프: 잔차를 목표로 트리 하나씩}
\begin{lstlisting}
def gbdt_fit(X, y, n_trees, learning_rate):
    trees = []
    predictions = [0.0] * len(y)  # F_0(x) = 0
    for t in range(n_trees):
        residuals = [y[i] - predictions[i] for i in range(len(y))]
        tree = fit_single_tree(X, residuals)  # 잔차를 목표로 트리 하나
        trees.append(tree)
        for i in range(len(y)):
            predictions[i] += learning_rate * tree.predict(X[i])
    return trees
\end{lstlisting}
  \vskip0.5em
  \begin{itemize}
    \item 실무 라이브러리의 스텝 단위는 보통 \textbf{깊이 3$\sim$6짜리
      작은 트리}(아래 1차원 실습은 스템/깊이 1) — 핵심은 ``각 스텝이
      \textbf{잔차를 목표로 작은 함수 하나}를 추가한다''는 구조, 스텝의
      크기는 하이퍼파라미터
    \item $F_{t-1}$이 어느 정도 맞히고 있다면 남은 잔차(오차)는 원래 $y$보다
      \textbf{작다} — 그 작은 오차를 새 트리가 다시 줄이면
      $F_t = F_{t-1} + \eta f_t$의 오차는 \textbf{한 단계 더 작아짐}
    \item 계속 추가하면 학습 데이터 오차는 이론상 계속 줄지만, 실전은
      \textbf{검증 성능이 나빠지기 시작하는 시점}(과적합)에서
      \textbf{early stopping}(Ch.\,6 검증 데이터 원칙과 동일)
  \end{itemize}
\end{frame}

\begin{frame}{왜 ``Gradient''인가: 잔차 = (음의) 기울기}
  회귀에서 손실을 \textbf{제곱오차} $L(F) = \tfrac{1}{2}\sum_i (y^{(i)} -
  F(x^{(i)}))^2$로 두면, $L$을 $F(x^{(i)})$로 미분하면:
  \[ \frac{\partial\, L}{\partial\, F(x^{(i)})}
     = F(x^{(i)}) - y^{(i)} = -\,r^{(i)} \]
  \textbf{기울기는 정확히 잔차의 음수}(제곱손실 기준).
  \vskip0.8em
  \begin{itemize}
    \item GBDT 스텝을 다시 쓰면:
      $F_t(x) = F_{t-1}(x) - \eta\, g_t(x)$,
      여기서 $g_t$는 $-r^{(i)}$(= 기울기)를 맞추는 트리
    \item Ch.\,2 경사하강법 $\theta \leftarrow \theta - \eta\nabla_\theta L$
      과 \textbf{구조가 완전히 같은 식} — 차이 하나: 경사하강법은
      \textbf{파라미터 공간}에서 $\theta$, GBDT는 \textbf{함수 공간}에서
      $F$ 자체를 옮김
    \item \textbf{Friedman(2001)의 핵심 통찰}: ``함수를 파라미터로 가지는
      모델을 경사하강으로 학습한다'' — 손실만 미분가능하면 어떤 손실이든
      적용 가능(분류는 로지스틱 손실의 기울기 $\to$
      \texttt{predict\_proba}가 나오는 분류기)
  \end{itemize}
\end{frame}

  \texttt{mean concave points 0.275}, \texttt{worst perimeter 0.071}}

    \item GBDT가 RF보다 \textbf{근소하게 더 높음}(0.977 vs 0.959) —

    \item \texttt{feature\_importances\_}는 \textbf{모델 전체}의 설명이지

  \end{itemize}

\begin{frame}[fragile]{GBDT만이 치르는 가격: overfitting 곡선}

  학습/100 검증)을 \textbf{500단계} 학습, \texttt{staged\_predict}로
  \textbf{매 단계} train/test MSE 추적:

reg = GradientBoostingRegressor(n_estimators=500, learning_rate=0.1,

\end{lstlisting}

  \texttt{500단계까지 계속하면: train\_MSE=0.005 \quad test\_MSE=1.290}}

    \end{column}
    \begin{column}{0.45\textwidth}

      \begin{enumerate}
        \item \textbf{train MSE는 0.005까지 계속 하락} — 500개 트리가
          학습 데이터의 잡음 하나까지 \textbf{거의 완벽히 외움}
        \item \textbf{test MSE는 10단계에서 최저(0.727)}
        \item \textbf{500단계까지 하면 1.290까지 다시 상승} — 최저점 대비
          \textbf{약 두 배}로 악화, ``더 오래 학습한 모델''이 정점의
          \textbf{절반 수준}
      \end{enumerate}
    \end{column}
  \end{columns}
  \vskip0.5em
  \textbf{왜 GBDT만?} RF의 101번째 트리는 데이터를 \textbf{처음부터 다시}
  보고 다른 트리의 노이즈 오류를 ``따라잡아야 할 의무''가 없다(다수결이
  희석). GBDT의 101번째 트리는 앞 100개 합이 \textbf{아직 못 맞춘 것
  (= 학습 데이터의 노이즈 잔차)을 \textbf{의도적으로} 목표}로 삼는다 —
  적당히 쌓이면 잔차 대부분은 \textbf{라벨링 노이즈}인데, GBDT는 그것을
  ``정당한 학습 목표''로 받아들인다.
\end{frame}

\begin{frame}{실무 규칙: early stopping이 필수}
  \begin{itemize}
    \item \texttt{n\_estimators}는 RF처럼 ``많을수록(안정적으로) 좋다''가
      \textbf{아님} — 검증 곡선의 \textbf{정점(최저 MSE)을 찾아 거기서
      멈추는 early stopping이 필수}(sklearn \texttt{validation\_fraction}
      옵션으로 내장)
    \item \texttt{learning\_rate}를 \textbf{작게}(0.1 $\to$ 0.01) 하면 각
      스텝이 작아져 같은 수준까지 \textbf{더 많은 단계}가 필요 $\to$
      정점의 위치·높이가 \textbf{완만해져 early stopping이 정밀}해짐
  \end{itemize}
  \vskip0.8em
  \begin{block}{표준 워크플로}
    \textbf{``learning\_rate를 작게 하고, n\_estimators를 early stopping으로
    결정''} — 실무 표준.
  \end{block}
  \vskip0.6em
  \begin{itemize}
    \item 이 ``정점을 지나 다시 올라가는 곡선'' = Ch.\,1.1 \textbf{편향-분산
      트레이드오프}: 트리 적으면(불충분 학습) \textbf{편향}이 크고,
      많으면(과적합) \textbf{분산}이 큼
    \item RF는 오른쪽(분산) 끝이 \textbf{거의 평탄}해서 ``많게 해두면
      된다'', GBDT는 \textbf{가파르게 올라서} ``정점을 찾아 멈춰야 한다''
  \end{itemize}
\end{frame}

\begin{frame}{정확한데 설명이 안 된다는 문제 $\to$ \kb{SHAP}}
  GBDT는 강력하지만 \textbf{수백 개의 트리가 얽혀} 있어 ``이 예측 하나''에
  어떤 특징이 얼마나 기여했는지 알기 어렵다.
  \vskip0.6em
  \begin{itemize}
    \item \kb{SHAP}(SHapley Additive exPlanations) = \kb{게임이론의
      섀플리값}(Shapley value)을 빌려, 예측 하나를 \textbf{``각 특징이
      기여한 양''으로 정확히 분해}
    \item 원래 섀플리값 = ``여러 플레이어가 \textbf{협력 게임}에 참여할
      때, 각 플레이어의 \textbf{공정한 몫}은 얼마인가'' — 모든 가능한
      \textbf{참여 순서}에서 그 플레이어가 합류함으로써 \textbf{늘어난
      가치를 평균}
    \item \textbf{역사}: 1953년 로이드 섀플리, ``Value of a Cooperative
      Game'' — ``10명의 건설업자가 협력해 건물을 지으면 각각에 얼마를
      공정하게?'' 2012년 섀플리, \textbf{노벨 경제학상}. 60년 뒤,
      2017년 Lundberg \& Lee가 이 도구를 \textbf{신경망·GBDT를 포함한
      거의 모든 모델}의 예측 설명으로 가져옴:
  \end{itemize}
  \begin{block}{정리}
    ``어떤 모델이든, 예측 하나를 특징들의 기여로 분해하려면
    \textbf{섀플리값이 정답}.'' 게임이론의 ``공정한 분배''와 기계학습의
    ``예측 설명''이 \textbf{동일한 수식}으로 풀리는 도구.
  \end{block}
\end{frame}

\begin{frame}{SHAP의 핵심: 가산성(additivity)}
  \begin{itemize}
    \item SHAP은 \textbf{특징들}을 \textbf{플레이어들}로 바꿔 적용:
      특징 $j$의 SHAP값 $\phi_j$ = 특징들이 하나씩 추가되는
      \textbf{모든 가능한 순서}에서 ``특징 $j$가 추가됨으로써 예측값이
      변한 정도''의 \textbf{평균}
  \end{itemize}
  \vskip0.6em
  \kb{핵심 성질(가산성, additivity)}:
  \[ f(x) = \phi_0 + \sum_{j=1}^{n} \phi_j \]
  \begin{itemize}
    \item $\phi_0$ = \textbf{기준값}(baseline, 전체 데이터의 평균 예측)
    \item $\phi_j$ = 특징 $j$가 기준에서 예측값을 얼마나
      \textbf{밀어올렸는지}(양) 또는 \textbf{밀어내렸는지}(음)
    \item 모든 $\phi_j$의 합이 \textbf{정확히} 예측값 $-$ 기준값과 같다는
      것이 SHAP의 \textbf{핵심 보장}
    \item 예: ``신용점수 기여 $-0.3$, 소득 기여 $+0.1$, $\dots$''처럼
      \textbf{합이 정확히 맞는} 분해
  \end{itemize}
\end{frame}

\begin{frame}{2개 특징 예제: 순서 두 가지의 한계기여 평균}
  특징이 2개($A, B$)뿐이면 가능한 추가 순서는 \textbf{두 가지}:
  $A \to B$와 $B \to A$.
  \[ \phi_A = \tfrac{1}{2}\left[ \big(f(\{A\}) - f(\{\})\big)
     + \big(f(\{A,B\}) - f(\{B\})\big) \right] \]
  \begin{itemize}
    \item 특징이 많아지면 순서 수는 $n!$로 \textbf{폭발} $\to$ 실제 SHAP은
      \textbf{샘플링으로 근사}
    \item 장난감 모델($f(\{\})=10, f(\{A\})=16, f(\{B\})=13,
      f(\{A,B\})=20$)에 대입:
    \[ \phi_A = \tfrac{1}{2}\big[(16-10) + (20-13)\big]
         = \tfrac{1}{2}[6+7] = 6.5 \qquad
       \phi_B = \tfrac{1}{2}\big[(13-10) + (20-16)\big]
         = \tfrac{1}{2}[3+4] = 3.5 \]
  \end{itemize}
  \vskip0.6em
  \begin{block}{검산 + 교차항}
    $\phi_0 + \phi_A + \phi_B = 10 + 6.5 + 3.5 = 20 = f(\{A,B\})$ —
    \textbf{정확히} 맞음.
    $\phi_A = 6.5$는 단순히 $f(\{A\})-f(\{\}) = 6$이 \textbf{아님} —
    ``A 혼자 합류(6)''와 ``B가 이미 있는데 A 합류(7)''의 \textbf{평균}.
    1의 차이는 B가 이미 기여한 부분이 A를 ``흡수''한 \textbf{교차항
    (synergy)}을 두 플레이어에 \textbf{공정하게} 나눈 것 — 이 분배가
    섀플리값을 단순 ``순서별 차이''에서 \textbf{공정한 분배 규칙}으로
    만들어주는 핵심.
  \end{block}
\end{frame}

\begin{frame}[fragile]{실습: 코드로 섀플리값 직접 계산}
  특징 2개라 \textbf{정밀 계산}이 가능:
\begin{lstlisting}
f = {frozenset(): 10, frozenset("A"): 16,
     frozenset("B"): 13, frozenset("AB"): 20}

phi_A = 0.5 * ((f[frozenset("A")] - f[frozenset()])
              + (f[frozenset("AB")] - f[frozenset("B")]))
phi_B = 0.5 * ((f[frozenset("B")] - f[frozenset()])
              + (f[frozenset("AB")] - f[frozenset("A")]))

print(f"phi_A = {phi_A},  phi_B = {phi_B}")
assert 10 + phi_A + phi_B == 20
\end{lstlisting}
  \vskip0.3em
  {\footnotesize \texttt{phi\_A = 6.5, \quad phi\_B = 3.5} \\
  \texttt{검산: 10 + 6.5 + 3.5 = 20.0 \quad (f(\{A,B\}) = 20)} }
  \vskip0.5em
  \begin{itemize}
    \item 특징이 \textbf{30개}가 되면 순서 $30! \approx 2.65\times10^{32}$개
      — 우주 나이만큼 초박사들 동원해도 못 셈
    \item 실전 SHAP의 두 \textbf{근사 전략}:
    \begin{enumerate}
      \item \textbf{퍼머테이션 샘플링}: $n!$개 중 작은 수(수백$\sim$수천개)를
        무작위로 골라 평균
      \item \textbf{트리 구조를 이용한 정밀 계산}: GBDT의 분기 경로를
        따라 ``특징 $j$ 합류 시 예측 변화''를 \textbf{모든 경로의 합}으로
        효율 계산 $\to$ RF/GBDT에 한해 \textbf{근사 없이} 정밀
        (\texttt{shap} 패키지 \texttt{TreeExplainer})
    \end{enumerate}
    \item ``경로''가 없는 신경망 등: 퍼머테이션 또는 \textbf{통합
      기대(KernelSHAP)} 같은 근사
  \end{itemize}
\end{frame}

\begin{frame}{실습: 실제 GBDT에 퍼머테이션 근사 SHAP}
  \textbf{알고 만든} 4개 특징 합성 데이터:
  신호 $z = 0.9 f_0 - 0.8 f_1 + 0.5\sin f_2 - 0.3 f_3 +$ 잡음,
  양수면 클래스 1. GBDT(80트리, depth 2) 학습 후, 예측 확률
  0.3$\sim$0.7 사이인 점 하나를 골라 \textbf{400회 퍼머테이션 근사}:
  \vskip0.5em
  {\small
  \begin{tabular}{@{}ll@{}}
    \toprule
     & 값 \\
    \midrule
    test accuracy & 0.758 \\
    고른 점 & index=113, pred\_prob=0.485, true\_label=0 \\
    $f_0$ (강한 +) & $\phi = \mathbf{+0.113}$ \\
    $f_1$ (강한 $-$) & $\phi = \mathbf{-0.380}$ \\
    $f_2$ ($\sin$, 비선형) & $\phi = +0.190$ \\
    $f_3$ (약한 $-$) & $\phi = +0.108$ \\
    baseline $\phi_0$ & 0.454 \\
    prediction & 0.485 \\
    prediction $-$ baseline & $+0.032$ \\
    sum of all $\phi$ & \textbf{+0.032} \quad (가산성 오차: \textbf{0.000000}) \\
    \bottomrule
  \end{tabular}}
  \vskip0.6em
  \begin{itemize}
    \item \textbf{가산성이 정확히 성립}: 0.454 + $(+0.113 - 0.380 +
      0.190 + 0.108)$ = 0.485 — 400회 근사인데도 \textbf{오차 0}
    \item \textbf{부호가 신호 방향과 대체로 일치}: $f_0$ 양 신호 $\to$
      $+0.113$, $f_1$ 음 신호 $\to$ $-0.380$
    \item $f_3$는 이 점에서 $+0.108$으로 ``예상(음)과 반대'' — 약한 신호는
      \textbf{교차 효과에 쉽게 덮임} $\to$ \kq{SHAP 값은 ``이 예측
      하나''에 대한 \textbf{국지적} 설명이지, ``특징의 본질적 방향''이
      아님}
  \end{itemize}
\end{frame}

\begin{frame}{자주 하는 실수: GBDT의 \texttt{n\_estimators}를 RF처럼 ``크게''}
  RF는 $n$ 1$\to$200까지 늘려도 정확도가 \textbf{떨어지는 지점 없음}
  $\to$ ``100$\sim$300으로 여유 있게'' 잡으면 됐지만,
  \textbf{GBDT에는 이 규칙이 적용되지 않음}.
  \vskip0.6em
  \small
  \begin{tabular}{@{}llp{5.6em}p{5.6em}@{}}
    \toprule
     & 랜덤 포레스트 & GBDT \\
    \midrule
    트리의 관계 & \textbf{독립적}(다수결) & \textbf{순차적}(잔차 추격) \\
    트리를 늘리면 & 정확도 $\uparrow$(수확체감), 분산 $\downarrow$
      & 정확도 $\uparrow$ $\to$ \textbf{정점} $\to$ 정확도 $\downarrow$ \\
    \texttt{n\_estimators} & ``많을수록 안정'' — 100$\sim$300 권장
      & \textbf{early stopping으로} 결정 (검증 곡선 최저점) \\
    \texttt{learning\_rate} & 해당 없음 & 0.01$\sim$0.1, 작을수록
      정밀한 early stopping \\
    \bottomrule
  \end{tabular}
  \vskip1em
  \textbf{실무 GBDT 워크플로}: (1) \texttt{learning\_rate=0.01$\sim$0.1}로
  시작 $\to$ (2) \texttt{n\_estimators}를 충분히 크게(예: 1000) 두고
  \texttt{staged\_predict}로 \textbf{매 단계 검증 MSE 추적}(또는
  \texttt{validation\_fraction}) $\to$ (3) \textbf{검증 MSE 최저점에서
  멈춤}. \kq{``더 오래 학습하면 더 좋다''는 직감은 GBDT에서
  \textbf{틀린 직감}.}
\end{frame}

\begin{frame}{자주 묻는 질문}
  \small
  \begin{itemize}
    \item \textbf{GBDT가 왜 RF보다 과적합에 취약한가?} RF는 트리들이
      독립적이라 새 트리가 이전 트리의 실수를 ``따라 하지'' 않음. GBDT는
      \textbf{매 트리가 이전까지의 잔차를 목표}로 학습 — 라벨링 오류가
      섞여 있으면 뒤따르는 트리가 계속 ``그 오류를 맞히려'' 노력해,
      트리를 너무 많이 추가하면 \textbf{노이즈까지 학습}.
      $\to$ \texttt{learning\_rate} 작게 + \texttt{n\_estimators}
      early stopping이 특히 중요.
    \item \textbf{SHAP 값이 크다고 ``좋은'' 기여인가?} \textbf{아님} —
      \textbf{부호}가 방향, \textbf{크기}가 세기만 알려줄 뿐 ``좋다/
      나쁘다''를 말해주지 않음. 예: 대출 심사 ``신용점수 $-0.3$'' =
      ``이 신청자의 경우 승인 확률을 0.3 낮추는 \textbf{방향}''이지,
      신용점수라는 특징이 나쁘다는 뜻이 아님 (다른 특징 소득은 $+0.5$로
      반대 방향 가능, \textbf{합이 정확히 최종 예측}이 됨 = 가산성).
    \item \texttt{learning\_rate}를 0.1 $\to$ 0.01로? 스텝 크기가
      $1/10$ $\to$ \textbf{약 10배 더 많은 단계} 필요. 장점:
      (1) 정점이 \textbf{완만}해져 정확히 멈추는 단계가 극단적으로
      일찍 오지 않음, (2) 한 트리의 과적합이 전체에 미치는 영향 감소.
      표준 조합: \texttt{lr=0.01$\sim$0.05} + \texttt{n=500$\sim$2000}
      (early stopping 결정) — 학습 10배가 ``정밀 early stopping''을 위한
      투자.
    \item \textbf{SHAP값의 합이 예측값과 정확히 같은 게 우연인가?}
      \textbf{아님} — \textbf{가산성} 성질이 보장. 각 순서에서
      ``빈집합 $\to$ 전체 집합''으로 이어지는 한계기여의 합은
      \textbf{항상} $f(x) - \phi_0$와 정확히 같고(경로의 총 변화량은
      \textbf{경로 순서에 무관}), 그 평균도 동일 $\to$ 400회 근사에도
      \textbf{오차 0.000000}.
  \end{itemize}
\end{frame}

\begin{frame}{이 장의 한 줄: 정확도의 대가, 그리고 되찾기}
  \begin{block}{한 줄}
    결정 트리는 ``사람이 읽을 수 있는 if-then 규칙의 나무''이면서 동시에,
    ``얼마나 잘 나누는가''를 \textbf{정확히 수식으로 잴 수 있는}
    알고리즘이다.
  \end{block}
  \vskip0.8em
  \begin{itemize}
    \item 트리를 \textbf{여러 개 합치면}(랜덤 포레스트, GBDT)
      \textbf{정확도는 올라가지만} 그 읽기 쉬움을 \textbf{잃는다}
    \item \kb{SHAP}은 그 대가로 잃은 설명가능성을,
      \textbf{게임이론이라는 전혀 다른 도구}로 \textbf{되찾아오는} 시도
  \end{itemize}
  \vskip0.8em
  \begin{itemize}
    \item \textbf{7.1} — 좋은 질문 = \textbf{지니불순도 감소량(정보이득)
      최대}. U자 곡선: \textbf{test 정점}이 최적 깊이.
    \item \textbf{7.2} — \textbf{독립적 다수결}. 두 무작위성이 $\rho$를
      줄여 \textbf{분산} 감소. 계단 $\to$ 부드러운 곡선.
    \item \textbf{7.3} — \textbf{순차적 잔차 추격} = 함수 공간의
      경사하강. 더 정밀한 대신 \textbf{early stopping}이 필수.
      SHAP(가산성)으로 예측 하나를 \textbf{정확히} 분해.
  \end{itemize}
  \vskip0.8em
  {\footnotesize 7.1절의 ``트리 하나를 잘 나누는 원리''는 이 장의
  \textbf{전부에서 재사용}되었다.}
\end{frame}

\begin{frame}{이번 주차 정리}
  \begin{enumerate}
    \item \kb{7.1} — \textbf{질문을 숫자로}: 지니불순도($1-\sum p_k^2$,
      ``잘못 고를 확률'')와 정보이득으로 ``좋은 질문''을 고름. 결정 경계는
      \textbf{계단}. 최적 깊이 = U자 곡선의 \textbf{test 정점}
      (0.947), \texttt{max\_depth=None}(test 0.912)은 과적합.
    \item \kb{7.2} — \textbf{독립적 다수결}: 배깅 + 특징 무작위화로
      상관계수 $\rho$를 줄여 분산을 감소 ($\sigma/\sqrt{T}$에
      가까워짐). OOB로 \textbf{검증 세트 없이} 성능 평가. 트리를 늘려도
      \textbf{과적합 안 옴}.
    \item \kb{7.3} — \textbf{순차적 잔차 추격} = 함수 공간의 경사하강
      (잔차 = 음의 기울기). RF보다 정확(0.977)하지만 \textbf{early
      stopping 필수}(10단계 최저 $\to$ 500단계에서 2배 악화). SHAP(섀플리값,
      가산성)으로 잃은 설명가능성 \textbf{되찾기}.
  \end{enumerate}
