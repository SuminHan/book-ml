\begin{frame}{이번 주차 개요}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      Chapter 4의 동적계획법은 \textbf{전이확률
      $P(s'|s,a)$ 를 정확히 안다}는 전제에 기대고 있었다.
      \kb{몬테카를로 (MC)}는 그 모델을 통째로 버린다 ---
      \textbf{에피소드를 끝까지 실제로 플레이하고, 결과(실제로
      받은 리턴)를 그대로 가치의 추정치}로 쓴다.
      \vskip0.4em
      \begin{block}{이 챕터를 마치면}
        \begin{itemize}
          \item 리턴의 \kb{평균}이 왜 가치함수로 수렴하는지
            설명하고, 전이확률 단 한 줄도 쓰지 않은 채
            \kb{첫방문 MC 예측}으로 상태가치를 추정한다.
          \item 예측에서 \kb{제어}로 $Q(s,a)$ 를 추정해 그리디
            정책 개선으로 더 나은 정책을 만들고,
            $\varepsilon$-greedy 탐험이 왜 필요한지 설명한다.
          \item off-policy 에서 리턴을 그냥 평균내면
            \kb{편향}이 생기는 이유와 \kb{중요도 샘플링}
            (확률의 비율 $\rho$)으로 보정하는 원리를 말한다.
          \item $\rho$ 의 \kb{분산이 에피소드 길이에 대해
            지수적으로 폭발}하는 이유를 짚고, on/off-policy
            트레이드오프를 정리한다.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{세 차시}
      \begin{enumerate}
        \item \kb{5.1 모델 없이 배운다: MC 예측}
          \,\--- 잔디깎는 기 MDP 에서 DP 의 정확해와 첫방문 MC 가
          \textbf{같은 숫자}에 도달하는 과정, 무편향성 증명,
          ``마지막 방문'' 함정.
        \item \kb{5.2 MC 제어와 블랙잭 실습}
          \,\--- $Q(s,a)$ 추정 + 그리디 개선, $\varepsilon$-greedy
          로 \textbf{블랙잭 기본 전략표}를 재현.
        \item \kb{5.3 중요도 샘플링과 MC 의 한계}
          \,\--- off-policy 편향 보정, 분산 지수 폭발,
          continuing task 한계.
      \end{enumerate}
      \vskip0.5em
      {\footnotesize 한 줄 감각: \textbf{모델 없이 배운다}(이
      장), \textbf{기다림 없이 배운다}(Ch.\,6 TD) --- 강화학습
      지도의 두 축이 하나씩 풀려나간다.}
    \end{column}
  \end{columns}

\section{5.1 모델 없이 배운다: MC 예측}

\begin{frame}{DP 의 전제와 MC 의 질문}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \kb{DP 의 전제}
      \begin{itemize}
        \item 벨만방정식을 \textbf{풀어야} 답에 도달.
        \item 전이확률 $P(s'|s,a)$ 가 \textbf{정확히} 필요.
        \item model-based 강화학습의 시작.
      \end{itemize}
      \vskip0.6em
      카드 게임 비유: 딜러의 ``확률표''를 미리 손에 쥐고
      있는 사람은 없다 --- 그런데도 사람은 \textbf{게임을
      반복해서 해보는 것만으로} 점점 좋은 전략을 배운다.
    \end{column}
    \begin{column}{0.5\textwidth}
      \kb{MC 의 질문}
      \begin{itemize}
        \item 모델을 \textbf{전혀 몰라도} 배울 수 있는가?
        \item 에피소드를 끝까지 플레이하고, \textbf{실제로
          받은 리턴}을 가치의 추정치로 쓴다.
        \item model-free 학습의 시작점.
      \end{itemize}
      \vskip0.6em
      \begin{block}{여전히 남은 제한}
        리턴을 계산하려면 \textbf{에피소드가 끝날 때까지
        기다려야 한다}. 이 기다림을 없애는 것이 Ch.\,6
        \textbf{시간차 학습 (TD)}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{DP vs MC --- ``모델의 유무''라는 축}
  \small
  \begin{tabular}{@{}lll@{}}
    \toprule
     & \kb{동적계획법 (DP)} & \kb{몬테카를로 (MC)} \\
    \midrule
    전제 & $P$ 를 \textbf{안다고} 가정 & $P$ 를 \textbf{모른다고} 가정 \\
    방법 & 벨만방정식을 \textbf{푼다} & 경험을 \textbf{평균낸다} \\
    가치 갱신 & (모델로 즉시) & \textbf{에피소드 종료 시} \\
    접근 & value-based (가치함수 중심) & value-based (가치함수 중심) \\
    \bottomrule
  \end{tabular}
  \vskip1em
  두 방법 모두 \textbf{가치함수}를 중심으로 하는
  \kb{value-based} 접근이며, ``가치를 어떻게 구할 것인가''에서만
  갈라진다. \textbf{모델의 유무}가 강화학습 알고리즘을 나르는
  가장 큰 축이며, 그 위에 ``가치 갱신을 언제 하는가''(MC:
  에피소드 종료 시 / TD: 매 스텝)라는 두 번째 축이 겹쳐진다.
\end{frame}

\begin{frame}{핵심 아이디어: 기댓값을 ``평균''으로 치환한다}
  가치함수의 정의:
  \[
  V^\pi(s) = \mathbb{E}_\pi[\,G_t \mid s_t = s\,]
  \]
  ``이 상태에서 시작해 정책을 따랐을 때 받을 리턴의
  \textbf{기댓값}.'' 기댓값을 정확히 계산하려면 전이확률이
  필요하지만 ---
  \vskip0.6em
  \begin{block}{통계학의 기본 원리 (대수의 법칙)}
    기댓값은 \textbf{많은 샘플의 평균}으로도 근사할 수 있다.
  \end{block}
  \vskip0.6em
  정책 $\pi$ 를 따라 에피소드를 여러 번 플레이하고, 상태 $s$ 를
  방문했을 때 이후 \textbf{실제로 받은 리턴}을 모아 평균 내면
  그것이 곧 $V^\pi(s)$ 의 추정치.
  \vskip0.4em
  \kq{$\mathbb{E}$ (확률분포를 ``알 때''의 연산)를 ``실제로 해본
  다음 평균 내는 것''(분포를 ``모를 때''의 연산)으로 치환해
  버린 것 --- 그것이 MC.}
\end{frame}

\begin{frame}{왜 강화학습에서 ``혁명적''이었는가}
  \begin{itemize}
    \item 우리가 마주한 대부분의 현실(로봇의 마찰계수가 정확히
      몇인지, 환자의 치료 반응이 어떤 분포인지)은
      \textbf{분포를 모를 때}의 문제.
    \item Ch.\,4 에서 벨만방정식을 ``풀기'' 위해 전이확률 $P$ 가
      필요했다면, MC 는 그 방정식을 \textbf{풀지 않는다}.
    \item 벨만 \textbf{동등성} --- 방정식의 양변이 실제로
      플레이할 때 같아야 한다는 사실 --- 을 이용해
      \textbf{플레이한 기록 자체가 답}이 되도록 한다.
  \end{itemize}
  \vskip0.8em
  \begin{block}{정리}
    MC 예측은 ``\textbf{기댓값 = 대수의 법칙에 의한 샘플
    평균}''이라는 가장 기본적인 원리를 강화학습에 그대로
    옮긴 것 --- 전이확률이라는 ``모델''을 전혀 쓰지 않고도
    $V^\pi(s)$ 에 수렴한다.
  \end{block}
\end{frame}

\begin{frame}{역사: 카지노 도시에 이름을 건 알고리즘}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item 이름의 기원: 모나코 공국의 카지노 도시
          \kb{몬테카를로}.
        \item 1940년대, MIT·맨해튼 프로젝트: Ulam,
          von Neumann, Metropolis (Ulam \& Metropolis, 1949).
        \item 풀어야 할 문제: \kb{중성자 이동} --- 물질 안에서
          중성자가 무작위로 여기저기 튕기는 과정.
        \item 이산 확률방정식을 정확히 풀어 계산하는 것은
          불가능에 가까움.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{해법}
        중성자가 어디로 가는지 \textbf{주사위처럼 던져서}
        충분히 많이 추적해보자 --- \textbf{무작위성을 도구}로
        쓰는 계산법, \kb{몬테카를로 방법}.
      \end{block}
      \vskip0.6em
      \begin{block}{RL에서의 MC도 본질은 같음}
        전이확률을 몰라도, \textbf{무작위 전이를 실제로
        겪어본 기록(에피소드)의 평균}이 기댓값을 대신한다.
        카지노에서 주사위를 돌리는 것과 시뮬레이터에서
        에피소드를 돌리는 것은 \textbf{같은 행위}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{DP 로 풀면: 정확값 (정답 먼저 보고두기)}
  Chapter 4 의 정책평가 반복을 이 환경에 적용하면, 터미널
  $V((1,1)) = 0$ 부터 벨만방정식을 반복 대입해 \textbf{모든
  상태의} $V^{\pi_0}$ 를 정확히 구할 수 있다 (전이확률을 우리가
  알므로). 선형방정식 $(I - \gamma P)V = R$ 의 해:
  \vskip0.5em
  \small
  \begin{tabular}{@{}ll@{}}
    \toprule
    상태 & $V^{\pi_0}(s)$ (정확값) \\
    \midrule
    $(0,0), (2,2)$ (모퉁이) & $+0.286$ \\
    $(0,1), (2,1)$ (모퉁이 옆) & $+0.465$ \\
    $(0,2), (2,0)$ (루프의 시작) & $+0.713$ \\
    $(1,2), (1,0)$ (터미널 바로 앞) & $+0.951$ \\
    $(1,1)$ (터미널) & $0$ \\
    \bottomrule
  \end{tabular}
  \vskip0.8em
  터미널이 가까울수록 \textbf{덜 할인된 채} $+1$ 을 받으므로
  $0.286 \to 0.465 \to 0.713 \to 0.951$ 으로 올라간다.
  \vskip0.3em
  \kq{이 $V^{\pi_0}$ 를, 0.8/0.2 전이확률 \textbf{단 한 줄도
  코드로 쓰지 않은 채} 에피소드만으로 추정할 수 있는가?}
\end{frame}

\begin{frame}{첫방문 MC 예측: 알고리즘}
  한 상태를 한 에피소드 안에서 \textbf{여러 번} 방문할 수 있다.
  \kb{첫방문 (first-visit)} MC 는 각 에피소드에서 그 상태를
  \textbf{처음} 방문했을 때의 리턴만 카운트한다 (모든 방문을 다
  쓰는 \kb{모든방문} MC 도 있음 --- 이번 학기는 첫방문만 다룸).
  \vskip0.5em
  \begin{enumerate}
    \item 에피소드 $\pi$ 로 샘플링.
    \item 에피소드를 \textbf{뒤에서부터} 훑으며 리턴을 재귀
      계산: $G_t = R_t + \gamma\, G_{t+1}$.
    \item 각 상태의 \textbf{첫 방문} 시점의 $G_t$ 로 $V(s)$ 를
      갱신(증분 평균).
  \end{enumerate}
  \vskip0.5em
  \begin{block}{왜 뒤에서부터 훑나}
    리턴 $G_t = R_t + \gamma G_{t+1}$ 이 재귀적으로 정의되므로,
    맨 마지막 스텝부터 거꾸로 누적하면 \textbf{한 번의 순회}로
    매 스텝의 $G_t$ 를 전부 계산 --- Ch.\,4 ``터미널부터
    거꾸로 대입'' 패턴과 같은 아이디어.
  \end{block}
\end{frame}

\begin{frame}[fragile]{첫방문 MC 예측: 코드}
\begin{lstlisting}
def mc_prediction(policy, env_sample_episode, n_episodes, gamma):
    # env_sample_episode(policy) -> [(state, action, reward), ...]
    returns_sum = {}
    returns_count = {}
    for _ in range(n_episodes):
        episode = env_sample_episode(policy)
        G = 0.0
        visited = set()
        for t in reversed(range(len(episode))):   # 뒤에서부터: G_t 재귀
            s, a, r = episode[t]
            G = r + gamma * G
            if s not in visited:                  # 첫 방문만 카운트
                visited.add(s)
                returns_sum[s] = returns_sum.get(s, 0.0) + G
                returns_count[s] = returns_count.get(s, 0) + 1
    return {s: returns_sum[s] / returns_count[s] for s in returns_sum}
\end{lstlisting}
  \vskip0.2em
  {\footnotesize \kq{주의:} 위 코드는 뒤에서부터 훑으면서
  ``아직 안 본 상태''를 카운트하는 구조라 사실상
  \textbf{마지막 방문}을 세고 있다 --- 이 구별은 아래
  ``자주 하는 실수''에서 정확히 짚는다.}
\end{frame}

\begin{frame}{손으로: 한 에피소드의 리턴 계산 (미끄러짐 없음)}
  시작 $(0,0)$, 매 스텝 의도한 방향 그대로. $\gamma = 0.9$:
  \small
  \begin{tabular}{@{}cllll@{}}
    \toprule
    스텝 & 위치 & 이동 & 보상 $R$ & 리턴 $G$ \\
    \midrule
    0 & $(0,0)$ & $\to (0,1)$ & $-0.1$ & $-0.1 + 0.9 \times 0.620 = +0.458$ \\
    1 & $(0,1)$ & $\to (0,2)$ & $-0.1$ & $-0.1 + 0.9 \times 0.800 = +0.620$ \\
    2 & $(0,2)$ & $\to (1,2)$ & $-0.1$ & $-0.1 + 0.9 \times 1.000 = +0.800$ \\
    3 & $(1,2)$ & $\to (1,1)$ 터미널 & $+1.0$ & $+1.0$ \\
    \bottomrule
  \end{tabular}
  \vskip0.8em
  리턴은 터미널 스텝부터 거꾸로: $G_3 = +1.0$, $G_2 = -0.1 +
  0.9 \times 1.0 = 0.8$, $G_1 = 0.62$, $G_0 = 0.458$.
  \vskip0.5em
  이 \textbf{에피소드 하나}가 $(0,0)$ 에는 $+0.458$, $(0,1)$ 에는
  $+0.620$, $(0,2)$ 에는 $+0.800$, $(1,2)$ 에는 $+1.000$ 을
  기여 --- \textbf{같은 에피소드 안에서도 상태마다 다른
  리턴}이 각각의 방문 추정치에 들어간다.
  \vskip0.3em
  \kb{한 번의 경험}이 그 에피소드를 \textbf{지나간 모든
  상태}의 가치에 동시에 증거를 추가한다 --- 이것이 MC 의
  구조.
\end{frame}

\begin{frame}{첫방문이 실제로 ``한번''이 아님을 보는 예}
  같은 $(0,0)$ 에서 시작, 이번에는 미끄러짐이 몇 번:
  \small
  \begin{tabular}{@{}clll@{}}
    \toprule
    스텝 & 위치 & 보상 $R$ & 리턴 $G$ \\
    \midrule
    0 & $(0,0)$ & $-0.1$ & $-0.139$ \\
    1 & $(0,1) \to$ (미끄러짐) $(0,0)$ & $-0.1$ & $-0.043$ \\
    2 & $(0,0)$ (두 번째 방문) & $-0.1$ & $+0.063$ \\
    3 & $(0,1)$ & $-0.1$ & $+0.181$ \\
    4 & $(0,0)$ (세 번째, 다시 미끄러짐) & $-0.1$ & $+0.312$ \\
    5$\sim$8 & (정상 루프 $\to$ 터미널) & $\cdots$ & $\to +1.0$ \\
    \bottomrule
  \end{tabular}
  \vskip0.8em
  $(0,0)$ 는 이 에피소드에서 \textbf{세 번} 방문, 매 방문마다
  리턴이 달랐다 ($-0.139,\ +0.063,\ +0.312$). 첫방문 MC 는
  \textbf{$-0.139$ 만} 한 번 더치고 나머지 두 방문은 무시한다 ---
  ``이 에피소드가 $(0,0)$ 에 대해 말하는 것''을
  \textbf{첫 방문 시점의 리턴 하나로 압축}.
  \vskip0.3em
  (마지막 방문을 쓰면 $+0.312$, 세 방문 모두 쓰면
  $(-0.139+0.063+0.312)/3 = +0.079$ --- \textbf{같은 에피소드에서
  세 가지 다른 추정치}가 나온다.)
\end{frame}

  \end{block}

\begin{frame}{왜 처음엔 흔들리는가 --- $\sigma/\sqrt{N}$}
  \begin{itemize}
    \item \textbf{단일 에피소드의 리턴은 잡음이 큰 단일
      관측치}. 평균의 표준오차는
      \[
      \frac{\sigma}{\sqrt{N}}
      \]
      으로 줄어든다 --- $N = 10{,}000$ 이면 오차 범위
      $\approx \pm 0.002$ 로 좁혀진다.
  \end{itemize}
  \vskip0.6em
  \kq{``MC 가 정확하다''는 말은 ``한 에피소드가
  정확하다''가 아니라 \textbf{``충분히 많은 에피소드의
  평균''이 정확하다}는 뜻이며, 그 ``충분히''는
  $\sigma/\sqrt{N}$ 공식이 숫자로 말해준다.}
  \vskip0.6em
  \begin{block}{핵심 정리 (5.1)}
    MC 예측은 전이확률이라는 ``모델''을 전혀 쓰지 않고도
    에피소드를 충분히 많이 돌리면 $V^\pi(s)$ 에 수렴한다.
    대가는 (1) 에피소드가 \textbf{끝날 때까지} 결과를 모른 채
    기다려야 하고, (2) 리턴의 \textbf{분산}이 크므로 같은
    정확도에 더 많은 에피소드가 필요하다 --- (1)은 Ch.\,6 TD,
    (2)는 5.3 중요도 샘플링에서 다시 문제된다.
  \end{block}
\end{frame}

\begin{frame}{왜 첫방문 MC 는 무편향인가 --- 두 줄의 직관}
  MC 예측의 가장 중요한 성질: \kb{무편향성 (unbiased)} ---
  에피소드 수 $N \to \infty$ 일 때 추정치가 \textbf{진짜}
  $V^\pi(s)$ 로 수렴한다.
  \vskip0.5em
  첫방문 MC 의 추정치 $\hat V(s)$ 는 ``상태 $s$ 를 \textbf{처음}
  방문한 에피소드들 중, 그 첫 방문 시점 리턴들의 평균''. 그런데:
  \[
  \mathbb{E}_\pi[\,G_t \mid s_t = s,\ \text{s가 t 이전에 안
  나온}\,] \;=\; V^\pi(s)
  \]
  \begin{enumerate}
    \item \textbf{마르코프 성질} --- ``이전에 $s$ 가 안
      나왔다''는 조건이 붙어도 미래는 \textbf{현재 상태만으로
      결정}되므로 값이 바뀌지 않는다.
    \item \textbf{대수의 법칙} --- 각 에피소드의 ``첫 방문
      리턴''은 같은 분포(기댓값 $V^\pi(s)$)에서 나온
      \textbf{독립 샘플}. 독립 샘플의 평균은 그 분포의 기댓값으로
      수렴.
  \end{enumerate}
  \vskip0.4em
  전이확률 $P$ 는 \textbf{어디에도 등장하지 않는다}.
\end{frame}

\begin{frame}[fragile]{자주 하는 실수: 뒤에서부터 훑으면 ``마지막 방문''}
  이 절의 \texttt{mc\_prediction} 을 처음 만들 때(이 교재의 이전
  버전에도) \textbf{실제로} 그런 버그가 있었다:
\begin{lstlisting}
for t in reversed(range(len(episode))):   # 뒤에서부터
    s, a, r = episode[t]
    G = r + gamma * G
    if s not in visited:   # "아직 못 봤다" = 이 에피소드에서
                           # 가장 늦게(마지막으로) 나온 방문
        ...
\end{lstlisting}
  \texttt{reversed} 순서에서 ``아직 visited 에 없는 상태''를
  만나면, 그 방문은 \textbf{그 상태에서 가장 늦게(마지막으로)
  방문한 것} --- 앞(시간 순서)에서부터 훑어야 첫방문을 센다.
  \vskip0.5em
  왜 \textbf{단순 코딩 실수가 아니라 추정치의 정합성 문제}인가를
  숫자로 본다(같은 20만 에피소드, 시드 42):
  \small
  \begin{tabular}{@{}clrr@{}}
    \toprule
    상태 & 정확값 & 첫방문 MC & \textbf{마지막 방문 MC} \\
    \midrule
    $(0,0)$ & $+0.286$ & $+0.288$ & \textbf{$+0.388$} \\
    $(0,1)$ & $+0.465$ & $+0.466$ & \textbf{$+0.542$} \\
    $(0,2)$ & $+0.713$ & $+0.713$ & \textbf{$+0.756$} \\
    $(1,2)$ & $+0.951$ & $+0.951$ & \textbf{$+1.000$} \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{마지막 방문 MC 가 왜 높게 추정되는가}
  마지막 방문 MC 는 \textbf{모든 상태에서 정확값보다
  높다} --- 특히 터미널 바로 앞 $(1,2)$ 는 정확값 0.951 인데
  $+1.000$ (``미끄러짐과 시간 페널티가 전혀 없는 것처럼''
  동작).
  \vskip0.6em
  이유: 마지막 방문 시점의 리턴은 ``그 \textbf{이후} 같은
  상태를 또 만난다''는 조건이 붙은 리턴 ---
  \[
  \mathbb{E}[\,G_t \mid s_t = s,\ \text{그 이후 } s \text{ 를
  다시 만남}\,]
  \]
  잔디깎는 기에서 $(0,2)$ 를 ``마지막으로'' 방문했다는 것은
  ``그 직후 미끄러져서(또는 다시 순회해서) 같은 상태를 거쳤음''
  을 의미 --- \textbf{단순 순회보다 오래 걸려서 리턴이
  달라진다}.
  \vskip0.5em
  (수식으로: $\mathbb{E}[G \mid \text{last visit}] = 0.756$ 이
  무조건 기대 $V^\pi(0,2) = 0.713$ 과 \textbf{0.043 차이} ---
  위 표의 $+0.756$ 과 정확히 일치.)
  \vskip0.4em
  \begin{block}{결론}
    ``뒤에서부터 훑는'' 코드는 \textbf{편향된(biased)
    추정치}를 만든다 --- 에피소드를 어마어마하게 늘려도 0.756
    근처에 머물고 0.713 에는 \textbf{영원히} 도달하지 못한다.
    \textbf{코딩 실수가 통계적 편향으로 바뀌는 대표적 예.}
  \end{block}
\end{frame}

\begin{frame}{두 번째 실수: ``평균 에피소드 길이''로 방문 횟수
추정하기}
  위 표의 마지막 열 --- $(0,0)$ 은 50,546회, $(0,2)$ 는
  100,155회 방문됨. ``에피소드당 평균 길이가 대략 5.3스텝이고
  20만 에피소드이므로 모든 상태가 대략 같은 횟수($\approx$10만)
  를 방문했을 것''이라고 \textbf{생각하기} 쉽다.
  \vskip0.6em
  그런데 실제로는 \textbf{어떤 상태가 ``몇 에피소드에서 몇
  번'' 방문되느냐}가 상태마다 다르다 --- $(0,2)$ 는 ``위쪽
  루프를 타는 에피소드''마다 2번 이상(시작 + 루프 중간)
  나타나고, $(0,0)$ 은 ``아래쪽 루프 에피소드''에서는 0번일 수
  있다.
  \vskip0.6em
  \begin{block}{원칙: 방문 횟수 = 신뢰도}
    (5.2절 블랙잭에서 보았던 그 원칙) 이 ``실제 횟수''는
    \textbf{계산해봐야} 알 수 있다. ``평균 길이 $\times$ 에피소드
    수''로 \textbf{대충 세는} 것은 방문이 \textbf{불균형한}
    환경(루프가 여러 개인 잔디깎는 기, 낮은 합이 드문 블랙잭)에서
    신뢰도를 \textbf{잘못 판단}하게 만든다.
  \end{block}
\end{frame}

\begin{frame}{자주 묻는 질문 (5.1) --- MC vs DP}
  \begin{itemize}
    \item \textbf{MC 예측은 Ch.\,4 정책평가와 결국 같은 것을
      계산하나요?} --- 목표($V^\pi(s)$)는 같지만 \textbf{방법}이
      다르다. 정책평가는 $P$ 를 안다는 전제로 벨만방정식을
      반복 대입(모델 기반), MC 예측은 $P$ 를 전혀 모른 채
      관찰된 리턴의 \textbf{평균}으로 추정(모델 프리) --- 대수의
      법칙에 의해 에피소드가 많아질수록 진짜 $V^\pi(s)$ 에
      가까워진다.
    \item \textbf{에피소드가 짧으면 추정이 더 정확한가요?} ---
      아니다. 정확도를 높이는 건 그 상태를 \textbf{방문한 횟수}
      (더 많은 에피소드)이지 한 에피소드의 길이가 아니다. 다만
      매우 긴 에피소드는 $G_t$ 분산이 커져 같은 정확도에
      더 많은 에피소드가 필요할 수 있다.
  \end{itemize}
  \vskip0.4em
  {\footnotesize \textbf{MC 가 ``모델을 모른다''는 것이 DP보다
  항상 이득인가요?} --- 아님. 모델이 정확하고 작다면(이
  잔디깎는 기처럼) DP 가 \textbf{훨씬} 빠르고 정확하다 (20만
  에피소드 + 오차 0.002 vs 8개 연립방정식을 \textbf{정확히}
  풀기). MC 의 진짜 무대: ``상태가 수천 수만 개, 전이를 사람이
  손으로 쓸 수 없는'' 문제 --- 이 판단은 Ch.\,15(모델 기반
  RL과 Dyna)에서 다시 만난다.}
\end{frame}

\begin{frame}{자주 묻는 질문 (5.1) --- $\gamma$ 는 MC 에 영향을
주나}
  주겠다 --- 단, \textbf{리턴의 정의}를 통해서만.
  \begin{itemize}
    \item $\gamma = 0.9$ 일 때 $(0,0)$ 의 리턴은 $-0.1 +
      0.9 \times G_{t+1}$ 처럼 미래 보상을 10\%씩 할인해
      계산.
    \item $\gamma = 1$ 이면 할인 없이 모든 보상을 통째로 더 ---
      에피소드가 \textbf{유한하고 끝이 나는} 과업(잔디깎는 기
      처럼)이라면 $\gamma = 1$ 도 괜찮다. (5.2 블랙잭에서
      $\gamma = 1$ 을 쓴 이유와 같은 맥락.)
  \end{itemize}
  \vskip0.6em
  \begin{block}{정리}
    $\gamma$ 는 ``정답이 무엇인가''를 바꾸고, MC 는 그
    ``바뀐 정답''을 평균으로 재추정한다 --- $\gamma$ 를 바꾸면
    \textbf{DP 의 정확값과 MC 의 수렴 목표가 동시에 바뀐다}.
  \end{block}
\end{frame}

\section{5.2 MC 제어와 블랙잭 실습}

\begin{frame}{예측에서 제어: $V$ 대신 $Q$ 를}
  예측(주어진 정책의 가치)에서 제어(더 나은 정책 찾기)로
  넘어가려면, Ch.\,4 의 정책 반복과 같은 틀을 쓴다 --- 다만
  이제는 $V(s)$ 대신 \textbf{$Q(s,a)$ 를 추정}(전이확률을
  모르니 $V$ 만으로는 ``어느 행동이 최선인지'' 판단할 수
  없음).
  \vskip0.6em
  \begin{block}{탐험을 어떻게 보장할 것인가}
    순수 탐욕적으로만 행동하면 \textbf{한 번도 시도하지 않은}
    (상태, 행동) 쌍은 영원히 가치를 알 수 없다 (Ch.\,2.1
    ``탐욕적 선택의 함정''이 재발).
  \end{block}
\end{frame}

\begin{frame}{두 가지 탐험 전략: exploring starts vs
$\varepsilon$-greedy}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \kb{(1) 탐험적 시작 (exploring starts)}
      \begin{itemize}
        \item 매 에피소드를 \textbf{무작위로 고른} (상태,
          행동) 쌍에서 \textbf{강제로 시작}.
        \item 모든 (상태, 행동) 쌍이 결국 충분히 방문됨.
        \item 이론적으로 깔끔하지만, 임의의 상태·행동에서
          ``시작''할 수 있어야 하므로 \textbf{실전 환경에서는
          항상 가능하지 않다}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \kb{(2) $\varepsilon$-greedy} (Ch.\,2.1 의 그 전략)
      \begin{itemize}
        \item 매 스텝 \textbf{작은 확률로 무작위 행동}을 섞음.
        \item 시작 상태와 무관하게 모든 (상태, 행동) 쌍을
          계속 조금씩 방문.
        \item \textbf{이번 실습은 $\varepsilon$-greedy.}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{MC 제어: 코드 (첫방문 기준 $Q$ 갱신)}
\begin{lstlisting}
def mc_control(env_step, n_states, n_actions, n_episodes,
               epsilon, gamma, start_state):
    Q = [[0.0] * n_actions for _ in range(n_states)]
    counts = [[0] * n_actions for _ in range(n_states)]

    def generate_episode():
        s, episode = start_state, []
        for _ in range(100):
            a = random.randrange(n_actions) if random.random() < epsilon \
                else max(range(n_actions), key=lambda x: Q[s][x])
            ns, r, done = env_step(s, a)
            episode.append((s, a, r))
            s = ns
            if done:
                break
        return episode

    for _ in range(n_episodes):
        episode = generate_episode()
        G, visited = 0.0, set()
        for t in reversed(range(len(episode))):
            s, a, r = episode[t]
            G = r + gamma * G
            if (s, a) not in visited:          # (상태, 행동) 쌍 방문
                visited.add((s, a))
                counts[s][a] += 1
                Q[s][a] += (G - Q[s][a]) / counts[s][a]  # 증분 평균
    return Q
\end{lstlisting}
  {\footnotesize MC 예측과 \textbf{거의 같다} --- $V[s]$ 가
  아니라 $Q[s][a]$ 를 갱신하고, ``방문'' 기준이 상태 $s$ 가
  아니라 \textbf{(상태, 행동) 쌍} $(s,a)$ 임. 나머진 동일(뒤에서
  훑어 $G_t$ 재귀 계산, 증분 평균).}
\end{frame}

\begin{frame}{MC 제어로 학습하는 것의 정체: $Q^\pi$}
  위 코드가 실제로 학습하는 것은 ``최적 정책의 가치''가
  아니라, \textbf{지금 $\varepsilon$-greedy 로 행동하는 그
  정책 자체의 가치 $Q^\pi(s,a)$}.
  \vskip0.6em
  \begin{itemize}
    \item 탐험을 위해 섞어 넣은 ``무작위 행동''이 받은 (보통은
      \textbf{더 낮은}) 보상이 $Q^\pi$ 에 그대로 반영됨.
    \item $Q^\pi$ 는 진짜 최선의 행동 가치 $Q^{*}(s,a)$ 보다
      \textbf{계속해서 약간 낮게} 측정됨 --- $\varepsilon$ 이
      크면 클수록 낮춤이 심해진다.
    \item \textbf{탐험(정보 수집)은 가치를 얻는 대가로 보상을
      희생}하는 것.
  \end{itemize}
  \vskip0.5em
  그럼에도 최종적으로 $Q$ 에서 \textbf{탐욕적 정책}
  $\pi(s) = \arg\max_a Q(s,a)$ 를 뽑아 써서(실전에서 무작위
  행동을 섞지 않고) 최적 정책에 가까운 행동을 한다.
\end{frame}

\begin{frame}{on-policy 라는 뜻: 행동 정책 = 목표 정책}
  \begin{block}{핵심 정리 (5.2)}
    $\varepsilon$-greedy MC 제어는 (1) 탐험을 섞은 \textbf{행동
    정책 $\pi$} 의 $Q^\pi$ 를 추정하고, (2) 거기서 $\arg\max$
    를 취해 \textbf{결정론적 정책}을 만들어낸다. ``추정
    대상''과 ``최종 배포 정책''이 살짝 달라지는 이 미묘함은
    Ch.\,9 DQN 과 Ch.\,11 PPO 에서도 계속 따라다닌다.
  \end{block}
  \vskip0.5em
  ``학습하는 정책($\pi$)''과 ``그 정책의 가치를 측정하는
  기준''이 \textbf{같은 정책}이라는 점이, 이번 절의 MC 제어가
  \kb{on-policy} 방법이라는 뜻. 5.3절에서
  ``\textbf{무작위로 모은 데이터로 다른 정책의 가치를} 알고
  싶다''는 \kb{off-policy} 문제로 넘어갈 때, 바로 이
  ``행동 정책과 목표 정책이 같아야 한다는'' 전제가 어떤
  불편함을 만드는지 살펴본다.
\end{frame}

\begin{frame}{실습: 블랙잭에서 MC 제어로 학습하기}
  Sutton \& Barto 의 RL 교재에서 MC 제어를 소개할 때 쓰는
  \textbf{표준 예제}가 블랙잭이다.
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \kb{상태} $s = (\text{플레이어 합},\ \text{딜러의
          보이는 카드})$
        \item \kb{행동}: 0 = \kb{스틱}(멈추기), 1 = \kb{히트}(카드
          더 받기)
        \item \kb{보상}: $+1$(이김), $-1$(지면/버스트), $0$(무)
        \item 스틱하면 \textbf{딜러가 17 이상}이 될 때까지
          받는 과정을 한꺼번에 시뮬레이션.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{왜 블랙잭인가}
        MC 제어의 두 가설(모델 불필요 + 에피소드가 있어야
        함)을 \textbf{동시에} 가장 선명하게 보여준다.
        \begin{itemize}
          \item 카드 덱의 전이확률은 손으로 계산하기 번거롭지만
            게임을 ``돌려보면'' 알 수 있다.
          \item 한 판은 \textbf{분명히 끝난다} (버스트 또는
            딜러 마무리) --- MC 의 ``에피소드 끝까지 기다려
            리턴 계산''이 그대로 적용.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{블랙잭: 에피소드 생성 코드}
\begin{lstlisting}
def gen_episode(Q, epsilon):
    player_sum = draw_card() + draw_card()
    dealer_showing = draw_card()
    s = (player_sum, dealer_showing)
    episode = []
    while True:
        if s[0] < 12:
            a = 1  # 12 미만이면 무조건 히트 (자명한 선택)
        else:
            a = random.randrange(2) if (s not in Q or random.random() < epsilon) \
                else max(range(2), key=lambda x: Q[s][x])  # 0=스틱, 1=히트
        ns, r, done = step(s, a)
        episode.append((s, a, r))
        if done:
            break
        s = ns
    return episode
\end{lstlisting}
  {\footnotesize (아래는 학습 목적의 단순화 버전 --- 실제
  카지노의 ``소프트 에이스'' 처리는 생략.)}
\end{frame}

\begin{frame}{할인율 $\gamma = 1$ 을 쓰는 이유}
  \begin{itemize}
    \item Ch.\,4 에서는 $\gamma < 1$ 을 써서 ``무한히 길어질 수
      있는 미래 보상''을 수렴되게 했었다.
    \item 블랙잭의 한 판은 \textbf{결정적으로 끝나는 유한
      호라이즌} --- 누군가 버스트하거나 딜러가 17 이상으로
      마무리되는 순간 에피소드가 끝나므로, 보상이 ``무한히 먼
      미래''까지 이어지지 않는다.
    \item 그런 \textbf{유한하고 끝이 나는 과업}에서는
      $\gamma = 1$ (보상을 통째로 가중하지 않는 것)이
      자연스러운 선택.
  \end{itemize}
  \vskip0.5em
  \begin{block}{보상을 $-1, 0, +1$ 로 두는 이유}
    ``기대 수익이 0 근처의 게임''이라는 직관이 $Q$ 값의
    \textbf{부호와 크기}에 그대로 담기게 한다. $Q(s,a) > 0$
    은 기대적으로 \textbf{이길} 확률이 더 크다, $Q(s,a) < 0$
    은 \textbf{지실} 확률이 더 크다.
  \end{block}
\end{frame}

\begin{frame}{결과: 카드 덱 확률분포를 계산하지 않은 전략}
  플레이어 합이 12 이상인 상태에서만 $Q$ 를 학습, 30만
  에피소드(시드 0):

  \begin{tabular}{@{}llll@{}}
    \toprule

    \midrule
    (20, 3)  & $+0.633$ & $-0.839$ & 스틱 \\
    (20, 10) & $+0.520$ & $-0.811$ & 스틱 \\
    (12, 10) & $-0.539$ & $-0.253$ & 히트 \\

    \item 플레이어 합이 20이면 딜러 카드가 무엇이든
      (3이든 10이든) \textbf{스틱}의 $Q$ 가 압도적으로

    \item 합이 12뿐이고 딜러가 강한 카드(10)를 보여주면

  \end{itemize}
  \vskip0.3em
  카드 덱의 확률분포나 딜러의 정확한 행동 규칙을 \textbf{단
  한 줄도 명시적으로 계산하지 않았는데도}, 30만 번의

\begin{frame}{$\varepsilon$ 를 줄여가며 학습: 탐험에서 착취로}
  $\varepsilon = 0.1$ 을 고집해서 40만 에피소드까지 돌리면,
  학습은 ``대략 좋은 정책''에서 멈추기 쉽다 --- 매 스텝 10\%
  의 무작위 행동을 계속 섞어주니 $Q$ 추정치가 항상 조금씩
  흔들리고 \textbf{경계 상태의 판단이 굳어지지 않기} 때문이다.
  \vskip0.5em
  실전에서는 보통 \textbf{학습이 진행될수록 $\varepsilon$ 를
  줄인다} (\kb{$\varepsilon$-decay}, 탐험 스케줄) --- 초반에
  $\varepsilon$ 을 크게 해서 전략 공간을 넓게 훑고, 후반에는
  줄여 ``지금까지의 학습''을 믿고 행동.
  \vskip0.5em
  이 실습: 30만 에피소드($\varepsilon = 0.1$) 뒤에 10만
  에피소드를 $\varepsilon = 0.02$ 로 추가 학습. 효과(작지만
  실제로 측정됨):
  \begin{itemize}
    \item 정확 정책과의 일치율: \textbf{104/110 $\to$ 105/110}.
    \item $|Q - Q^{*}|$ 의 평균: 0.029 $\to$ 0.027.
  \end{itemize}
\end{frame}

\begin{frame}{$\varepsilon$ 는 ``상한선''인가, ``속도''인가}
  ``110개 중 1개''라는 소박한 변화지만, 이 숫자가
  \kq{$\varepsilon$ 이 ``학습의 \textbf{상한선}''을 제약하는가,
  아니면 단순히 ``\textbf{수렴 속도와 노이즈 크기}''만 조정하는
  가?} 를 생각하게 해준다.
  \vskip0.5em
  \textbf{정답은 전자(대부분)}. $\varepsilon > 0$ 인 한 행동
  정책은 무작위 행동을 섞으므로, 그 정책의 가치 $Q^\pi$ 는
  원래 $Q^{*}$ 보다 낮다. $\varepsilon$ 을 0에 가깝게 하면
  $\pi \to$ 그리디 정책에 가까워져 $Q^\pi \to Q^{*}$ 가
  된다.
  \vskip0.5em
  \begin{block}{$\varepsilon$ 를 0으로 \textbf{완전히} 줄이면
  어떨까}
    그때는 더 이상 새 (상태, 행동) 쌍을 방문하지 못해, 아직
    제대로 추정되지 않은 상태의 $Q$ 가 \textbf{영원히} 그
    자리에서 멈춘다 --- 탐험의 대가(낮은 즉석 보상)를 치르지
    않으면 새로운 지식을 얻을 기회도 사라진다.
    \textbf{``탐험 vs 착취'' 트레이드오프의 가장 정직한 모습.}
  \end{block}
\end{frame}

\begin{frame}{자주 하는 실수: 방문 횟수가 적은 상태의 $Q$ 를
그대로 믿기}
  플레이어 합이 낮고(12$\sim$13) 애매한 상태들은 20 근처
  상태보다 \textbf{방문 빈도가 낮아}, $Q$ 값 추정의 노이즈가
  상대적으로 더 많다 --- 30만 에피소드로도 교과서적 기본
  전략표와 정확히 일치하지 않는 경계 상태가 남아 있을 수
  있다.
  \vskip0.5em
  \begin{itemize}
    \item MC 추정치는 ``그 상태를 \textbf{몇 번 방문했는가}''에
      직접적으로 의존.
    \item 방문 횟수가 적은 (상태, 행동) 쌍의 $Q$ 값은 곧이곧대로
      신뢰하기보다 \textbf{``더 많은 에피소드가 필요하다''는
      신호}로 받아들이는 것이 안전.
    \item 이 문제는 Ch.\,6 TD, Ch.\,9 DQN 에서도 형태를 바꿔
      계속 등장.
  \end{itemize}
  \vskip0.4em
  위 비교표의 마지막 열이 이 ``방문 횟수''를 그대로 보여준다:
  스틱은 ``12 이상이면 거의 무조건 멈추는'' 학습된 행동이라
  방문이 드물고(82$\sim$3100회), 히트는 ``아직 카드 더 받을
  수 있는'' 상태의 주된 경로라 방문이 잦다(168$\sim$4210회).
  그런데 정확히 \textbf{방문이 드문 쪽이 편차가 크다}.
\end{frame}

\begin{frame}{두 번째 실수: 학습된 $Q^\pi$ 를 곧 $Q^{*}$ 처럼
읽기}
  $\varepsilon$-greedy 로 학습한 $Q$ 는 탐험의 희생을 반영한
  \textbf{그 행동 정책의 가치}이므로, 같은 (상태, 행동)의
  $Q^{*}$ 보다 \textbf{항상 조금 낮게} 측정된다 (예: (20, 3)
  스틱의 0.633 이 정확 값 0.656 보다 낮은 것).
  \vskip0.6em
  그래서 ``학습된 $Q$ 가 $-0.5$ 인데, 이건 그 상태가 그만큼
  나쁜 것인가?''라고 해석하면 \textbf{안 되고},
  \[
  \text{``그 $\varepsilon$-greedy 정책 아래에서 그 상태·행동의}
  \]
  \[
  \textbf{평균 리턴이 약 $-0.5$ 다''라고 읽어야 한다.}
  \]
  \vskip0.5em
  \begin{block}{최적 정책의 ``진짜'' 가치를 알고 싶다면}
    5.3절의 \kb{off-policy (중요도 샘플링)}이나 Ch.\,6 의
    \kb{Q-learning} 이 필요하다.
  \end{block}
\end{frame}

\begin{frame}{자주 묻는 질문 (5.2) --- MC 예측 vs MC 제어}
  \textbf{코드적으로 얼마나 다른가요?} --- \textbf{거의
  같다, 두 줄만 바뀐다}.
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \kb{MC 예측 (5.1)}
      \begin{itemize}
        \item $V[s]$ 를 갱신.
        \item 상태 $s$ 만 ``방문''으로 셈.
      \end{itemize}
      \vskip0.5em
      \kb{MC 제어 (5.2)}
      \begin{itemize}
        \item $Q[s][a]$ 를 갱신.
        \item \textbf{(상태, 행동) 쌍} $(s,a)$ 를 방문
          기준으로 삼음.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      나머지는 \textbf{동일} (에피소드를 뒤에서부터 훑어
      $G_t$ 를 재귀 계산, 증분 평균으로 갱신).
      \vskip0.6em
      이 ``거의 같다''가 MC 의 미덕 --- \textbf{예측을 할 줄
      안다면 제어는 자동으로 따라온다.}
      \vskip0.6em
      \textbf{왜 $V$ 만으로는 안 되고 $Q$ 여야 하는가?} ---
      $V(s)$ 만으로는 ``다음에 \textbf{어느 행동}을
      골랐는지''가 사라져 다음 행동을 결정할 수 없다.
      $Q(s,a)$ 는 ``지금 행동까지 포함''한 가치로 그
      부족함을 메운다.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{자주 묻는 질문 (5.2) --- 실제 블랙잭에 쓸 수
있나, Q 부호 해석}
  \begin{itemize}
    \item \textbf{학습한 정책을 실제 블랙잭에 쓰면 이길 수
      있나요?} --- 이 절의 단순화 모델(소프트 에이스 무시,
      무한 카드 덱)로 학습한 정책은 \textbf{기본 전략의
      골격}은 재현하지만, 실제 카지노 규칙(소프트 17,
      멀티덱, 히트 온 17 등)에는 정확히 맞지 않는다. MC 의
      강점은 ``규칙이 아무리 복잡해도(전이확률이 계산
      불가능해도) 시뮬레이션만 하면 전략을 배운다''는 것이지,
      ``학습한 정책이 무조건 현역 프로보다 낫다''는 것이
      아니다.
    \item \textbf{$Q$ 값의 부호(+/−)는 어떻게 해석하나?} ---
      블랙잭은 (공정한 게임이므로) 전체 기대 수익이 0 근처
      이므로, 모든 상태·행동의 $Q$ 가 동시에 양수/음수일 수
      없고 \textbf{상태마다 양/음이 섞여 있어야 한다}. 학습된
      $Q$ 에서 (20, 3) 스틱 $+0.63$, (12, 10) 스틱 $-0.54$ 로
      부호가 반대인 것이 바로 이 ``게임 전체는 0 근처''
      구조를 반영한다.
  \end{itemize}
\end{frame}

\section{5.3 중요도 샘플링과 MC 의 한계}

\begin{frame}{off-policy 라는 문제: 그냥 평균내면 편향}
  지금까지의 MC 는 ``행동에 쓰는 정책''과 ``가치를 평가하려는
  정책''이 같았다 (\kb{on-policy}) ---
  $\varepsilon$-greedy 로 행동하면서 그 $\varepsilon$-greedy
  정책 \textbf{자신}의 가치를 학습했다.
  \vskip0.5em
  그런데 ``\textbf{무작위로 행동해서 모은 데이터}로
  \textbf{탐욕적 정책의 가치}를 알고 싶다''면(\kb{off-policy}),
  그냥 평균을 내면 \textbf{안 된다} ---
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \kb{행동 정책} $b$ (behavior)

      데이터를 \textbf{모을 때} 쓴 정책.
    \end{column}
    \begin{column}{0.5\textwidth}
      \kb{목표 정책} $\pi$ (target)

      가치를 \textbf{평가하려는} 정책.
    \end{column}
  \end{columns}
  \vskip0.4em
  $b \neq \pi$ 면 관찰된 리턴을 \textbf{그대로} 평균내는 것은
  \textbf{편향된(biased) 추정}이다.
\end{frame}

\begin{frame}{중요도 샘플링: 확률의 비율 $\rho$ 로 보정}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \kb{중요도 샘플링} 은 각 에피소드의 리턴에 ``이 궤적이
      $\pi$ 에서 나왔을 확률 대비 $b$ 에서 나왔을 확률의
      \textbf{비율}''을 가중치로 곱해 편향을 보정:
      \[
      \rho \;=\; \prod_t \frac{\pi(a_t | s_t)}{b(a_t | s_t)}
      \]
      \kb{왜 ``확률의 비율''인가} --- 평가하려는 값은
      $V^\pi(s_0) = \mathbb{E}_\pi[G_0]$ 인데, $b$ 에서
      샘플을 모으려면 $\pi$ 에 대한 기댓값을 $b$ 에 대한
      기댓값으로 바꿔야 한다 --- 확률론의 기본 정리를
      쓰는 것뿐:
      \[
      \mathbb{E}_\pi[f] =
      \mathbb{E}_b\!\left[f \cdot \frac{p_\pi}{p_b}\right]
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{직관}
        목표 정책 $\pi$ 라면 거의 선택하지 않았을 행동을
        행동 정책 $b$ 가 우연히 선택해서 얻은 궤적은, 그
        \textbf{희소함}($\rho$ 가 작음)만큼 가중치를
        \textbf{낮춰서} 반영.
      \end{block}
      \vskip0.4em
      \begin{itemize}
        \item 가중치가 \textbf{에피소드 전체의 비율}이고 단
          한 스텝의 비율이 아닌가 --- 마르코프 성질 덕분에
          스텝별 비율의 \textbf{곱}으로 분해.
        \item $\rho$ 는 0 또는 1이 아니라 \textbf{4나 1/8처럼
          ``10배''}가 될 수 있다.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.4em
  \kq{``비율 2가 매 스텝마다 곱해져 $L$ 스텝 뒤에는
  $2^L$ 이 된다''는 사실이 곧 \textbf{분산 폭증}의 근원.}
  \vskip0.3em
  {\footnotesize 이 아이디어는 Ch.\,9(DQN)의
  \kb{경험재현 버퍼}(과거 정책들이 모은 데이터를 재사용)와,
  Ch.\,11(PPO)의 \kb{확률비}에서 형태를 바꿔 다시
  등장한다. Ch.\,6 SARSA(Sutton, 1991)의 \kb{off-policy
  TD(2)} --- 각 스텝의 TD 오류를 새 행동 정책이 실제로 고를
  확률 $b(a'_{t+1}|s_{t+1})$ 로 가중 --- 도 같은 분산
  문제를 한 스텝 단위로 다스린 사례.}
\end{frame}

\begin{frame}{손으로 하는 중요도 샘플링: 5칸 회랑}
  5칸 GridWorld(칸 0$\sim$4, 0과 4가 종료, 왼쪽 끝 $-1$,
  오른쪽 끝 $+1$, $\gamma = 0.9$, 시작 칸 2)에서 두 정책:
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \kb{행동 정책 $b$}
      매 스텝 왼쪽/오른쪽을 \textbf{50-50} 무작위.
    \end{column}
    \begin{column}{0.58\textwidth}
      \kb{목표 정책 $\pi$}
      \textbf{항상 오른쪽}(탐욕적).
      \vskip0.3em
      두 정책의 가치를 \textbf{정확히} 알 수 있다(Ch.\,4 의
      역순 대입):
      \[
      V^\pi = [0.729,\ 0.810,\ 0.900,\ 1.000,\ 0]
      \]
      \[
      Q^\pi(2,\ \text{right}) = 0.900 > Q^\pi(2,\
      \text{left}) = 0.729
      \]
    \end{column}
  \end{columns}
  \vskip0.5em
  즉 ``칸 2에서 오른쪽이 더 좋은 정책''이 $V^\pi(2) = 0.9$
  다.
  \vskip0.3em
  \kq{핵심 질문: \textbf{$b$ 의 데이터만} 모아서
  $V^\pi(2) = 0.9$ 를 추정할 수 있는가?}
\end{frame}

\begin{frame}{편향의 함정: ``그냥 평균''이 추정하는 것}
  $b$ 에서 시작 칸 2의 리턴을 모아서 평균내면
  $\mathbb{E}_b[G_0 \mid s_0 = 2]$ 가 나온다 --- 이건
  \textbf{$V^b(2)$ 의 추정}이지 $V^\pi(2)$ 의 추정이
  \textbf{아니다}.
  \vskip0.5em
  이 예에서는 우연히 $V^b(2) = 0$ (시작이 정확히 중앙이라
  ``왼쪽 끝 $-1$''과 ``오른쪽 끝 $+1$''의 가능성이 대칭으로
  상쇄) --- 그런데 $\varepsilon$-greedy 같은 \textbf{비대칭}
  행동 정책이라면 이 평균은 $V^\pi$ 에서 \textbf{원래
  위치에서 일정한 거리만큼 벗어나는} 값.
  \vskip0.5em
  \begin{block}{편향 (bias)}
    에피소드를 \textbf{아무리 많이 돌려도 해소되지
    않는다}. (실습 노트북에서 2만 에피소드로 직접 확인:
    그냥 평균은 $0.004 \to 0$ 에 수렴, 0.9에는 \textbf{절대
    안 간다}.)
  \end{block}
\end{frame}

\begin{frame}{중요도 보정: 스텝별 비율}
  $b$ 가 50-50이고 $\pi$ 가 항상 오른쪽이므로 \textbf{스텝별
  비율}은:
  \[
  \frac{\pi(a_t|s_t)}{b(a_t|s_t)} \;=\;
  \begin{cases} 2 & a_t = \text{right} \\ 0 & a_t =
  \text{left} \end{cases}
  \]
  다시 말해, \textbf{왼쪽이 한 번이라도 나오면 그 에피소드의
  $\rho = 0$} --- ``$\pi$ 라면 절대 이렇게 행동하지
  않았을 궤적''이라 평가에서 \textbf{완전히 제외}. 오른쪽만
  나오면(시작 2에서 오른쪽만 $L$번 $\to$ 종료 4) $\rho = 2^L$
  다.
  \vskip0.5em
  \small
  \begin{tabular}{@{}llll@{}}
    \toprule
    에피소드 (시작 2) & 리턴 $G_0$ & $\rho$ & $\rho \cdot G_0$ \\
    \midrule
    (2,오) $\to$ (3,오, 종료) & $0 + 0.9 \times 1 = 0.9$ & $2 \times 2 = 4$ & $3.6$ \\
    (2,좌) $\to$ (1,오) $\to$ (2,오) $\to$ (3,오, 종료) & $0.9^3 = 0.729$ & $0$ (첫 스텝에 왼쪽) & $0$ \\
    (2,좌) $\to$ (1,$\cdots$) $\to$ (0, 종료) & $\approx -1$ & $0$ & $0$ \\
    \bottomrule
  \end{tabular}
  \vskip0.8em
  목표 정책이 \textbf{결정론적}이라 시작 2에서 $\rho \neq 0$
  인 궤적은 \textbf{``오른쪽, 오른쪽'' 단 하나뿐} ---
  $\rho G_0$ 가 오직 두 값 $\{3.6, 0\}$ 만 취한다.
\end{frame}

\begin{frame}{그래도 정확히 맞는다 (편향 없음) --- 대가는
분산}
  \[
  \mathbb{E}[\,\rho G_0\,] = 0.25 \times 3.6 = 0.9 =
  V^\pi(2)
  \]
  \textbf{정확히} 맞는다 (편향 없음).
  \vskip0.5em
  다만:
  \[
  \text{Var}[\,\rho G_0\,] = 0.25 \times 3.6^2 - 0.9^2 =
  2.43
  \]
  표준편차 $\approx 1.56$ --- \textbf{평균(0.9)의 1.7배}.
  ``희귀하지만 거대한 값''이 평균을 지배하는 구조가 여기까지
  축소되어도 보인다.
  \vskip0.5em
  일반적으로 에피소드 길이 $L$ 일 때 $\pi$ 가 그 궤적을
  실제로 뽑을 확률 대비 $b$ 가 뽑을 확률은 $2^L$ 배이므로:
  \[
  \mathbb{E}_b[\rho] = 1, \qquad \text{Var}_b[\rho] = 2^L - 1
  \]
  \vskip0.3em
  실전에서는 에피소드가 더 길고($\rho = 2^L$ 지수적으로
  커짐), 목표 정책이 확률적이라($\rho$ 가 0/4 외에도 무한한
  값) 이 문제는 \textbf{훨씬 극단적}.
\end{frame}

\begin{frame}{$\text{Var}[\rho] = 2^L - 1$: 지수 폭발}
  \kb{$\text{Var}[\rho] = 2^L - 1$ 이 \textbf{지수함수적으로}
  커진다는 것이 핵심.}
  \small
  \begin{tabular}{@{}cl@{}}
    \toprule
    $L$ & $\text{Var}[\rho] = 2^L - 1$ \\
    \midrule
    $L = 2$  & $3$ \\
    $L = 4$  & $15$ \\
    $L = 6$  & $63$ \\
    $L = 8$  & $255$ \\
    $L = 10$ & $1023$ \\
    \bottomrule
  \end{tabular}
  \vskip0.6em
  \textbf{에피소드가 2배 길어질 때마다 분산이 4배로 뛴다}.
  \vskip0.4em
  2만 에피소드 시뮬레이션(시드 고정)에서 실제로 확인:
  \begin{itemize}
    \item 그냥 평균: $0.004$ ($V^b = 0$ 으로 수렴, \textbf{편향}).
    \item $\rho$ 가중 평균: $0.892$ (진짜값 $0.9$ 로 수렴,
      \textbf{편향 없음}), $\rho \cdot G_0$ 의 표준편차
      $\approx 1.55$ --- 평균 0.9 의 거의 2배.
  \end{itemize}
  \vskip0.3em
  \begin{block}{본질}
    \textbf{편향은 없애주되 대가로 분산을 극단적으로
    키운다} --- 이것이 중요도 샘플링의 본질.
  \end{block}
\end{frame}

\begin{frame}{중요도 샘플링의 두 가지 실패 모드}
  \begin{center}
    \includegraphics[width=0.92\linewidth]{ch05_3_importance_sampling.png}
    \vskip0.3em
    {\footnotesize 좌: 그냥 평균($b$ 데이터)은 편향되어
    $V^b = 0$ 에 수렴. 중앙: $\rho$ 가중 평균은 편향 없이
    $V^\pi = 0.9$ 에 수렴하나 점의 분산이 매우 큼. 우:
    $\rho$ 분포가 에피소드 길이 $L$ 에 따라 지수적으로
    퍼짐(Var[$\rho$] = $2^L - 1$).}
  \end{center}
\end{frame}

\begin{frame}{``정확한 수렴'' $\neq$ 실용적 --- 에피소드 수가
지수적으로 폭증}
  수학적으로는 $\rho$ 가중 MC 도 에피소드 수 $\to \infty$ 에서
  $V^\pi$ 로 수렴한다(편향이 없으므로). 그런데
  ``\textbf{수렴한다}''와 ``\textbf{실용적}''은 다른 이야기 ---
  표준오차는 $\sigma/\sqrt{N}$ 인데, 매 에피소드별 표준편차
  $\sigma$ 자체가 \textbf{에피소드 길이에 대해 지수적으로}
  커진다 ($\sigma^2 \propto \mathbb{E}[\rho^2] \sim 2^L$).
  \vskip0.5em
  길이 $L$ 인 에피소드에서 $\pm 0.05$ 정확도를 원하면 필요한
  에피소드 수는 대략
  \[
  N \approx \left(\frac{2}{\delta}\right)^2 \cdot 2^L \cdot
  \mathbb{E}[G^2]
  \]
  $L$ 이 1 커질 때마다 \textbf{2배}. $L = 5$ 이면 $10^4$ 급,
  $L = 20$ 이면 $10^9$ 급 --- \textbf{15 스텝이 길어진 대가로
  필요한 데이터가 약 $2^{15} \approx 3$만 배}.
  \vskip0.4em
  \begin{block}{off-policy 학습 전반의 근본 난제}
    에피소드가 조금만 길어져도 \textbf{``수학적으로는
    가능하지만 실제로는 불가능''}이 된다.
  \end{block}
  \vskip0.3em
  {\footnotesize Ch.\,11(PPO)의 \kb{클리핑된 확률비}가 바로
  이 문제를 ``$\rho$ 가 너무 크면 \textbf{자른다}''는 가장
  단순한 다스림법 ---
  $\text{clip}(\rho_t, 1-\varepsilon, 1+\varepsilon)$ 는
  ``$\rho$ 가 100배가 되는 궤적을 100배로 반영''하지 않고
  ``최대 1.1배까지만'' 반영. 편향을 들여들이는 대가(클리핑된
  후에는 더 이상 정확한 IS가 아님)를 치르되 분산을 극단적으로
  줄이는 \textbf{편향-분산 트레이드오프의 의도적 선택}.}
\end{frame}

\begin{frame}{``분산이 크다''가 실전에서 무엇을 의미하나}
  수학식으로는 ``분산 $\sigma^2$'' 한 줄이지만, 실전에서는
  이렇게 느껴진다:
  \begin{itemize}
    \item \kb{학습 곡선이 ``안 정해진다''}: 에피소드 1만 개
     까지 평균이 $0.85 \to 0.92 \to 0.87 \to 0.91$ 로
      출렁인다. ``아직 안 수렴한 건가, 이미 수렴한 건가''를
      구분하기 어렵다.
    \item \kb{한두 에피소드가 전체를 지배한다}: $\rho$ 가
      극단적으로 큰 에피소드 1$\sim$2개가 평균의 절반 이상을
      차지할 수 있다 (시뮬레이션에서 $\rho \cdot G_0$ 값이
      가장 큰 상위 1\% 에피소드가 전체 합의 상당 부분을
      담당). ``학습이 한두 운 좋은 게임의 결과에 좌우된다''.
    \item \kb{$\varepsilon$ 을 키우면 더 악화된다}:
      $\varepsilon$-greedy 의 $\varepsilon$ 이 크면 $b$ 와
      $\pi$ 의 차이가 커져 $\rho$ 의 분포가 더 넓어진다.
      ``탐험을 많이 하면 데이터가 좋아진다''는 on-policy
      직관이 off-policy 에서는 \textbf{탐험이 많을수록
      분산이 커진다}는 반대 방향으로 작용한다.
  \end{itemize}
\end{frame}

\begin{frame}{자주 하는 실수: ``분모가 0'' 무시 --- 그리고
지원 조건}
  $\rho = \prod_t \frac{\pi(a_t|s_t)}{b(a_t|s_t)}$ 에서
  \textbf{분모} $b(a_t|s_t) = 0$ 인 상태·행동이 나타나면
  $\rho$ 가 \textbf{정의되지 않는다}.
  \begin{itemize}
    \item $\varepsilon$-greedy $b$ ($\varepsilon > 0$)에서는
      모든 행동을 (작지만) nonzero 확률로 고르므로
      \textbf{괜찮}.
    \item \textbf{결정론적} 행동 정책을 쓰면 $b$ 가 0인
      행동이 $\pi$ 에서 nonzero 라면 $\rho = \infty$.
  \end{itemize}
  \vskip0.4em
  $\rho$ 의 분모 0은 ``이 궤적은 $b$ 에서는 불가능하지만
  $\pi$ 에서는 가능''이라는 \kb{지원(support) 불일치}를
  의미 --- 이런 경우 중요도 샘플링은 \textbf{원래부터
  성립하지 않는다}(편향도 분산도 논할 여지 없이 추정 자체가
  무의미).
  \vskip0.4em
  \begin{block}{실전 규칙}
    $b(a|s) = 0$ 인 (상태, 행동) 쌍은 \textbf{평가 대상에서
    제외}하거나, 그 에피소드를 \textbf{통째로 버린다}.
    ``분모가 0이면 그냥 1로 치자''는 \textbf{가장 흔한
    실수}.
  \end{block}
  \vskip0.4em
  \kb{지원 조건 (support condition)}:
  \[
  \text{supp}(\pi) \;\subseteq\; \text{supp}(b)
  \]
  목표 정책이 행동하는 (상태, 행동) 쌍의 집합이 행동 정책의
  지원에 \textbf{포함}되어야 한다. 이 조건이 깨지면 어떤
  가중치로도 \textbf{보정 불가}. $\varepsilon > 0$ 으로 모든
  행동을 (작게라도) 포함하는 것이 표준.
\end{frame}

\begin{frame}{on-policy vs off-policy: 트레이드오프}
  \scriptsize
  \begin{tabular}{@{}p{0.24\linewidth}p{0.33\linewidth}p{0.33\linewidth}@{}}
    \toprule
     & \textbf{On-policy} (예: $\varepsilon$-greedy MC) & \textbf{Off-policy} (예: 중요도 샘플링) \\
    \midrule
    데이터 수집 정책 & 학습하는 정책 \textbf{자체} & 임의의 다른 정책 ($b \neq \pi$) \\
    편향 & 없음 (평가 대상과 동일) & 보정 전에는 있음, 보정 후에는 없음 \\
    분산 & \textbf{낮음} & \textbf{높음} ($\rho$ 폭증) \\
    데이터 효율 & 낮음 (탐험 비용) & 높음 (기존 데이터 재사용 가능) \\
    구현 난이도 & 간단 & $\rho$ 계산 + 분산 다스림 필요 \\
    실용 예 & 5.2절 블랙잭 MC 제어 & Ch.\,9 DQN 경험재현, Ch.\,11 PPO \\
    \bottomrule
  \end{tabular}
  \vskip0.8em
  \begin{block}{off-policy 의 진짜 가치}
    ``사람의 시연 데이터, 시뮬레이터에서 모은 데이터, 과거
    버전의 정책이 모은 데이터''를 \textbf{재사용}할 수
    있다는 점 --- Ch.\,12(모방학습)에서 사람의 동작 데이터를
    강화학습에 어떻게 쓸지 다룰 때 이 off-policy 도구가
    \textbf{필수 전제}.
  \end{block}
\end{frame}

\begin{frame}{자주 묻는 질문 (5.3) --- 단일 샘플 vs 전체 궤적}
  \textbf{``단일 샘플''과 ``전체 궤적'' 중요도 샘플링의
  차이는?}
  \begin{itemize}
    \item 위에서 쓰던 $\rho = \prod_t \frac{\pi(a_t|s_t)}{b(a_t|s_t)}$
      은 \kb{전체 궤적 (full trajectory)} 중요도 --- 에피소드
      전체 확률의 비율.
    \item 상태가치 $V^\pi(s)$ 를 추정할 때는 \kb{상태 $s$ 이후
      부분}(partial trajectory)만 쓰면 충분 --- 분모가 ``$s$
      에서 시작하는 궤적 확률의 비율''이라 전체 궤적보다
      \textbf{분산이 작다}.
    \item $Q^\pi(s,a)$ 를 추정할 때는 해당 $(s,a)$ 이후 부분만
      쓴다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{실전 원칙}
    \textbf{가능한 한 짧은(목표와 관련된) 부분 궤적만 가중치로
    쓴다} --- 이것이 분산을 줄이는 가장 간단한 방법.
    (5.2절 블랙잭에서 ``플레이어 합 12 이상 상태에서만 $Q$ 를
    학습''한 것도 같은 원리 --- 시작 상태부터가 아니라
    \textbf{관심 상태 이후}의 리턴만 쓴다.)
  \end{block}
\end{frame}

\begin{frame}{자주 묻는 질문 (5.3) --- 뭐가 더 나은가}
  \textbf{On-policy 와 off-policy 중 뭐가 더 나은가요?}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \kb{On-policy}
      \begin{itemize}
        \item 구현 \textbf{간단}, 분산 \textbf{작음}.
        \item 단, \textbf{학습하는 정책 자체로} 행동해야
          하므로 탐험(무작위 행동)의 \textbf{비용}을
          감수.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \kb{Off-policy}
      \begin{itemize}
        \item 다른 정책(심지어 \textbf{사람의 시연 데이터})이
          모은 경험을 \textbf{재사용} $\Rightarrow$ 데이터
          효율이 좋을 \textbf{수} 있다.
        \item 단, 중요도 샘플링의 \textbf{분산 문제}를
          다뤄야.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.6em
  Ch.\,6 의 \kb{Q-learning} 은 이 트레이드오프를
  \textbf{다른 방식}(중요도 샘플링 없이 off-policy 를 구현)으로
  풀어낸다.
  \vskip0.4em
  {\footnotesize ``off-policy 의 데이터 효율이 항상
  on-policy 보다 좋다''는 말은 \textbf{항상 참은
  아니다} --- $\rho$ 의 분산이 극단적으로 커지는 경우(행동
  정책과 목표 정책의 지원이 겹치는 정도가 낮음)에는
  \textbf{오히려 나빠질 수 있다}.}
\end{frame}

\begin{frame}{MC 의 한계: 에피소드가 끝나야만 학습}
  MC 는 모델이 필요 없다는 강력한 장점이 있지만,
  \textbf{에피소드가 끝나야만} 학습할 수 있다는 근본적 제약이
  있다 --- 리턴 $G_t$ 자체가 \textbf{에피소드 끝까지의 보상을
  다 더한 값}이기 때문.
  \vskip0.4em
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{block}{continuing task(끝이 없는 과업)라면 MC 는
      아예 쓸 수 없다}
        ending 이 없으면 ``에피소드 끝까지의 보상 합''이라는
        $G_t$ 자체가 \textbf{정의되지 않는다}.
      \end{block}
      \vskip0.4em
      대신 \kb{H 스텝 절단 리턴}
      \[
      G_t^{(H)} = R_t + \gamma R_{t+1} + \cdots +
      \gamma^{H-1} R_{t+H-1}
      \]
      로 대체할 수 있지만 \textbf{H 를 몇으로 잡을지}를
      사람이 정해야 한다 --- H 가 너무 \textbf{짧으면
      편향}(먼 미래 보상 무시), 너무 \textbf{길면
      분산}(먼 미래 무작위성 누적).
    \end{column}
    \begin{column}{0.54\textwidth}
      매 스텝 $+1$, $\gamma = 0.9$ 인 continuing task ---
      \textbf{진짜 리턴} $\frac{1}{1-\gamma} = 10$:
      \small
      \begin{tabular}{@{}cl@{}}
        \toprule
        $H$ & 절단 리턴 $G_t^{(H)}$ \\
        \midrule
        5   & 4.0951 \\
        20  & 8.7842 \\
        50  & 9.9485 \\
        200 & 10.0000 \\
        \bottomrule
      \end{tabular}
      \vskip0.4em
      ``에피소드가 끝나면 \textbf{알아서 결정되는} 문제''가
      ``H 를 \textbf{수동으로 고르는} 문제''로 변했다.
    \end{column}
  \end{columns}
  \vskip0.4em
  \begin{block}{Ch.\,6 의 시간차 학습}
    ``에피소드가 끝나기를 기다리지 않고, \textbf{한 스텝만
    보고도 즉시 갱신한다}''는 완전히 다른 절충안으로 이
    문제를 푼다.
  \end{block}
\end{frame}

\begin{frame}[fragile]{실습: 중요도 샘플링 직접 구현}
\begin{lstlisting}
GAMMA = 0.9

def sample_episode(rng, start=2, n_max=200):
    """행동 정책 b(50-50)로 한 에피소드 샘플링."""
    s, ep = start, []
    for _ in range(n_max):
        a = 0 if rng.random() < 0.5 else 1
        ns, r, done = step(s, a)
        ep.append((s, a, r)); s = ns
        if done: break
    return ep

def estimate(rng, n_episodes, start=2):
    """세 가지 추정치: (1) 그냥 평균 (2) 중요도 가중 평균."""
    naive, isw = [], []
    for _ in range(n_episodes):
        ep = sample_episode(rng, start)
        G = 0.0
        for t in reversed(range(len(ep))):
            G = ep[t][2] + GAMMA * G
        naive.append(G)                          # 편향됨: V^b(start)
        all_right = all(a == 1 for _, a, _ in ep)
        L = len(ep)
        rho = (2.0 ** L) if all_right else 0.0   # b=50-50, pi=right
        isw.append(rho * G)                      # 편향 없음: V^pi
    return np.mean(naive), np.std(naive), np.mean(isw), np.std(isw)
\end{lstlisting}
  {\footnotesize 같은 데이터, 다른 가중치 $\Rightarrow$
  \textbf{다른 정책의 가치를 추정} --- 중요도 샘플링의
  본질.}
\end{frame}

\begin{frame}{추가 확인: $\rho$ 분포가 길이에 따라 퍼진다}
  \scriptsize
  \begin{tabular}{@{}cl@{}}
    \toprule
    $L$ & P($\rho > 0$) = $2^{-L}$ \quad / \quad Var[$\rho$] = $2^L - 1$ \\
    \midrule
    $L = 1$  & $0.50000$ \quad / \quad $1$ \\
    $L = 2$  & $0.25000$ \quad / \quad $3$ \\
    $L = 3$  & $0.12500$ \quad / \quad $7$ \\
    $L = 4$  & $0.06250$ \quad / \quad $15$ \\
    $L = 6$  & $0.01562$ \quad / \quad $63$ \\
    $L = 8$  & $0.00391$ \quad / \quad $255$ \\
    $L = 10$ & $0.00098$ \quad / \quad $1023$ \\
    \bottomrule
  \end{tabular}
  \vskip0.8em
  20만 에피소드 샘플링(시드 2)에서 직접 확인:
  \[
  P(\rho = 4) = 0.250, \qquad P(\rho = 0) = 0.750
  \]
  \[
  \mathbb{E}[\rho] = 1.001 \ \text{(이론 } 1\text{)}, \qquad
  \text{Var}[\rho] = 3.002 \ \text{(이론 } 2^2 - 1 = 3\text{)}
  \]
  \vskip0.4em
  $L$ 이 \textbf{커질수록} $\text{Var}[\rho] = 2^L - 1$
  (지수적 폭증).
\end{frame}

\begin{frame}{자주 하는 실수: $\rho$ 의 분산을 ``평균 에피소드
길이''에 대입하기}
  $\text{Var}[\rho] = 2^L - 1$ 은 \textbf{$L$ 이 같으면} 같은
  분산이지만, 실전에서는 \textbf{에피소드 길이가 고정되어
  있지 않다}. 5.2절 블랙잭처럼 ``언제 멈출지''가 변수인
  환경에서는 에피소드 길이가 2에서 100까지 \textbf{분포}.
  \vskip0.5em
  이 경우 $\text{Var}[\rho]$ 는 에피소드 \textbf{길이 분포}와
  행동/목표 정책의 차이 \textbf{모두}를 반영. ``에피소드 평균
  길이가 5니까 $\text{Var} \approx 31$ 이겠지''라고
  \textbf{평균 길이에 대입}하는 것은 \textbf{잘못}이다 ---
  \[
  \mathbb{E}[2^L] \ \ge\ 2^{\,\mathbb{E}[L]}
  \]
  (Jensen 부등식, $2^L$ 이 볼록) 이므로 \textbf{평균 길이에
  넣은 값보다 실제 분산이 더 크다}. 길이 분포가
  \textbf{넓을수록}(표준편차가 클수록) 격차는 커진다.
  \vskip0.4em
  \begin{block}{의미}
    이것이 ``에피소드 길이가 불규칙한 환경에서 off-policy 가
    \textbf{특히 어렵다}''는 말의 정확한 의미.
  \end{block}
\end{frame}

  \begin{block}{Ch.\,6 예고: 시간차 학습 (Temporal-Difference)}
    ``기다림 없이 배운다'' --- 에피소드가 끝나기를 기다리지
    않고 \textbf{한 스텝만 보고도 즉시 갱신}. 이번 장의
    ``에피소드를 끝까지 기다린다''는 대가를 없애는
    것이 바로 TD.
  \end{block}
\end{frame}

\begin{frame}{이번 주차 정리}
  \begin{enumerate}
    \item \kb{MC 예측 (5.1)} --- \textbf{기댓값 = 샘플
      평균}이라는 통계학 원리를 강화학습에 그대로 이식.
      잔디깎는 기 MDP 에서 전이확률 단 한 줄도 쓰지 않은 채
      DP 의 정확해와 \textbf{같은 숫자}($0.286$ 등)에
      도달. \textbf{첫방문} MC 가 무편향인 이유는
      \textbf{마르코프 성질 + 대수의 법칙} 두 줄. 뒤에서부터
      훑으면 ``마지막 방문''을 세게 되어 \textbf{편향된
      추정치}(0.756 vs 0.713)가 됨 --- 코딩 실수가 통계적
      편향으로 바뀌는 대표적 예.
    \item \kb{MC 제어 (5.2)} --- $V$ 대신 $Q(s,a)$ 를
      추정하고 $\arg\max$ 로 결정론적 정책을 뽑음.
      $\varepsilon$-greedy 로 블랙잭 30$\sim$40만 에피소드를
      돌리면 \textbf{교과서 기본 전략표}(17 이상 스틱,
      12$\sim$13 히트)가 학습만으로 재현(104/110 $\to$ 105/110
      일치). 학습 대상은 \textbf{행동 정책 $\pi$ 의
      $Q^\pi$} (on-policy) --- $Q^{*}$ 보다 항상 약간
      낮게 측정. ``방문 횟수 = 신뢰도''.
    \item \kb{중요도 샘플링 (5.3)} --- off-policy 의
      편향을 확률의 비율
      $\rho = \prod_t \frac{\pi(a_t|s_t)}{b(a_t|s_t)}$
      로 보정. 5칸 회랑에서 손으로 확인: 편향은 없애주되
      $\text{Var}[\rho] = 2^L - 1$ 로 \textbf{지수적으로
      분산 폭증}. ``분모 0 무시'' 금지(지원 조건
      $\text{supp}(\pi) \subseteq \text{supp}(b)$),
      on/off-policy 트레이드오프를 정리.
  \end{enumerate}
\end{frame}

\begin{frame}{다음 주차 예고: Chapter 6. 시간차 학습 (TD)}
  \begin{itemize}
    \item 이번 장의 MC 는 \textbf{모델 없이} 배운다는 대가로
      \textbf{에피소드를 끝까지 기다려야} 했다.
    \item TD 는 그 기다림을 없앤다 --- \textbf{매 스텝}에서
      ``한 스텝만 보고도 즉시 갱신''
      ($Q$ 를 한 스텝 분만 ``내다보고'' 갱신).
    \item \kb{SARSA} (on-policy TD)와 \kb{Q-learning}
      (off-policy TD, 중요도 샘플링 \textbf{없이} off-policy 를
      구현)를 배우고, CliffWalking 예제로 두 방법의 차이를
      체감한다.
    \item ``모델의 유무'' × ``갱신을 언제 하는가''라는
      강화학습 지도의 두 축이 모두 놓인다 --- Ch.\,9 DQN
      (신경망 + 경험재현)과 Ch.\,11 PPO (클리핑된 확률비)로
      이어지는 모든 것이 이번 두 주(Ch.\,5 MC + Ch.\,6 TD)의
      위에서 세워진다.
  \end{itemize}
\end{frame}
