  \footnotesize
  \begin{itemize}
    \item 등고선이 \textbf{늘어진 타원}인 이유 = 앞선 고유값
      비율 ($\approx 46$) 이 바로 그 타원의 축 비율이기 때문.
    \item 매 스텝 ``가장 가파르게 내려가는 방향''을 걷는데, 타원
      등고선에서 그 방향은 골짜기 바닥(긴 축)이 아니라
      \textbf{대각 방향} --- 그래서 경로는 짧은 축을 수직으로
      오가며 \kb{zigzag}(빨간 점들).
    \item 등고선이 원형(두 고유값이 같을 때)에 가까우면 진동이
      사라져 직선에 가까워짐.
  \end{itemize}
\end{frame}
