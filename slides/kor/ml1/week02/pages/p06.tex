\begin{frame}{모델: 특징의 가중합}
  선형회귀는 입력 $\boldsymbol{x} = (x_1, \ldots, x_n)$ 에서 출력을 이렇게 예측한다:
  \[
  h_{\boldsymbol{w}}(\boldsymbol{x}) = w_0 + w_1 x_1 + w_2 x_2 + \cdots + w_n x_n,
  \qquad h_{\boldsymbol{w}}: \mathbb{R}^n \to \mathbb{R}
  \]
  \begin{itemize}
    \item $w_0$ = \kb{절편}(bias), \; $w_1, \ldots, w_n$ = 각 특징의 \kb{가중치}
    \item 표기 단순화: $x_0 = 1$ 을 항상 추가하면
      \[h_{\boldsymbol{w}}(\boldsymbol{x}) = \boldsymbol{w}^T \boldsymbol{x} = \sum_{j=0}^n w_j x_j\]
      --- bias까지 하나의 \textbf{내적}으로 통일
  \end{itemize}
\end{frame}
