  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item 1801년 발견된 소행성 \kb{세레스}가 태양 뒤로 사라지자,
          24세 \kb{가우스}가 \textbf{단 며칠치 부정확한 관측만으로}
          재출현 위치를 정확히 예측.
        \item 쓴 방법 = \textbf{오차 제곱합 최소화} --- 지금 이 절의
          바로 그 $J(\boldsymbol{w})$.
      \end{itemize}
      \vskip0.3em
      \begin{block}{200년 넘게 쓰이는 이유}
        {\footnotesize ``아름다워서''가 아니라, 관측(데이터)이
        참값에 잡음이 섞인 구조가 \kb{실존 세계에 끊임없이
        반복}되기 때문.}
      \end{block}
    \end{column}

      {\footnotesize 궤도를 정하는 6개 숫자 $\boldsymbol{w}$:
      $a$(장반경), $e$(이심률), $i$(경사각), $\Omega$(승교점 경도),
      $\omega$(근일점 인수), $\theta$(위치) --- 직선의 $(w_0,w_1)$이
      6개로 늘어난 것과 본질이 같다.}

  \vskip0.2em
  {\footnotesize \kq{궤도는 타원인데 우리가 배운 건 직선 아닌가?}
  --- 최소제곱법이 재는 건 ``직선''이 아니라 \textbf{예측값과 관측값의
  오차 제곱합}. $h$가 직선이든 타원 궤도 방정식이든
  $J(\boldsymbol{w})=\sum(h_{\boldsymbol{w}}-\text{관측})^2$ 구조는
  동일 --- 가우스는 6개짜리 비선형 $h$로 풀었을 뿐, \textbf{같은
  원리, 더 많은 파라미터}.}
