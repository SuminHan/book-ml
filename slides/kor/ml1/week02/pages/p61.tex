\section{2.3 로지스틱회귀: 시그모이드에서 PR-AUC까지}
\begin{frame}{시그모이드: 실수를 확률로}
  로지스틱함수(시그모이드)는 정확히 ``$-\infty \sim +\infty$ 를
  $0 \sim 1$ 로 바꾸는'' 변환:
  \[
  \sigma(z) = \frac{1}{1+e^{-z}}
  \]
  \begin{itemize}
    \item $z \to +\infty \Rightarrow \sigma \to 1$, \;
      $z \to -\infty \Rightarrow \sigma \to 0$, \;
      $\sigma(0) = 0.5$.
    \item 어떤 실수든 이 함수를 통과하면 0$\sim$1 ---
      ``확률''로 해석 가능.
    \item \[h_{\boldsymbol{w}}(\boldsymbol{x}) = \sigma(\boldsymbol{w}^T\boldsymbol{x}) = \frac{1}{1+e^{-\boldsymbol{w}^T\boldsymbol{x}}}\]
      --- ``$\boldsymbol{x}$ 가 \textbf{양성 클래스(1)}일 확률''로 해석.
  \end{itemize}
\end{frame}
