      \vskip0.4em
      {\footnotesize $w_0=1$ 고정, 세 학습률($\alpha = 0.1, 0.01,
      0.001$)로 $w_1$만 경사하강 --- 스텝별 비용 $J$, 같은
      출발점.}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item 세 곡선은 \textbf{같은 점에서 출발해 같은 최솟값}
          (점선)으로 수렴.
        \item $\alpha = 0.1$ 이 가장 빠르게 바닥에 닿고,
          $\alpha = 0.01$ 은 약 \textbf{10배 느리다}.
        \item $\alpha = 0.001$ 은 300스텝이 지나도 바닥에서 꽤 먼
          채로 거의 움직이지 못함.
        \item 1D 그림은 ``속도''를 보여준다 --- 학습률은
          ``가는 방향''이 아니라 \textbf{``걸음 수''}를 정하는
          하이퍼파라미터.
      \end{itemize}
    \end{column}
  \end{columns}
