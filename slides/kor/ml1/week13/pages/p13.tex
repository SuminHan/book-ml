\begin{frame}{Zachary's Karate Club --- 표준 테스트 데이터셋}
  \begin{itemize}
    \item 1977년 인류학자 \kb{웨인 재커리}(Wayne Zachary)가
      한 가라테 동호회 \textbf{34명}의 친분 관계를 기록.
    \item 마침 이 동호회가 \textbf{감독(node 0)}과 \textbf{관장(node 33)}
      두 파벌로 \textbf{실제 분열}.
    \item 34개 노드, \textbf{78개}의 친분 관계(edge)뿐인 작은 그래프.
    \item 그래프 임베딩이 ``제대로 동작하는지'' 확인하는
      \textbf{표준 벤치마크}로 지금까지도 사용.
  \end{itemize}
  \vskip0.5em
  \begin{block}{성공의 기준}
    \textbf{라벨(파벌 소속) 없이 순수하게 그래프 구조만} 보고
    임베딩했는데도 두 파벌이 벡터 공간에서 자연스럽게 갈라지면 성공.
  \end{block}
\end{frame}
