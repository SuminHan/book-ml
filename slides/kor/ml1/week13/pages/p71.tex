\begin{frame}{노트북 Part 3: 합성 도로망}
  \begin{itemize}
    \item 작은 \textbf{합성 도로망}($4\times 4$ 격자, 교차로 = 노드)에서
      인공적인 ``\textbf{출근 파도}'' 시계열을 만들고.
    \item 두 모델을 비교:
    \item (a) \textbf{1홉 이웃 평균}만 보는 모델
    \item (b) \textbf{3홉 이웃 평균 + 과거 3단계 자기 평균}을
      보는 모델
    \item 결과: \textbf{이웃 범위를 넓히고 과거를 함께 보면}
      \textbf{오차가 줄어든다}.
  \end{itemize}
\end{frame}
