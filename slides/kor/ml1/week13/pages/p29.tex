\begin{frame}{해석: 방향은 보인다, 크기는 작다}
  \begin{itemize}
    \item 34개 노드라는 작은 스케일에서 세 설정의 차이는 \textbf{작고},
      원본 논문의 대규모 패턴까지 그대로 재현되지는 않는다
      (데이터가 작아 노이즈가 큰 것은 자연스러움).
    \item 그래도 방향을 짚자: \textbf{DFS형 $(10,1)$}에서 node 2의
      상위 3개 유사 노드(12, 19, 17)가 \textbf{모두 감독 파벌}에
      속하는 \textbf{가장 깔끔한 파벌 분리}.
    \item 카라테 클럽에서 파벌 = 커뮤니티이므로, 여기서는
      커뮤니티 분리 품질이 곧 ``잘 됐다''의 지표.
  \end{itemize}
\end{frame}
