\begin{frame}{재료가 그대로 실전에 옮겨진다}
  \begin{itemize}
    \item 네 케이스 모두, 이 장의 ``재료''인 \kb{그래프 자체}가
      \textbf{도메인 지식에서 직접 만들어지는} 형태
      (접촉 그래프, 거래 그래프, 도로망, 사용자-아이템 그래프).
    \item 이 장의 ``연산''인 \kb{반복해서 이웃과 값을 합친다}는
      것이 각 도구의 \textbf{핵심}에 들어 있다.
  \end{itemize}
  \vskip0.5em
  \begin{block}{그래프 학습에서 ``모델링''이 ``프로그래밍''보다
  중요한 이유}
    도메인이 달라질수록 \textbf{그래프를 어떻게 만드는가}
    (\textbf{어떤 것을 노드로, 어떤 것을 엣지로} 하는가)의 선택이
    \textbf{모델의 성능을 결정}한다.
  \end{block}
\end{frame}
