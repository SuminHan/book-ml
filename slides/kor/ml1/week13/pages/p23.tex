\begin{frame}{계수를 움직이면 탐색 스타일이 기울어진다}
  \begin{itemize}
    \item $p = q = 1$: 모든 가중치가 1 $\to$ 정확히
      \textbf{일괄적 무작위 걷기}.
    \item \kb{$p < 1$} ($1/p > 1$): 직전 노드로의 \textbf{뒤로 감}
      (backtracking) 확률이 올라간다.
    \item \kb{$q < 1$} ($1/q > 1$): 방금 탐색한 \textbf{로컬 영역 안에
      머물기} 확률이 올라간다.
    \item \kb{$q > 1$} ($1/q < 1$): 이미 본 영역을 다시 밟는 것이
      \textbf{억제}되어 걷기가 \textbf{새 영토로 확장}하려 한다.
  \end{itemize}
\end{frame}
