  \begin{center}
  \includegraphics[height=4.7cm]{ch05_1_geometry.png}
  \end{center}
  \vskip0.2em
  \footnotesize
  \begin{itemize}
    \item 검은 선 = 결정 경계($w^\top x+b=0$), 남색 점선 = 마진 경계($w^\top x+b=1$).
    \item 빨간 화살표 = $w$ 방향으로 이동한 거리, 정확히 $\dfrac{1}{\|w\|}$.
    \item 이 거리가 다음 쪽에서 마진 공식 $\dfrac{2}{\|w\|}$로 이어진다.
  \end{itemize}
\end{frame}
