\begin{frame}{그림으로: $\gamma$ 스윕 --- 과소적합에서 과적합까지}
  \begin{center}
  \includegraphics[height=5.2cm]{ch05_3_gamma_sweep.png}
  \end{center}
  \vskip0.2em
  \footnotesize
  \begin{itemize}
    \item $\gamma=0.05$: test acc $0.47$ --- 경계가 거의 직선, \textbf{과소적합}
      (동전 던지기보다 나쁨).
    \item $\gamma=0.3\sim1.0$: test acc $1.00$ --- 두 원을 매끈하게 갈라내는
      적당한 구불거림.
    \item $\gamma=30.0$: train/test acc $1.00$이지만 경계가 점마다 뾰족하게
      감쌈 --- 노이즈 있는 데이터라면 \textbf{과적합}으로 무너질 모양.
  \end{itemize}
\end{frame}
