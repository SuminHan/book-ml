      \begin{itemize}\small
        \item \textbf{실선} = 순전파: $z_1 \to a_1 \to z_2 \to a_2$
          (입력 $\to$ 은닉 $\to$ 출력)
        \item \textbf{점선} = 역전파: $\delta_2 \to \delta_1$
          (출력 $\to$ 은닉)
      \end{itemize}
      \vskip0.4em
      {\footnotesize 역전파는 이름뿐이 아니라,
      \textbf{오차 값이 실제로 층을 거슬러 흘러가는} 것.
      각 층의 그래디언트 $\partial L/\partial W_2$,
      $\partial L/\partial W_1$이 여기서 만들어진다.}
    \end{column}
  \end{columns}
