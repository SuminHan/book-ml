\begin{frame}{손으로 한 번: 2-2-1 순전파 + 손실}
  \small
  $x = (1,0)$, $y = 1$ (XOR(1,0)=1), \;
  $W_1 = \begin{bmatrix}0.5 & 0.3 \\ -0.5 & 0.8\end{bmatrix}$,
  $b_1 = (0.1, -0.2)$, \; $W_2 = (0.7, -0.6)$, $b_2 = 0.1$:
  \vskip0.5em
  \begin{itemize}
    \item \kb{순전파}: $z_1 = x^\top W_1 + b_1 = (0.6,\; 0.1)$, \;
      $a_1 = \sigma(z_1) \approx (0.646,\; 0.525)$
    \item $z_2 \approx 0.646 \times 0.7 + 0.525 \times (-0.6) + 0.1 \approx 0.237$, \;
      $a_2 \approx 0.559$
    \item \kb{손실}: $L = \tfrac{1}{2}(a_2 - y)^2 \approx 0.097$
  \end{itemize}
  \vskip0.5em
  \kb{역전파}:
  \[
  \delta_2 \approx (-0.441) \times 0.559 \times 0.441 \approx -0.109
  \]
  \[
  \delta_1 = W_2 \delta_2 \odot \sigma'(z_1)
  \approx (-0.076,\; 0.065) \odot (0.229,\; 0.249)
  \approx (-0.017,\; 0.016)
  \]
  {\footnotesize 행=입력 컨벤션: $\partial L/\partial W_1 = x \otimes \delta_1$,
  $x_2 = 0$이므로 두 번째 행은 0.}
\end{frame}
