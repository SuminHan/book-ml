\begin{frame}{``왜 딥러닝인가''를 어떻게 채점하는가}
  이 질문을 \textit{형식적으로만} 채우는 것과 \textit{내용적으로}
  채우는 것의 차이 = 8.2의 \textbf{``1점 vs 3점''} 구별과
  \textbf{정확히 같은 구조}.
  \vskip0.5em
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \kb{형식(2점)}
      \begin{itemize}
        \item ``왜 고전 ML이 아니라 딥러닝인가''에 \textbf{답변이
          있었는지} 확인만
        \item ``팀이 \textit{`데이터가 이미지라서'}라고 답했다''로
          끝나는 리뷰
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \kb{내용(3점)}
      \begin{itemize}
        \item 그 답변이 \textbf{이 학기의 개념으로 반박 가능한 근거}
          인가까지 따짐
        \item 예: ``이미지 데이터에 CNN을 썼는데, CNN의
          \textbf{국소 receptive field + 파라미터 공유}가 이
          데이터의 어떤 구조(Ch10.1)에 맞는지 설명할 수
          있나요?''
        \item 발표팀이 \textbf{Ch10.1 공식이나 자기 코드에 다시
          들어가야} 답할 수 있는 질문
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.6em
  {\footnotesize 판별 기준은 8.2 그대로: \textbf{발표팀이
  ``네, 알겠습니다''로 넘길 수 있으면 2점, 새 계산·새 확인이
  없으면 답할 수 없으면 3점.}}
\end{frame}
