\begin{frame}{발표: 마지막 1분을 ``왜 DL''에 전념}
  \begin{itemize}
    \item 형식은 8.1과 동일(문제 정의 $\to$ 방법론 $\to$ 결과와
      정당화 $\to$ 한계 분석, 팀당 5~7분)
    \item \textbf{시간 배분을 고정}: 문제 정의(1분) $\to$ 데이터와
      정형/비정형 판정(1분) $\to$ 적용한 기법과 구조(2분) $\to$ 결과
      + GBDT 비교·수식적 정당화(2분) $\to$ \kb{``왜 DL인가''(1분)}
  \end{itemize}
  \vskip0.5em
  \begin{block}{그 1분에 채울 것}
    §``왜 고전 ML이 아니라''의 \textbf{2축 + 2숫자}.
    ``정형인데도 신경망''이라면 GBDT와의 \textbf{숫자 비교 필수}.
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item 8.1처럼 ``\textbf{잘 안 된 부분}''을 정직하게 말하는 습관
      유지 — 학습이 안 된 부분을 \textbf{Ch09}(역전파/그래디언트)·
      \textbf{9.3}(정규화)·\textbf{10.2}(과적합)의 개념으로 진단하는
      것이 ``방법론을 실제로 이해했는가''의 증거
    \item 8.1이 ``어떤 고전 ML 모델이 이 데이터에 \textbf{맞는가}''를
      물었다면, 이번엔 ``\textbf{애초에} 딥러닝이 필요한 문제인가''
      부터 스스로 판단
  \end{itemize}
\end{frame}
