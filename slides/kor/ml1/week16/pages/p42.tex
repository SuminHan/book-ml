\begin{frame}{핵심 숫자 ③: 교정은 정확도와 별개}
  \small
  \kb{③ 교정은 \textbf{정확도와 별개}}
  \begin{itemize}
    \item ``\textbf{0.90+ 확률}로 예측한 샘플'' 100건의 실제
      악성 비율 = \textbf{0.980} — \textit{대략}은 맞지만
      \textbf{확률 스케일 전체}가 맞는다는 증거는 아님
    \item \textbf{정확도 0.953}(분류)과 \textbf{0.98}(확률
      신뢰도)는 \textbf{두 개의 다른 숫자} — ``이 모델의
      \textbf{확률}을 위험도 결정에 쓸 수 있는가''가 의미를
      갖는 이유
  \end{itemize}
  \vskip0.5em
  \begin{block}{이 실험이 §``손으로 한 번''의 비판을 검증}
    ``\textbf{test 0.95 vs 0.94는 잡음 수준}''이라는 리뷰어의
    말이 이 실험에서 \textbf{실제로 일어나는 일}이라 \textbf{
    반박 가능} — 세 숫자(①·②·③)가 가상 발표 비판의
    \textit{근거}.
  \end{block}
\end{frame}
