\begin{frame}{이번 주차 개요}
  \begin{itemize}
    \item 의대생은 해부학·생리학·약리학 같은 \textbf{방대한 기초}를
      먼저 익히고 나서야 특정 전공을 집중 훈련한다. LLM도
      비슷: 방대한 텍스트로 다음 단어 예측을 익혀(\kb{사전학습})
      폭넓은 지식을 흡수하고, 적은 데이터로 다듬고(\kb{파인튜닝}),
      ``사람이 실제로 원하는 방향''으로 정교하게 다듬는다
      (\kb{포스트트레이닝}).
    \item \kb{17.1 언어모델링} --- ``다음 토큰 예측''이 \kq{문장
      전체 분포 학습과 동일한 과제}임을 사슬 분해로 보이고,
      BPE·softmax·온도·교차 엔트로피·PPL을 작은 예제로 확인한
      뒤 사전학습/파인튜닝의 현대 도식으로 마무리.
    \item \kb{17.2 프롬프팅} --- 가중치 없이 \textbf{입력만으로}
      행동을 바꾸기: zero-shot $\to$ few-shot $\to$ CoT
      스펙트럼, temperature·top-k·top-p, 그리고 환각.
    \item \kb{17.3 포스트트레이닝} --- InstructGPT의 관찰에서
      출발해 SFT $\to$ RLHF(보상모델 + PPO) $\to$ DPO를 짚고,
      LoRA/PEFT와 RAG·에이전트까지 모은 전체 지도.
  \end{itemize}
  \vskip0.5em
  \begin{block}{이 장을 마치면 (1$\sim$16장 이후의 심화·응용 부록)}
    \begin{itemize}
      \item softmax·교차 엔트로피·퍼플렉시티를 작은 예제로
        직접 계산, BPE가 ``단어 조각'' 어휘를 만드는 이유를
        설명할 수 있다.
      \item SFT·RLHF·DPO를 그림으로 구분하고, ``그럴듯한
        텍스트''가 ``사람이 원하는 답변''과 왜 다른지
        설명할 수 있다.
    \end{itemize}
  \end{block}
\end{frame}
