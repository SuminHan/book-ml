\begin{frame}{손으로 한 번: 2D 밀집 검색}
  차원 768을 2로 낮춘 예. \; 질문 ``자동차 연비'' $\boldsymbol{q} = (4, 1)$,
  문서:
  \begin{itemize}
    \item $\boldsymbol{d}_{1} = (3, 2)$: ``차량의 연료 소비량은 1km당
      소모 연료로 측정한다''
    \item $\boldsymbol{d}_{2} = (2, 3)$: ``자동차 보험을 고르는 법''
    \item $\boldsymbol{d}_{3} = (1, 3)$: ``양자 컴퓨터는 정보를 어떻게
      처리하는가''
  \end{itemize}
  {\footnotesize $\boldsymbol{q}$와 $\boldsymbol{d}_{1}$은 \textbf{단어를
  하나도 공유하지 않는다}(자동차/차량, 연비/연료 소비량) ---
  \textbf{키워드 매칭이 못 찾는 경우}.}
  \vskip0.4em
  \textbf{1단계 — 내적}: \;
  $\boldsymbol{q}\cdot\boldsymbol{d}_{1} = 4\times3 + 1\times2 = \mathbf{14}$, \;
  $\boldsymbol{q}\cdot\boldsymbol{d}_{2} = 11$, \; $\boldsymbol{q}\cdot\boldsymbol{d}_{3} = 7$.
  \vskip0.3em
  \textbf{2단계 — 노름}: \;
  $\|\boldsymbol{q}\| = \sqrt{17} \approx 4.12$, \;
  $\|\boldsymbol{d}_{1}\| = \|\boldsymbol{d}_{2}\| = \sqrt{13} \approx 3.61$, \;
  $\|\boldsymbol{d}_{3}\| = \sqrt{10} \approx 3.16$.
\end{frame}
