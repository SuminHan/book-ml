    \begin{column}{0.55\textwidth}

      \vskip0.3em
      {\footnotesize 기준값(0.454)부터 기여를 크기순으로 쌓아 실제
      예측값(0.485)에 \textbf{정확히} 도달.}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item \kb{워터폴트(waterfall) 플롯} = 가산성을 \textbf{시각화}:
          기준값부터 각 특징의 기여를 하나씩(크기순) 쌓아가면 마지막에
          \textbf{정확히} 실제 예측값에 도착
        \item ``합이 정확히 일치한다'' = SHAP이 ``대략적인 설명''이
          아니라 \textbf{정확한 분해}
        \item 은행 예: ``신용점수 $-0.3$, 소득 $+0.1$''의 합이 정확히
          ``승인 확률이 0.2 낮아졌다''와 일치해야 하는 것과 같음
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
