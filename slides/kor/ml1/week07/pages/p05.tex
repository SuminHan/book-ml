\begin{frame}{경로 = 설명: 결정 트리의 강점}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \textbf{새 샘플: 루트 평가 $\to$ 이동 $\to$ 리프 값}
        \item \textbf{경로가 예측 설명}
      \end{itemize}
      \vskip0.6em
      \begin{block}{\textbf{설명가능성}(interpretability)}
        \small 루트에서 리프까지의 질문 경로를 사람이 읽을 수 있다.
        이 장(7.3)에서는 \textbf{수백 개의 트리}가 합쳐지면 이
        ``경로 = 설명'' 특성을 잃는 대가를 다시 논한다(SHAP).
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch07_1_path_explanation.png}
      \vskip0.3em
      {\footnotesize 루트에서 리프까지 예/아니오 경로를 따라가면 그
      자체가 예측의 설명이 된다.}
    \end{column}
  \end{columns}

\note{\begin{itemize}
\item 새 샘플 들어오면: (1) 루트에서 질문을 평가 $\to$ (2) 예/아니오에
      따라 좌·우 자식으로 이동 $\to$ (3) 리프면 그 값을 내놓고 끝
\item 이 \textbf{경로 자체가} ``왜 이렇게 예측했는가''의 설명이다
\item 대출 심사 예: ``소득 $<$ 3000만원 $\to$ 연령 $>$ 45세 $\to$
      \textbf{거절}''이라는 경로를 \textbf{그대로 읽어}주면 됨
\end{itemize}}
\end{frame}
