  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth]{ch07_1_boundary_staircase.png}
      \vskip0.3em
      {\footnotesize 왼쪽: 깊이 3, 오른쪽: 완전(제한 없음).}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{itemize}
        \item 2개 특징 400개 샘플(두 덩어리, 오버랩) —
          \textbf{둘 다 계단}
        \item 질문이 ``$x_j > t$?''뿐이라 경계는 \textbf{필연적으로}
          축에 평행한 직선 조각 — 대각선 한 줄도 못 그리다
        \item 왼쪽(깊이 3): 리프 8개의 비교적 단순한 계단
        \item 오른쪽(완전): 학습 데이터를 100\% 맞추려
          \textbf{학습 점 사이로 파고드는} 복잡한 계단(리프 59개)
        \item 숫자로: 깊이 3 $\to$ train 0.846 / test 0.792 \quad
          완전 $\to$ train \textbf{1.000} / test \textbf{0.733}
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.5em
  {\footnotesize ``학습 데이터를 더 잘 맞히는 모델이 더 좋은 모델''이라는
  직관이 여기서 처음 깨진다.}
\end{frame}
