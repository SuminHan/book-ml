\begin{frame}{이번 주차 개요}
  1996년 티브시라니(Tibshirani)의 한 생각 — 손실함수에 가중치 절댓값의 합을
  페널티로 더하면, 무관한 특징의 가중치가 \textbf{정확히 0}이 된다.
  특징이 수천 개여도 사람이 일일이 고를 필요가 없다.
  \vskip0.6em
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{이 주차를 마치면}
      \begin{itemize}\small
        \item L2는 축소, L1은 \textbf{정확히 0}(특징 선택)
        \item $\lambda$ = 편향-분산 \kb{손잡이}, 계수 경로
        \item k-겹 교차검증으로 $\lambda$ 고르기, U자 곡선
        \item train/val/test 3분할과 누수 진단
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{세 차시 한 줄}
      \begin{itemize}\small
        \item \kb{6.1} 원(L2) 대 마름모(L1)
        \item \kb{6.2} 검증 손실로 $\lambda$ 고르기
        \item \kb{6.3} test는 \textbf{딱 한 번}
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.4em
  {\footnotesize 이미 써본 그 손잡이: Week 5 소프트 마진 SVM의 목적함수
  $\frac{1}{2}\|w\|^2 + C\sum_i \xi^{(i)}$에서 $\|w\|^2$ 자체가 L2 페널티,
  $C$는 이 장의 $\lambda$와 같은 종류의 손잡이였다.}
\end{frame}
