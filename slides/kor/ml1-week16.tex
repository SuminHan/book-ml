\section{16.1 팀 프로젝트: 구조와 발표}

\begin{frame}{왜 Block B의 시작에 또 프로젝트를 하는가}
  \begin{itemize}
    \item Ch8.1(Block A 캡스톤)이 검증한 것: ``Ch02~07의 도구가 실제 데이터
      앞에서도 \textbf{작동하는가}'' — 정형 데이터 고전 ML의 완결
    \item 이번(Ch09~15의 도구)이 검증할 것: ``신경망과 그 이후의 도구가,
      이 데이터에 \textbf{정말 필요한가}''
    \item 모양은 같지만 무게가 다르다 — 전자는 ``도구가 맞는가'', 후자는
      ``\textbf{도구가 필요한가}''를 묻는다
    \item 후자는 ``\textbf{아예 안 쓰는 것(GBDT, Ch07)}과 비교해서''
      물어야 성립한다
  \end{itemize}
  \vskip0.6em
  \begin{block}{실전과의 연결}
    산업 현장 팀은 이미지·텍스트를 처음 만나면 ``신경망을 돌려보자''가 아니라
    ``이 문제를 정형 문제로 치환해서 GBDT로 풀 수 있는가''부터 묻는다 —
    정형 데이터에서는 지금도 GBDT 계열이 신경망과 비등하거나 이긴다.
    \kq{``딥러닝을 써야 한다''는 대명제로 시작하는 팀은 이미 감점 상태.}
  \end{block}
\end{frame}

\begin{frame}{두 층의 베이스라인: ``무작위''와 ``GBDT''}
  \begin{enumerate}
    \item \kb{1층: 무작위(또는 다수 클래스)} — 모델이 \textit{아무것도}
      배운 상태의 \textbf{하한}. 10클래스 균형 데이터면 무작위 정확도
      10\%
    \item \kb{2층: GBDT (Ch07.3)} — \textit{정형 도구의} \textbf{상한}
      에 가까운 기준. ``신경망이 정형 도구보다 나은가''를 묻는 기준선
    \item \kb{본인 모델(신경망)} — 위 두 층을 \textbf{둘 다} 넘어야
      ``이 데이터에 딥러닝이 필요했다''고 말할 수 있다
  \end{enumerate}

\begin{frame}{``예상대로 졌다''는 예상된 결과}
  \begin{itemize}
    \item digits 같은 \textbf{작은 정형} 데이터에서는 \textbf{신경망이
      GBDT를 (소폭) 이기지 못하는 것이 \textit{예상되는} 결과}
    \item Grinsztajn, Oyallon \& Varoquaux (2022), ``Why do tree-based
      models still outperform deep learning on typical tabular data?'':
      \textbf{저차원 정형 데이터에서는 GBDT가 신경망을 이긴다}
    \item 신경망의 우위는 \textbf{세 조건} 중 하나에서 온다 —
      (a) \textbf{차원이 높은} 데이터(텍스트를 원시 토큰으로 보면
      $10^4$$\sim$$10^5$차원), (b) \textbf{데이터가 아주 큰} 경우,
      (c) 특징이 \textbf{공유되는} 국소 구조(이미지)
  \end{itemize}
  \vskip0.5em
  \begin{block}{이 프로젝트의 태도}
    digits는 64차원(저차원) $\Rightarrow$ GBDT가 앞서는 쪽이
    \textit{예상}이고, 실험은 정확히 그렇게 나온다 (64차원에서
    GBDT 0.944 $\succ$ CNN 0.939, 격차 0.006).
    \kq{``왜 DL''의 정직한 답: ``신경망이 좋다''가 아니라
    ``\textbf{언제}(어떤 차원·구조에서) 신경망이 좋은지''를
    실증한다.}
  \end{block}
\end{frame}

\begin{frame}{변수 하나만 바꾼다: 8$\times$8 $\to$ 28$\times$28}
  \begin{itemize}
    \item \textbf{같은} digits 이미지를 8$\times$8에서
      \textbf{28$\times$28(784차원)으로 확대}해 다시 GBDT vs CNN
    \item 64차원: GBDT가 \textbf{0.006} 앞서던 것이
      $\Rightarrow$ 784차원: \textbf{CNN 0.961이 GBDT 0.944를
      0.017로 역전}
    \item 격차 \textbf{0.006 $\to$ $-$0.017} — 닫히고 심지어 뒤집힘
      = \kb{축 2(규모)}의 \textit{증거}
    \item 왜? 8$\times$8은 ``이미지''여도 64개 수치로 flattening되어
      \textbf{정형}이고, 28$\times$28(784개 픽셀)은
      \textbf{국소 구조를 공유하는 원시 픽셀}에 가까워지는 순간
      CNN의 \textbf{공유·평행불변 편향}이 일을 시작
  \end{itemize}
  \vskip0.5em
  \begin{block}{이 실험까지 세운 팀의 ``왜 DL'' 답변}
    ``64차원 정형에서는 GBDT 0.944가 상한이고 내 CNN(0.939)은 이를
    어깨너머로 확인했으며, 같은 데이터를 784차원으로 확대하자 CNN이
    오히려 0.961로 GBDT를 역전시켰다 — 내 문제가 차원·규모가 크다면
    (EMNIST, CIFAR) CNN의 우위는 이 추세의 연장선이라는 것을
    \textbf{숫자로} 보여준다.''
  \end{block}
\end{frame}

\begin{frame}{``왜 고전 ML이 아니라 딥러닝인가'' — 2축 + 2숫자}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \kb{(축 1) 데이터의 구조} — 정형인가, 원시(비정형)인가
      \begin{itemize}
        \item 정형(수치/범주 열) $\Rightarrow$ 기본 전제:
          ``GBDT와 비교하라''
        \item 비정형(픽셀/음파/토큰) $\Rightarrow$ ``신경망이 특징
          추출기''라는 주장이 성립할 \textit{가능}이 생김
      \end{itemize}
      \vskip0.5em
      \kb{(축 2) 데이터의 규모} — 특징이 \textbf{공유}되는가,
      크기만큼 비용이 오르는가
      \begin{itemize}
        \item Ch10.1 파라미터 수 공식이 ``왜 CNN''을 수학적으로 설명
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \kb{(숫자 1) GBDT 베이스라인} — ``신경망이 정형 상한을
      넘는가''
      \vskip0.5em
      \kb{(숫자 2) 학습 곡선의 train/val 격차} — 신경망이
      \textbf{데이터가 커질수록 val 곡선이 계속 상승}한다면
      (GBDT는 일찍 포화), ``규모가 커지면 DL에 유리하다''는
      \textbf{전망}까지 제시 가능
    \end{column}
  \end{columns}
  \vskip0.6em
  \begin{block}{네 가지를 보고서와 발표에 담으면}
    ``왜 딥러닝인가''는 막연한 의문이 아니라 \textbf{반박 가능한
    주장}이 된다 — 16.2의 peer-review가 이 주장을 \textit{심사}하는
    기준이기도 하다.
  \end{block}
\end{frame}

\begin{frame}{역사적 일화로 축을 붙잡자: AlexNet}
  \begin{itemize}
    \item 2011년까지 이미지 분류의 최선은 \textbf{사람이 손으로 만든
      특징(HOG, SIFT)} + 선형 분류자/GBDT
    \item 2012년 \kb{AlexNet}(Krizhevsky, Sutskever \& Hinton, 2012):
      \kb{ImageNet}(128만 이미지)에서 오차율을 기존보다
      \textbf{절대 10\%p 가까이 줄여} 경쟁을 사실상 종료
  \end{itemize}
  \vskip0.5em
  \begin{block}{이 전회의 본질}
    ``신경망이 마법''이 아니라 — \textbf{① 데이터(ImageNet)가 갑자기
    충분히 커지고} ② \textbf{GPU가 학습을 가능하게 한} 순간.
    \kb{축 2(규모)가 축 1(비정형)을 결정지운} 대표 사례.
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item 반대로, ``정형 표 데이터''에서는 2020년대에도 \textbf{GBDT가
      최선에 가까움}(Grinsztajn et al., 2022)
    \item 두 사례를 나란히 두면, ``왜 DL''의 답이
      \textbf{데이터의 구조와 규모}에서 나와야 하는 이유가
      직관적으로 보인다
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{손으로 한 번: CNN의 파라미터 수}
  실습 노트북의 구조(Ch10.1 공식으로 계산):
  \[
  \small 1\times8\times8
  \xrightarrow{\text{Conv2d}(1,32,3,\,\text{pad}=1,\,\text{str}=2)}
  32\times4\times4
  \xrightarrow{\text{MaxPool2d}(2)}
  32\times2\times2
  \]
  \[
  \small \xrightarrow{\text{Flatten}} 128
  \xrightarrow{\text{FC}} 256
  \xrightarrow{\text{ReLU}} 256
  \xrightarrow{\text{FC}} 10
  \]
  \small
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{tabular}{@{}llc@{}}
        \toprule
        층 & 공식 & 파라미터 수 \\
        \midrule
        Conv2d(1$\to$32, 3$\times$3) & $1\times3\times3\times32 + 32$
          & \textbf{320} \\
        MaxPool2d(2) & (파라미터 없음) & 0 \\
        FC 128$\to$256 & $128\times256 + 256$ & \textbf{33,024} \\
        FC 256$\to$10 & $256\times10 + 10$ & \textbf{2,570} \\
        \midrule
        \textbf{합계} & & \textbf{35,914} \\
        \bottomrule
      \end{tabular}
      \vskip0.4em
      {\footnotesize conv: $\text{in\_ch}\times k\times k\times
      \text{out\_ch}+\text{out\_ch}$, \; FC:
      $\text{in}\times\text{out}+\text{out}$ (Ch10.1)}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{lstlisting}
import torch
n = sum(p.numel()
        for p in model.parameters())
print(n)   # 35,914
\end{lstlisting}
      {\footnotesize PyTorch로 돌려 \textbf{정확히 일치(35,914)}
      확인 — ``10.1 공식으로 설계 정당화'' 요구사항의 뼈대가
      라이브러리 없이도 성립하는 것.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{이 숫자의 의미는 비교해야 보인다}
  \begin{itemize}
    \item 같은 8$\times$8을 \textbf{flatten(64-dim)}해서
      64$\to$64$\to$10 단순 MLP면 파라미터 \textbf{4,810}개뿐 —
      conv 층(320)보다 \textbf{작다}
    \item 즉, 작은 데이터에서는 ``깊게/넓게''가 아니라
      \textbf{입력 차원(flatten)이 작으면 GBDT}(특수한 파라미터가
      아예 없음)가 효율적이다
    \item ``왜 CNN''의 진짜 이유 = \textbf{해상도 $S$가 커질 때}
  \end{itemize}
  \vskip0.3em
  \small
  \begin{tabular}{@{}cccc@{}}
    \toprule
    해상도 $S$ & ``flatten $\to$ 256'' FC ($S^2\times256$) &
      3$\times$3 conv(1$\to$32) & 비율 \\
    \midrule
    8 & 16,384 & 320 & 51$\times$ \\
    32 & 262,144 & 320 & 819$\times$ \\
    64 & 1,048,576 & 320 & 3,277$\times$ \\
    224 & 12,845,056 & 320 & 40,141$\times$ \\
    \bottomrule
  \end{tabular}
  \vskip0.5em
  \begin{block}{``상수 vs 제곱''}
    flatten은 해상도 \textbf{제곱}($S^2$)으로 폭증하지만, conv는
    해상도와 무관하게 \textbf{320으로 일정} — 이 차이가
    ``이미지가 커질수록 CNN(파라미터 공유, 평행불변)이 필수''를
    \textbf{수학적으로} 말한다. 이 표 한 장이 발표용
    \textbf{수식적 정당화}로 그대로 쓰인다.
  \end{block}
\end{frame}

\begin{frame}{발표: 마지막 1분을 ``왜 DL''에 전념}
  \begin{itemize}
    \item 형식은 8.1과 동일(문제 정의 $\to$ 방법론 $\to$ 결과와
      정당화 $\to$ 한계 분석, 팀당 5~7분)
    \item \textbf{시간 배분을 고정}: 문제 정의(1분) $\to$ 데이터와
      정형/비정형 판정(1분) $\to$ 적용한 기법과 구조(2분) $\to$ 결과
      + GBDT 비교·수식적 정당화(2분) $\to$ \kb{``왜 DL인가''(1분)}
  \end{itemize}
  \vskip0.5em
  \begin{block}{그 1분에 채울 것}
    §``왜 고전 ML이 아니라''의 \textbf{2축 + 2숫자}.
    ``정형인데도 신경망''이라면 GBDT와의 \textbf{숫자 비교 필수}.
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item 8.1처럼 ``\textbf{잘 안 된 부분}''을 정직하게 말하는 습관
      유지 — 학습이 안 된 부분을 \textbf{Ch09}(역전파/그래디언트)·
      \textbf{9.3}(정규화)·\textbf{10.2}(과적합)의 개념으로 진단하는
      것이 ``방법론을 실제로 이해했는가''의 증거
    \item 8.1이 ``어떤 고전 ML 모델이 이 데이터에 \textbf{맞는가}''를
      물었다면, 이번엔 ``\textbf{애초에} 딥러닝이 필요한 문제인가''
      부터 스스로 판단
  \end{itemize}
\end{frame}

\begin{frame}{학습 곡선: Block B에서 x축이 ``에포크''인 이유}

  \end{itemize}

\begin{frame}{채점 기준 (rubric)}
  \small
  \begin{tabular}{@{}p{3.2cm}p{1.2cm}p{9.9cm}@{}}
    \toprule
    항목 & 비중 & 좋은 답 \\

    수식적 정당화 & 20\% & Ch09~15 공식(10.1 파라미터, 9.3 정규화,

    결과의 정직성과 반성 & 15\% & GBDT보다 못 한 부분·학습이 안 된

  \begin{block}{8.1과의 차이}
    ``방법론 적절성''이 ``이 데이터에 맞는 \textit{고전 ML 모델}인가''
   에서 ``\textbf{딥러닝이 필요한가, 그리고 그 중 왜 \textit{이}
    기법인가}''로 확장 — 성공의 기준이 ``신경망을 돌렸는가''가 아니라
    ``필요했는지를 \textbf{GBDT와 비교해 판정했는가}''이므로 25\%.
    성능이 GBDT보다 0.5\%p 낮아도 ``64차원 정형이므로 GBDT가 상한
   이고, EMNIST(784차원)로 확대해 검증했다''는 논리까지 세우면
    높은 점수.
  \end{block}
\end{frame}

\begin{frame}{자주 하는 실수 (5가지)}
  \begin{itemize}
    \item \kb{GBDT 베이스라인 없이 ``신경망 정확도 0.95'' 발표} —
      두 층의 2층이 빠짐 (Ch07.3 ``정형이면 GBDT와 비교'' 위반).
      리뷰 질문: ``같은 데이터에 GBDT를 돌리면 몇 \%인가요?
      0.95가 \textit{상대}로 얼마나 좋은지?''
    \item \kb{``신경망이 최신이라서''라는 이유} — 2축 + 2숫자가
      \textbf{전부} 빠져 있음. ``최신''은 근거가 아님 (2022
      Grinsztajn et al.: 정형에서는 GBDT가 여전히 경쟁적)
    \item \kb{에포크를 test로 정하기} — ``test 정확도가 더 높아지는
      에포크까지 돌렸다'' = Ch06.3 test-한-번 원칙 위반.
      에포크·학습률·구조는 \textbf{val 곡선만}으로
    \item \kb{flatten해서 GBDT에 비등한 것을 ``신경망이 졌다''로
      결론} — digits(64-dim) 비등은 \textit{예상된} 결과.
      ``64차원 정형''과 ``28$\times$28 원시 픽셀''을 구분하지 않고
      단순 결론내는 것 — \textbf{차원/구조의 효과를 분리}해 논의
    \item \kb{증강 없이 ``val 과적합''만 보고 dropout만 올리기} —
      과적합이 \textit{데이터 부족}(증강)인지 \textit{모델 과대}
      (dropout/L2)인지 먼저 구분 — Ch09.3의 \textbf{다른 처방}
  \end{itemize}
\end{frame}

\begin{frame}{실습: ``왜 DL''의 숫자를 코드로 검증하기}
  실습 노트북(sklearn digits만으로, CPU에서도)에서 이 절의 핵심 주장
  — ``정형·저차원에서는 GBDT가 상한이고, CNN의 우위는 \textbf{구조
  (공유)}와 \textbf{규모}에서 온다'' — 를 한 흐름으로 검증:
  \scriptsize
  \begin{enumerate}
    \item \textbf{문제 정의} — digits(1,797$\times$64), 0~9 분류,
      주 지표 정확도
    \item \textbf{3분할} — 1,437/180/180, stratify
    \item \textbf{두 층 베이스라인} — 다수 클래스(`1') val/test 0.100
    \item \textbf{GBDT 베이스라인} — 200트리: val 0.983 / test 0.944 /
      macro-F1 0.944
    \item \textbf{CNN 파라미터 공식} — §``손으로 한 번''의 표를 코드로
      재계산, torch와 \textbf{정확히 일치(35,914)} 확인
    \item \textbf{GBDT vs CNN (64차원)} — 작은 CNN test 0.939,
      GBDT(0.944)에 소폭 열세(0.006) — ``예상된 결과'' 확인
    \item \textbf{8$\times$8 $\to$ 28$\times$28 확대} — 같은 digits를
      784차원으로: GBDT 0.944 vs CNN \textbf{0.961} — 0.006 열세가
      \textbf{0.017로 역전}(축 2의 증거)
    \item \textbf{해상도 vs 파라미터} — flatten FC($S^2$)와 conv(상수)
      표 생성
    \item \textbf{학습 곡선} — GBDT $n\_estimators$ 곡선(train 50트리
      1.0 포화) + CNN \textbf{에포크} 곡선(train $\to$ 0, val은 그
      뒤로 꺾임)
    \item \textbf{선택 편향} — ``에포크·구조 후보를 test로 골랐다면''
      의 $\mathbb{E}[\max]$ 부풀림(8.1과 동일, Ch06.3)
  \end{enumerate}
  \vskip0.4em
  {\footnotesize 이 흐름이 보고서 6절로, 다시 16.2의 체크리스트
  (``왜 DL'' 추가 항목 포함)로 매핑 — \textbf{세 가지 언어(코드,
  보고서, 리뷰)}로 같은 절차를 말할 수 있어야 끝.}
\end{frame}

\begin{frame}{확인 문제 1~2}
  \textbf{1.} CIFAR-10(60,000개 32$\times$32) CNN으로 test 0.82를
  내고 ``신경망이 GBDT보다 좋다''고 결론낸 팀 — 이 결론이 성립하려면
  \textbf{먼저 확인해야 할 숫자 하나}와 이유?
  \vskip0.3em
  \small
  \begin{block}{답}
    \textbf{같은 flatten(3$\times$32$\times$32=3,072-dim) 데이터에
    GBDT를 돌려 얻은 test 정확도}. ``좋다''는 것은 GBDT보다
    \textit{상대}로 좋은지를 보여줘야 성립(2층 베이스라인).
    CIFAR 32$\times$32를 flatten하면 3,072차원(64차원 digits보다
    훨씬 고차원)이라 GBDT가 CNN에 육박할 수도 있음 — 그 비교 없이는
    ``신경망이 좋다''는 1층(무작위)만 이긴 것에 그칠 수 있다.
  \end{block}
  \vskip0.3em
  \textbf{2.} Ch10.1 공식으로, 입력 3$\times$32$\times$32에
  \texttt{Conv2d(3,64,3, padding=1)} 한 층의 파라미터 수를 계산하고,
  ``flatten FC와 달리 해상도와 무관하다''는 것을 $S{=}32$에서
  flatten($\to$256)과 비교해 보여라.
  \vskip0.3em
  \begin{block}{답}
    conv: $3\times3\times3\times64 + 64 = 1{,}792$ (해상도 32와
    무관). flatten FC(3,072$\to$256): $3{,}072\times256 + 256 =
    786{,}688$ — conv의 \textbf{약 439배}. 해상도 64가 되면 FC는
    $12{,}288\times256$으로 4배가 되는데 conv는 그대로 1,792 —
    \textbf{상수 vs 제곱}이 ``커질수록 CNN''의 근거.
  \end{block}
\end{frame}

\begin{frame}{확인 문제 3~4}
  \textbf{3.} train 손실 0.01, val 손실 0.40인 에포크까지
  ``\textbf{test가 더 좋아질 때까지} 돌렸다''고 보고 — Ch09.3 과적합
  과 Ch06.3 test-한-번 원칙을 들어 깨진 원칙 \textbf{두 가지}와
  올바른 방법?
  \vskip0.3em
  \small
  \begin{block}{답}
    (1) train/val 격차 0.39 = \textbf{과적합}(Ch09.3) — train이 0으로
    가는데 val이 오르는 구간. (2) ``test가 좋아질 때까지'' =
    \textbf{test를 튜닝에 쓰는} 선택 편향(Ch06.3) — test는
    \textit{한 번}만. 올바르게: \textbf{val 손실의 최솟값(또는
    plateau) 에포크}를 정한 뒤, 그 에포크로 \textbf{test를 딱 한 번}.
  \end{block}
  \vskip0.3em
  \textbf{4.} ``우리 팀은 정형 표 데이터(5,000행$\times$12열)인데 LLM을
  fine-tune했다'' — 2축(구조·규모)과 Grinsztajn et al.\ 2022 결과를
  근거로 \textbf{반박 가능한 질문 하나}.
  \vskip0.3em
  \begin{block}{답}
    축 1(구조)이 정형이고 축 2(규모)도 12열로 작아 GBDT가
    \textbf{경쟁적/우위}인 영역(2022). 질문: ``12열·5,000행 정형에서
    LLM fine-tune의 test 성능이 같은 데이터의 GBDT(Ch07.3)와
    비교해서 \textit{절대적으로} 얼마나 더 좋나요? 그 차이가
    ``특징의 국소 공유'' 같은 \textit{구조적} 이득이 아니라 단순히
    용량 때문이라면, 정형이므로 GBDT가 상한인데 왜 LLM을 선택했
    냐요?'' — ``최신이라서''로 답할 수 없는, Ch07(GBDT)를 근거로
    둔 질문.
  \end{block}
\end{frame}

\begin{frame}{자주 묻는 질문}
  \small
  \begin{itemize}
    \item \textbf{Q. 정형 데이터에 굳이 신경망을 써도 되나요?} \;
      된다 — 다만 정형에서는 지금도 GBDT 계열이 경쟁적인 경우가 흔함.
      선택했다면 ``왜 GBDT보다 이 문제에 더 적합하다고 판단했는지''
      (복잡한 상호작용 자동 학습, 다른 DL 모듈과 결합 계획 등)를
      발표에서 설명할 수 있어야 — ``최신이라서''는 근거가 아님.
      GBDT 비교 숫자가 그 \textit{증거}가 되어야 한다.
    \item \textbf{Q. Ch09~15 중 여러 기법을 동시에 써도 되나요?} \;
      \textbf{권장} — 예: CNN으로 이미지 특징 추출 $\to$ 어텐션으로
      조합. ``왜 이 조합인가''의 추가 설계 근거 기회. 단,
      \textbf{수식적 정당화 1개}는 그중 하나의 기법만 충족하면 됨.
    \item \textbf{Q. GPU가 없는데 CNN을 돌릴 수 있나요?} \; \textbf{되
      다} — 기본 코스(digits 8$\times$8, EMNIST 28$\times$28,
      CIFAR 32$\times$32)는 Colab 무료 CPU/T4로도 4주 안에 충분.
      작은 CNN(35,914)은 CPU에서 수 분 만에 수렴. 대규모
      (ResNet, BERT full fine-tune)가 필요하면 HF \textbf{pretrained}
      모델 transfer + 마지막 층만 학습 — 정당화는 ``왜 이 pretrained
      구조인가''(Ch13 사전학습)로 변경.
    \item \textbf{Q. ``왜 DL'' 질문에 GBDT가 이기면(신경망이 못
      이기면) 점수가 떨어지나요?} \; \textbf{떨어지지 않음} — 오히려
      \textbf{정직한 결과}. 64차원 정형에서 GBDT와 비등은 \textit{예상
      }이며, ``그래서 내 문제는 차원이 더 큰(784-dim) EMNIST여야 CNN
      우위를 검증할 수 있다''로 연결하면 \textbf{높은} 점수.
      감점은 GBDT를 돌려놓고도 ``신경망 0.94''만 보고 \textit{단순
      결론}내는 경우.
  \end{itemize}
\end{frame}

        = 이미지 파일''인 경우) 그 가정은 \textbf{아무 가치}를 주지
        못함
      \item \kb{리뷰 질문 예}: ``여러분의 이미지에서 \textit{위치}가
        실제로 정보인가요 — 이미지를 \textbf{무작위로 회전·이동}
        시켰을 때 val 정확도가 얼마나 변하나요?'' — 발표팀이
        \textbf{새 실험}(증강 효과를 측정하는 run)을 해야 답할
        수 있으므로 8.2 기준대로 \textbf{3점}
    \end{itemize}
  \end{block}
  \vskip0.3em
  {\footnotesize 8.2의 질문 등급 기준은 그대로: ``새 실험이 필요
  한 질문'' = 3점, ``말 몇 마디로 넘길 질문'' = 2점.}
\end{frame}

\begin{frame}{에폭과 교정: Block B의 두 시간 축}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \kb{(2) 학습의 시간 차원 — 에폭}
      \begin{itemize}
        \item 고전 ML은 \texttt{fit()} \textbf{한 번}으로 끝나지만,
          신경망은 에폭을 돌리는 \textbf{과정}이 모델의 일부
        \item 8.1 M3(하이퍼파라미터는 \textbf{val 곡선}으로만)가
          신경망에서는 \textbf{에폭 수를 정하는 것}까지 포함
        \item ``에폭 300을 돌렸다''는 \textbf{전제}(val 곡선이 300
          부근에서 포화)를 만족할 때만 성립
        \item \kb{질문 예}: ``에폭 300인데 val 곡선은 언제 정점?
          정점 이후 100개 에폭이 val에서 무엇을 바꿨나요?'' —
          답하려면 \textbf{학습 곡선(M3)}을 다시 꺼내 봐야 함
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \kb{(3) 확률의 신뢰도 — 교정(calibration)}
      \begin{itemize}
        \item ``정확도 95\%''를 넘겨도 ``\textbf{0.99 확률}로 예측한
          건 중 \textbf{99\%가 정말 맞았는가}''는 \textbf{별개}의
          질문
        \item 교차엔트로피가 낮아도 확률 \textbf{스케일 전체가 왜곡}
          된 모델(0.99를 말해도 실제로 0.87)은 ``\textbf{위험도 기반
          결정}''(의료, 사기)에서 정확도 결정과 \textbf{완전히
          다른} 결정
        \item Block A(로지스틱회귀)는 교정이 \textit{구조적으로}
          좋아 잘 안 나왔지만, \textbf{신경망+드롭아웃}은 \textbf{
          드롭아웃 ON 상태에서 확률을 뽑으면 교정이 악화}(Guo et
          al., 2017)
        \item ``\textbf{드롭아웃 off로 확률을 뽑았나요?}'' = 합
          리적인 3점 질문
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{자주 하는 실수: ``정확도가 높으니 좋은 모델''}
  \begin{itemize}
    \item 딥러닝에서 특히 잦은 부실한 리뷰: ``\textbf{test 정확도
      95\%라 정말 잘 만들었다}'' — 숫자 \textbf{하나}만으로 판단
    \item 그 정확도가 Ch06.3 원칙(train/val/test 분리, \textbf{
      누수 방지})을 지키고 나온 숫자인지, 불균형이면 Ch2.3
      \textbf{PR-AUC}를 함께 확인했는지를 먼저 따지는 것이 건설적
    \item 예: ``\textbf{검증셋·테스트셋을 나눈 기준}을 설명해
      주시겠어요?'' — 다시 확인해봐야 답할 수 있는 질문
  \end{itemize}
  \vskip0.5em
  \begin{block}{Block B에서 특히 위험한 이유}
    ``좋은 정확도''가 \textbf{우연일 가능성}이 Block A보다
    \textbf{훨씬 높음}. Ch6.3 선택 편향 — 후보 $m$개 중 max를 고르면
    $\mathbb{E}[\max]$만큼 \textbf{부풀어 오름} — $m$이 클수록
    커진다.
  \end{block}
  \vskip0.3em
  \small
  \begin{tabular}{@{}p{8.3cm}p{5.6cm}@{}}
    \toprule
    후보 공간(Block B) & $\sigma=0.02$일 때 부풀림(8.1 표) \\
    \midrule
    모델 2~3개 $\times$(에폭$\times$드롭아웃$\times$학습률) 10~30개
      $\Rightarrow m = 30\sim90$ & \textbf{0.045$\sim$0.052} 부근 \\
    \bottomrule
  \end{tabular}
  \vskip0.4em
  \kq{``test 정확도 0.95''가 실제로 0.90대 초반일 수 있다 —
  8.2의 ``0.1\%p가 잡음 수준'' 논리가 그대로 적용되되, Block B
  에서는 그 \textbf{잡음이 더 크고, 후보가 더 많다.}}
\end{frame}

\begin{frame}{실습 노트북이 정량화하는 ``잡음의 두 기원''}
  \begin{itemize}
    \item 실습 노트북(§4)이 이 잡음의 \textbf{두 기원}을 \textbf{동시에}
      정량화:
    \item ① \textbf{표본 잡음}: \textit{같은 모델}의 test 100건만
      바꿔도 정확도 범위 \textbf{약 0.06}
    \item ② \textbf{선택 편향}: 후보 $m$개 중 최고를 고르는
      $\mathbb{E}[\max]$가 \textbf{약 0.04$\sim$0.05}
  \end{itemize}
  \vskip0.5em
  \begin{block}{리뷰어의 한 줄 판정}
    \textbf{모델 간 정확도 차이가 이 두 숫자보다 작다면, 그 차이는
    모델이 아니라 표본·선택에서 나온 것이다.}
  \end{block}
  \vskip0.5em
  \begin{itemize}
    \item 이 프레임이 16.2 실습의 ``3점 질문 감각''을 만든다 —
      발표를 들으며 ``이 숫자가 나오면 3점 질문을 쓸 수 있다''는
      기준이 생김
    \item §``자주 하는 실수''의 $\mathbb{E}[\max]$ 계산($m \approx
      30$, $\sigma=0.02$)과 서로 \textbf{일치} — 후보가 30~90개면
      1\%p 차이는 원리적으로 잡음 범위
  \end{itemize}
\end{frame}

\begin{frame}{손으로 한 번: 가상의 딥러닝 발표}
  8.2의 ``가상 발표''와 동일한 구조로 Block B 발표 하나를 놓고
  체크리스트를 적용.
  \vskip0.4em
  \begin{alertblock}{발표 요약}
    \small ``우리 팀은 breast\_cancer 데이터(569건 $\times$ 30 특성,
    악성 63\% : 양성 37\%)로 악성 여부 분류 모델을 만들었다. 30차원
    특징을 \textbf{64개 히든 뉴런의 1층 신경망}에 입력하고,
    \textbf{dropout 0.5}를 넣어서 \textbf{50에폭} 학습했다.
    \textbf{test 정확도 0.95}가 나왔다 — 로지스틱회귀(0.94)와
    GBDT(0.94)보다 높다. 이미지/텍스트는 아니지만, 신경망이 특징의
    \textbf{복잡한 상호작용을 자동으로 학습}할 수 있어서 고른
    것이다.''
  \end{alertblock}
  \vskip0.4em
  \begin{itemize}
    \item 체크리스트 5개 항목을 하나씩 적용한다 (다음 프레임)
    \item 이 발표의 함정: \textbf{정형 표 데이터}에 신경망,
      비교 대상과 \textbf{1\%p 차이}, 수식적 정당화 ``숫자만''
  \end{itemize}
\end{frame}

    }(후보 $m$개 중 max를 고르는 부풀림)과 같은 크기인지 — 즉, 이
    1\%p 차이가 ``신경망이 더 좋다''는 결론을 뒷받침할 만큼 큰

  \end{block}

  \vskip0.4em

    GBDT랑 비교해 보셨으면 좋겠어요.'' — \textbf{관찰은 있지만

  \end{block}

\begin{frame}{확인 문제 1~2}
  \textbf{1.} breast\_cancer(569$\times$30)에 MLP로 test 0.95,
  ``로지스틱회귀(0.94)보다 높으니 MLP가 더 좋은 모델이다'' —
  Ch06.3 선택 편향과 Ch07.3 GBDT 비교를 근거로 \textbf{반박 가능한
  질문 하나}.
  \vskip0.3em
  \small
  \begin{block}{답}
    ``GBDT(Ch07.3)를 \textbf{같은 val 분할}에서 \textbf{F1}로
    비교했나요? 0.95 vs 0.94의 1\%p 차이는 Ch06.3 선택 편향
    부풀림(후보 $m$개 중 max, $m\approx30$일 때 $\approx0.04$)
    보다 \textbf{작거나 같은 크기}일 수 있습니다 — 이 차이가
    결론을 뒷받침할 만큼 큰 근거인지 val F1과 함께 \textbf{
    정량적으로} 보여주세요.''
  \end{block}
  \vskip0.3em
  \textbf{2.} ``CNN을 100에폭 학습했다'' — val 정확도가
  \textbf{에폭 40에서 정점(0.93)}에 도달한 뒤 \textbf{0.92}
  까지 내려옴. ``100에폭'' 결론의 \textbf{한계}와 \textbf{
  질문}?
  \vskip0.3em
  \begin{block}{답}
    val 곡선이 에폭 40에서 \textbf{꺾였으므로}, 100에폭 중 60개
    는 \textit{val을 떨어뜨린} 과적합 구간(Ch09.3 신호). ``100
    에폭'' 주장이 성립하려면 ``왜 val이 더 나빠졌는지''를 Ch09.3
    개념(정규화·dropout 부족? 학습률?)으로 \textbf{설명}해야
    함. 질문: ``val 정점(40) 이후 60개 에폭이 val을 0.93$\to$
    0.92로 \textit{내려}보냈는데, 이 하락을 진단해줄 수 있나요?
    에폭 40에서 멈췄다면 0.93이었을 텐데, 왜 0.92를 발표 수치로
    쓰셨나요?''
  \end{block}
\end{frame}

\begin{frame}{확인 문제 3~4}
  \small
  \textbf{3.} ``텍스트에 RNN'' — 시퀀스를 \textbf{무작위로
  70/15/15} 분할. Ch06.3의 어떤 원칙이 깨졌고, 왜 test가 낙관
  적인가?
  \begin{block}{답}
    ``시계열/시퀀스는 \textbf{시간순(또는 그룹 단위)}으로 나눠야
    한다''. 무작위 분할이면 같은 시퀀스 앞뒤가 train/test에
    섞여 RNN이 \textit{학습한 문맥의 잔해}를 보고 ``예측'' —
    \textbf{그룹 단위 누수}. test가 ``새 시퀀스에 잘 일반화하는
    가''가 아니라 ``\textbf{같은 시퀀스의 나머지를 얼마나 외웠
    는가}''를 재게 되어 낙관적.
  \end{block}
  \vskip0.2em
  \textbf{4.} ``데이터가 이미지라서 CNN을 썼다''가 \textbf{형식
  (2점)}에 머무는 이유, 3점으로 \textbf{바꾸는 법}?
  \begin{block}{답}
    ``이미지라서 CNN''은 Ch10.1의 \textit{왜}를 설명하지 않음 —
    국소 receptive field + 파라미터 공유가 \textit{어떤 구조적
    특성}(인접 픽셀 상관)에 맞는지 말해야 ``구조-편향 매칭''이
    성립. 3점: ``\textit{위치}가 실제로 정보인가요 — \textbf{
    무작위 회전·이동} 시키면 val이 얼마나 변하나요?''
  \end{block}
\end{frame}

\begin{frame}{확인 문제 5}
  \small
  \textbf{5.} ``dropout 0.5를 넣었더니 val F1 0.88$\to$0.91'' —
  이 설명이 \textbf{타당한 조건}?
  \begin{block}{답}
    ``정규화했으니까 올랐다''(나쁜 설명)로는 안 되고,
    \textbf{train/val 격차가 줄었는지}를 함께 보여줘야 —
    ``train F1 0.99/val 0.88 $\to$ 0.5 넣자 0.95/0.91로
    \textbf{격차가 줄었다}''고 말해야 Ch09.3 ``정규화가
    \textbf{분산}을 줄인다''가 관측과 일치. 격차가 \textit{안
    줄고} val만 올랐다면 그 0.03은 \textbf{잡음 범위}일 수
    있음을 스스로 인정.
  \end{block}
\end{frame}

  \begin{itemize}
    \item \textbf{Q. Ch8의 peer-review와 뭐가 다른가요?} \;

      \textbf{딥러닝이 정말 필요했는가}''까지 함께 판단.
    \item \textbf{Q. 정형 데이터에 DL을 쓴 팀에게 ``GBDT로 다시
      돌려라''는 질문이 가혹하지 않나요?} \; \textbf{가혹하지
      않음} — 16.1 FAQ가 발표팀에게 ``GBDT와 비교했을 때의 근거를
      설명하라''고 요구했으므로, 리뷰어가 ``그 비교가 실제로
      있는지''를 묻는 것은 \textbf{16.1 요구사항을 집행}하는 것
      에 불과. 비교 \textit{결과}가 GBDT가 이기더라도(정형에서는
      흔함, Ch07.3) 발표가 ``실패''하는 것은 아님 — ``GBDT가
      이기긴 했지만, 다른 DL 모듈(예: Ch12 어텐션)과 \textbf{결합
      }할 계획이 있어 신경망을 고랐다''도 \textit{충분한} 근거.
      리뷰어가 판정할 것은 ``\textbf{비교를 했는가, 근거가 개념
      에 뿌리를 박는가}''이지 ``비교에서 신경망이 이겼는가''가
      아님.
    \item \textbf{Q. 학습 곡선이 val도 계속 상승 중(정점 도달 전)
      이면 어떻게 리뷰하나요?} \; ``\textbf{에폭 $N$이 정점이 아
      니다}''를 지적 — val이 $N$에서 여전히 상승 중이면 ``$N$이
      최적''이 아니라 ``$N$이 \textbf{관찰된 범위 내 최선}''일
      뿐, ``$2N$까지 돌리면 더 좋아질 것''이 성립할 수 있음. 질문:
      ``에폭 \textbf{$2N$}까지 돌린 val 곡선을 보여줄 수 있나요?''
      — \textbf{새 실험}이 필요하므로 3점.
    \item \textbf{Q. 이미지인데 GBDT만 쓴 팀(= DL을 \textit{안} 쓴
      팀)은 어떻게 채점하나요?} \; 16.1 ``적용'' 요구사항
      (Ch09~15 중 \textbf{최소 하나의 DL 기법})을 \textbf{미충
      족} — ``왜 DL인가'' 항목은 ``\textbf{해당 없음(요구사항
      미충족)}''으로 채점하고 사유를 리뷰에 명시. ``왜 DL을
      \textit{안} 썼는가''를 리뷰어가 판단하는 것이 아니라,
      \textbf{요구사항 충족 여부}가 이 항목의 범위.
    \item \textbf{Q. Block B의 프로젝트는 Block A보다 규모가 크면
      안 되나요 — 같은 수준이면 되나요?} \; \textbf{같은 수준이
      면 된다} — 요구사항(DL 기법 최소 1개 + 수식적 정당화 최소
      1개)이 Block A와 \textit{구조적으로} 동일. 유일한 차이:
      ``\textbf{딥러닝 특유의} 하이퍼파라미터(에폭, dropout,
      학습률 스케줄)를 \textbf{val 곡선}으로 정했는가''를 추가로
      확인(위 §(2)(3)).
  \end{itemize}
\end{frame}

\begin{frame}{복습 문제 1~2: 장을 넘나드는 연결}
  \textbf{1.} Ch03의 \kb{GDA}와 Ch15의 \kb{GMM}은 둘 다
  ``\textbf{정규분포를 가정한 생성적 모델}'' — 가장 \textbf{핵
  심적인 차이}?
  \vskip0.3em
  \small
  \begin{block}{답}
    GDA는 클래스 라벨 $y$가 \textbf{관측되는}(지도) 상황에서 각
    클래스의 분포를 추정하지만, GMM은 클러스터 소속($z$)이
    \textbf{관측되지 않는}(비지도) 상황에서 \textbf{EM}으로 그
    \textbf{잠재변수까지 함께} 추정해야 한다.
  \end{block}
  \vskip0.3em
  \textbf{2.} Ch07의 \kb{트리}와 Ch10의 \kb{CNN}은 둘 다
  ``\textbf{국소적인 정보}를 활용하는 \textbf{귀납적 편향}'' —
  각각 \textbf{어떤 방식}으로 구현?
  \vskip0.3em
  \begin{block}{답}
    트리는 \textbf{한 번에 특징 하나씩}만 보고 분기해서
    \textbf{저차원적 규칙을 조합}하고, CNN은 \textbf{작은 필터}
    가 이미지의 \textbf{국소 영역}만 보고 그 필터를 \textbf{전체
    이미지에 재사용}(\textbf{파라미터 공유}) — 형태는 다르지만
    ``\textbf{전체를 한 번에 보지 않고 지역적으로 나눠서 본다}''
    는 방향은 \textbf{같다}.
  \end{block}
\end{frame}

\begin{frame}{복습 문제 3~4: 같은 해법의 재발견, 같은 연산}
  \textbf{3.} Ch09의 \textbf{그래디언트 소실}이 Ch11(\kb{RNN})
  과 Ch10(\kb{ResNet})에서 각각 \textbf{어떤 해법}으로 재등장?
  \vskip0.3em
  \small
  \begin{block}{답}
    ResNet은 \textbf{스킵 연결}($y=F(x)+x$)로 그래디언트가
    ``+1'' 항을 통해 \textbf{그대로 전달}될 경로를 만들고,
    \textbf{LSTM/GRU}(Ch11.3)도 \textbf{게이트}를 통해 정보가
    \textbf{곱셈 대신 덧셈}으로 흐르는 경로를 만듦 — 서로 다른
    구조(이미지 vs 시퀀스)에서 \textbf{같은 해법}(\textbf{곱
    승 사슬에 덧셈 지름길을 낸다})이 독립적으로 재발견된 것.
  \end{block}
  \vskip0.3em
  \textbf{4.} Ch12의 \kb{어텐션}과 Ch15의 \kb{VAE}은 둘 다
  ``\textbf{가중합}''을 핵심으로 — 각각 \textbf{무엇의 가중합}?
  \vskip0.3em
  \begin{block}{답}
    어텐션은 \textbf{softmax로 정규화된 관련도}(가중치)로 각
    단어의 \textbf{Value 벡터들}의 가중합 — ``관련 있는 정보를
    섞은'' 새 벡터. VAE의 \textbf{ELBO}는 (가중합은 아니지만)
    ``\textbf{복원 항}''과 ``\textbf{KL 정규화 항}''이라는
    \textbf{서로 다른 두 목표의 균형}을 하나의 손실로 \textbf{
    합친다}는 점에서, ``\textbf{여러 신호를 하나로 종합한다}''
    는 더 넓은 패턴을 공유.
  \end{block}
\end{frame}

    ``0.97''이 실제로는 \textbf{0.94 부근}일 수 있음. 질문:
    ``8개 조합 중 \textbf{val F1로} 고른 것과 \textbf{test F1
    로} 고른 것의 차이는 얼마인가요? val-test 격차가 0.02 미만
    이면 그 격차는 선택 편향 크기보다 \textbf{작아} — test로
    고르지 않았다는 \textbf{근거가 없습니다}.''
  \end{block}
  \vskip0.3em
  \textbf{6.} 실습 §8에서 ``\textbf{누수된 test}''의 정확도
  \textbf{1.000}, 실제 test \textbf{0.947}(차이 \textbf{+0.053}) —
  모델과 하이퍼파라미터를 \textbf{하나도 바꾸지 않고} 이 차이가
  가능한가? 리뷰어는 무엇을 물어보는가?
  \vskip0.3em
  \begin{block}{답}
    \textbf{가능} — \textbf{분할 방식 하나}가 바뀌면 됨.
    train에 이미 있는 \textbf{같은 단위}(환자/행)를 test에 넣는
    \textbf{그룹 누수}(Ch06.3). 리뷰 질문은 \textbf{판단이
    아니라 기계적 점검}: ``train과 test에 \textbf{같은 환자가
    동시에} 들어갔나요? \texttt{GroupKFold}로 재분할하면 test
    F1이 어떻게 변하나요?'' — \textbf{분할 코드 한 줄}을 보면
    즉시 판정.
  \end{block}
\end{frame}

\begin{frame}{복습 문제 7: ``최신이라서''의 등급}

  \end{block}

\begin{frame}{ML2로 가기 전에}
  \begin{itemize}
    \item ML1 = \textbf{``정답이 있는 데이터로부터 배운다}'' —
      \textbf{지도학습·비지도학습}의 세계
    \item ML2는 \textbf{완전히 다른 질문}에서 출발 — 정답 대신
      \textbf{\textbf{보상}}만 주어지고, 데이터가 \textbf{미리
      있는 게 아니라 에이전트가 스스로 행동하며 만들어지는}
      \textbf{\textbf{강화학습}}의 세계
    \item 무대는 텍스트나 이미지가 아니라, \textbf{물리 법칙이
      지배하는 \textbf{로봇 시뮬레이션 환경}}(Sutton \& Barto,
      2018)
  \end{itemize}
  \vskip0.5em
  \begin{block}{``ML1을 안 듣고 ML2를 바로 들어도 되나요?''}
    \textbf{무방} — ML2 1주차가 이번 학기에 필요한 \textbf{신경
    망 기초를 압축해서} 다시 짚어줌. 다만 \textbf{Ch09(역전
    파)}과 \textbf{Ch06(정규화, 특히 교차검증)}의 감각이 남아
    있다면 ML2를 \textbf{훨씬 수월하게} 따라갈 수 있음.
  \end{block}
\end{frame}

\begin{frame}{같은 구조, 다른 질문 — 한 학기가 남기는 선물}
  \begin{block}{4단계는 \textbf{그대로} 살아 있다}
    ① 모델 정의 $\to$ ② 손실 $\to$ ③ $\theta$ 조정 $\to$ ④
    통제는 ML2에서도 \textbf{그대로} — 바뀌는 것은 \textbf{
    ①의 ``모델''}(\textbf{정책 $\pi(a|s)$})과 \textbf{④의
    ``데이터''}(\textbf{에이전트의 경험})\textbf{뿐}.
  \end{block}
  \vskip0.4em
  \kq{\textbf{같은 구조, 다른 질문} — 이번 학기가 ML2에 남기는
  \textbf{정확한 선물}. \; 어떤 모델을 쓸지보다 \textbf{먼저}
  ``\textbf{무엇을 데이터로 갖고, 무엇을 맞히고 싶은가}.''}
  \vskip0.5em
  \small
  \begin{itemize}
    \item \textbf{Q. 기말고사 대비 — 어디서부터?} \; 개념 연결
      4문항 + 절차 3문항을 그 순서로. \textbf{6번(누수 질문)}을
      노트북 없이 답하면 준비 끝.
    \item \textbf{Q. Ch02~15가 섞여 있는데 기말은 Block B
      만?} \; 복습 문제가 \textbf{의도적으로} 두 블록을 넘나
      듦 — ``\textbf{연결을 아는가}''가 ``개별 장을 외우는 것
      ''보다 시험·ML2에서 훨씬 많이 쓰임.
    \item \textbf{Q. ML1의 수식 유도를 전부 외워야?} \;
      \textbf{아니오} — 4단계 구조의 \textbf{감각}이 자동인 것
      이 목표. \textbf{유도}(GDA$\leftrightarrow$로지스틱,
      ELBO)는 필요할 때 장에서 \textbf{찾아보는 것}이 정상,
      ``\textbf{손실을 뭐로 나타내지?}''가 \textbf{자동 반사}
      여야.
    \item \textbf{Q. ML2의 프로젝트는 뭐가 다른가요?} \; 뼈
      대(발표+리뷰+채점)는 \textbf{동일}. ``\textbf{결과의
      정직성}''의 \textbf{초점}만 바뀜 — RL에서 ``학습이 안
      된 부분''은 실패가 아니라 \textbf{데이터} — 왜 안 됐
      는지 Ch09~15의 개념으로 진단하는 것이 \textbf{좋은
      보고서}.
  \end{itemize}
\end{frame}
